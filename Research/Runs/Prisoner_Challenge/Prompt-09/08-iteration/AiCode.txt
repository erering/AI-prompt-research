#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    int L = 0, p = 1;
    vector<int> pw(1, 1);
    while (p <= N) {
        p *= 3;
        pw.push_back(p);
        ++L;
    }

    vector<vector<int>> s(1 + 3 * L, vector<int>(N + 1, -1));

    auto digit = [&](int v, int pos) -> int {
        return (v / pw[pos]) % 3;
    };

    if (L == 0) {
        s[0][0] = 0;
        for (int j = 1; j <= N; ++j) s[0][j] = -1;
        return s;
    }

    auto id = [&](int level, int d) -> int {
        return 1 + level * 3 + d;
    };

    s[0][0] = 0;
    int top_pos = L - 1;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = id(0, digit(j, top_pos));
    }

    for (int level = 0; level < L; ++level) {
        int pos = L - 1 - level;
        bool inspectB = (level % 2 == 0);

        for (int d = 0; d < 3; ++d) {
            int row = id(level, d);
            s[row][0] = inspectB ? 1 : 0;

            for (int j = 1; j <= N; ++j) {
                int e = digit(j, pos);

                if (inspectB) {
                    if (e > d) {
                        s[row][j] = -1;
                    } else if (e < d) {
                        s[row][j] = -2;
                    } else {
                        if (level == L - 1) {
                            s[row][j] = -1;
                        } else {
                            int next_pos = pos - 1;
                            s[row][j] = id(level + 1, digit(j, next_pos));
                        }
                    }
                } else {
                    if (e < d) {
                        s[row][j] = -1;
                    } else if (e > d) {
                        s[row][j] = -2;
                    } else {
                        if (level == L - 1) {
                            s[row][j] = -1;
                        } else {
                            int next_pos = pos - 1;
                            s[row][j] = id(level + 1, digit(j, next_pos));
                        }
                    }
                }
            }
        }
    }

    return s;
}
