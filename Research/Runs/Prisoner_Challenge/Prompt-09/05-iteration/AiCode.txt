#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    if (N <= 1) {
        vector<vector<int>> s(1, vector<int>(N + 1, -1));
        s[0][0] = 0;
        return s;
    }

    const int INF = 1e9;
    vector<int> dp(N + 1, INF), prv(N + 1, -1), prvr(N + 1, -1);
    dp[1] = 0;

    for (int p = 1; p < N; ++p) {
        if (dp[p] == INF) continue;
        for (int r = 2; r <= N; ++r) {
            int np;
            if ((long long)p * r >= N) np = N;
            else np = p * r;
            if (dp[p] + r < dp[np]) {
                dp[np] = dp[p] + r;
                prv[np] = p;
                prvr[np] = r;
            }
        }
    }

    vector<int> rad;
    int cur = N;
    while (cur > 1) {
        rad.push_back(prvr[cur]);
        cur = prv[cur];
    }
    reverse(rad.begin(), rad.end());

    int D = (int)rad.size();
    vector<int> place(D, 1);
    for (int i = D - 2; i >= 0; --i) {
        place[i] = place[i + 1] * rad[i + 1];
    }

    auto digit = [&](int value, int pos) -> int {
        int v = value - 1;
        return (v / place[pos]) % rad[pos];
    };

    vector<vector<int>> id(D);
    int total = 0;
    for (int p = 0; p < D; ++p) {
        id[p].resize(rad[p]);
        for (int d = 0; d < rad[p]; ++d) {
            id[p][d] = ++total;
        }
    }

    vector<vector<int>> s(total + 1, vector<int>(N + 1, -1));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = id[0][digit(j, 0)];
    }

    for (int p = 0; p < D; ++p) {
        for (int d = 0; d < rad[p]; ++d) {
            int st = id[p][d];
            s[st][0] = (p % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; ++j) {
                int od = digit(j, p);

                if (p % 2 == 0) {
                    if (d < od) {
                        s[st][j] = -1;
                    } else if (d > od) {
                        s[st][j] = -2;
                    } else {
                        if (p + 1 == D) s[st][j] = -1;
                        else s[st][j] = id[p + 1][digit(j, p + 1)];
                    }
                } else {
                    if (od < d) {
                        s[st][j] = -1;
                    } else if (od > d) {
                        s[st][j] = -2;
                    } else {
                        if (p + 1 == D) s[st][j] = -1;
                        else s[st][j] = id[p + 1][digit(j, p + 1)];
                    }
                }
            }
        }
    }

    return s;
}
