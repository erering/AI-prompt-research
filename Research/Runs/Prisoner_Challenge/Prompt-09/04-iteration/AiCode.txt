#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    vector<int> base = {5, 4, 4, 4, 4, 4};
    int L = (int)base.size();

    vector<int> mult(L, 1);
    for (int i = L - 2; i >= 0; --i) mult[i] = mult[i + 1] * base[i + 1];

    vector<int> astate(L);
    for (int i = 0; i < L; ++i) astate[i] = i;

    vector<vector<int>> bstate(L);
    int idx = L;
    for (int i = 0; i < L; ++i) {
        bstate[i].resize(base[i]);
        for (int d = 0; d < base[i]; ++d) bstate[i][d] = idx++;
    }

    vector<vector<int>> s(idx, vector<int>(N + 1, -1));

    auto digit = [&](int val, int level) {
        return (val / mult[level]) % base[level];
    };

    for (int level = 0; level < L; ++level) {
        int st = astate[level];
        s[st][0] = 0;
        for (int j = 1; j <= N; ++j) {
            int d = digit(j - 1, level);
            s[st][j] = bstate[level][d];
        }

        for (int d = 0; d < base[level]; ++d) {
            int bst = bstate[level][d];
            s[bst][0] = 1;
            for (int j = 1; j <= N; ++j) {
                int e = digit(j - 1, level);
                if (e > d) s[bst][j] = -1;
                else if (e < d) s[bst][j] = -2;
                else {
                    if (level + 1 < L) s[bst][j] = astate[level + 1];
                    else s[bst][j] = -1;
                }
            }
        }
    }

    return s;
}
