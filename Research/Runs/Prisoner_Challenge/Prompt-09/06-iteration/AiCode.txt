#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int L = 8;
    const int R = 3;
    vector<int> pow3(L, 1);
    for (int i = L - 2; i >= 0; --i) pow3[i] = pow3[i + 1] * R;

    auto digit = [&](int val, int pos) -> int {
        return (val / pow3[pos]) % R;
    };

    auto id = [&](int pos, int d) -> int {
        return 1 + pos * R + d;
    };

    int X = L * R;
    vector<vector<int>> s(X + 1, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        int v = j - 1;
        s[0][j] = id(0, digit(v, 0));
    }

    for (int pos = 0; pos < L; ++pos) {
        for (int d = 0; d < R; ++d) {
            int st = id(pos, d);
            s[st][0] = (pos % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; ++j) {
                int v = j - 1;
                int od = digit(v, pos);

                if (pos % 2 == 0) {
                    if (d < od) {
                        s[st][j] = -1;
                    } else if (d > od) {
                        s[st][j] = -2;
                    } else {
                        if (pos + 1 == L) s[st][j] = st;
                        else s[st][j] = id(pos + 1, digit(v, pos + 1));
                    }
                } else {
                    if (od < d) {
                        s[st][j] = -1;
                    } else if (od > d) {
                        s[st][j] = -2;
                    } else {
                        if (pos + 1 == L) s[st][j] = st;
                        else s[st][j] = id(pos + 1, digit(v, pos + 1));
                    }
                }
            }
        }
    }

    return s;
}
