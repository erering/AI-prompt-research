#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    if (N <= 1) {
        vector<vector<int>> s(1, vector<int>(N + 1, -1));
        s[0][0] = 0;
        return s;
    }

    const int INF = 1e9;
    vector<int> dp(N + 1, INF), par(N + 1, -1), parg(N + 1, -1);
    dp[1] = 0;

    for (int p = 1; p < N; ++p) {
        if (dp[p] == INF) continue;
        for (int g = 2; g <= N; ++g) {
            long long prod = 1LL * p * g;
            int q = (prod >= N ? N : (int)prod);
            int nd = dp[p] + g + 1;
            if (nd < dp[q]) {
                dp[q] = nd;
                par[q] = p;
                parg[q] = g;
            }
        }
    }

    vector<int> radix;
    int cur = N;
    while (cur != 1) {
        radix.push_back(parg[cur]);
        cur = par[cur];
    }
    reverse(radix.begin(), radix.end());

    int L = (int)radix.size();
    vector<long long> suffix(L + 1, 1);
    for (int i = L - 1; i >= 0; --i) suffix[i] = suffix[i + 1] * radix[i];

    vector<int> A(L);
    vector<vector<int>> B(L);
    int rows = 0;
    for (int p = 0; p < L; ++p) {
        A[p] = rows++;
        B[p].resize(radix[p]);
        for (int d = 0; d < radix[p]; ++d) B[p][d] = rows++;
    }

    vector<vector<int>> s(rows, vector<int>(N + 1, 0));

    auto digit = [&](int j, int p) -> int {
        long long v = j - 1;
        return (int)((v / suffix[p + 1]) % radix[p]);
    };

    for (int p = 0; p < L; ++p) {
        s[A[p]][0] = 0;
        for (int j = 1; j <= N; ++j) {
            int d = digit(j, p);
            s[A[p]][j] = B[p][d];
        }

        for (int d = 0; d < radix[p]; ++d) {
            int st = B[p][d];
            s[st][0] = 1;
            for (int j = 1; j <= N; ++j) {
                int bd = digit(j, p);
                if (bd < d) s[st][j] = -2;
                else if (bd > d) s[st][j] = -1;
                else {
                    if (p + 1 < L) s[st][j] = A[p + 1];
                    else s[st][j] = -1;
                }
            }
        }
    }

    return s;
}
