#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    if (N == 2) {
        vector<vector<int>> s(1, vector<int>(N + 1, 0));
        s[0][0] = 0;
        s[0][1] = -1;
        s[0][2] = -2;
        return s;
    }

    const int INF = 1e9;
    vector<int> dp(N + 1, INF), pre(N + 1, -1), fac(N + 1, -1);
    dp[1] = 0;

    for (int it = 0; it < 30; ++it) {
        bool changed = false;
        for (int p = 1; p <= N; ++p) {
            if (dp[p] == INF) continue;
            for (int r = 2; r <= N; ++r) {
                int np = (p > (N + r - 1) / r) ? N : p * r;
                if (dp[np] > dp[p] + r) {
                    dp[np] = dp[p] + r;
                    pre[np] = p;
                    fac[np] = r;
                    changed = true;
                }
            }
        }
        if (!changed) break;
    }

    vector<int> base;
    int cur = N;
    while (cur != 1) {
        base.push_back(fac[cur]);
        cur = pre[cur];
    }
    reverse(base.begin(), base.end());

    int L = (int)base.size();
    vector<int> suffix(L + 1, 1);
    for (int i = L - 1; i >= 0; --i) suffix[i] = suffix[i + 1] * base[i];

    vector<vector<int>> id(L);
    int states = 1;
    for (int i = 0; i < L; ++i) {
        id[i].resize(base[i]);
        for (int d = 0; d < base[i]; ++d) id[i][d] = states++;
    }

    vector<vector<int>> s(states, vector<int>(N + 1, 0));

    auto digit = [&](int value, int pos) {
        int v = value - 1;
        return (v / suffix[pos + 1]) % base[pos];
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = id[0][digit(j, 0)];
    }

    for (int pos = 0; pos < L; ++pos) {
        for (int d = 0; d < base[pos]; ++d) {
            int st = id[pos][d];
            s[st][0] = (pos % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; ++j) {
                int c = digit(j, pos);

                if (pos % 2 == 0) {
                    if (c > d) s[st][j] = -1;
                    else if (c < d) s[st][j] = -2;
                    else {
                        if (pos + 1 == L) s[st][j] = -1;
                        else s[st][j] = id[pos + 1][digit(j, pos + 1)];
                    }
                } else {
                    if (c < d) s[st][j] = -1;
                    else if (c > d) s[st][j] = -2;
                    else {
                        if (pos + 1 == L) s[st][j] = -1;
                        else s[st][j] = id[pos + 1][digit(j, pos + 1)];
                    }
                }
            }
        }
    }

    return s;
}
