#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int INF = 1e9;

    vector<int> dp(N + 1, INF), parP(N + 1, -1), parB(N + 1, -1);
    dp[1] = 0;

    for (int p = 1; p <= N; ++p) {
        if (dp[p] == INF) continue;
        for (int b = 2; p * b <= N; ++b) {
            int np = p * b;
            if (dp[p] + b < dp[np]) {
                dp[np] = dp[p] + b;
                parP[np] = p;
                parB[np] = b;
            }
        }
    }

    vector<int> bestArg(N + 2, 1);
    int best = INF, arg = 1;
    for (int i = N; i >= 1; --i) {
        if (dp[i] < best) {
            best = dp[i];
            arg = i;
        }
        bestArg[i] = arg;
    }

    int bestCost = INF, bestPrefix = 1, bestLast = 2;
    for (int last = 2; last <= N; ++last) {
        int need = (N + last - 1) / last;
        int pref = bestArg[need];
        int cost = dp[pref] + (last + 1) / 2;
        if (cost < bestCost) {
            bestCost = cost;
            bestPrefix = pref;
            bestLast = last;
        }
    }

    vector<int> bases;
    int cur = bestPrefix;
    while (cur > 1) {
        int b = parB[cur];
        bases.push_back(b);
        cur = parP[cur];
    }
    reverse(bases.begin(), bases.end());
    bases.push_back(bestLast);

    int L = (int)bases.size();

    vector<long long> suffix(L + 1, 1);
    for (int i = L - 1; i >= 0; --i) suffix[i] = suffix[i + 1] * bases[i];

    auto digit = [&](int j, int p) -> int {
        long long v = j - 1;
        return (int)((v / suffix[p + 1]) % bases[p]);
    };

    vector<int> offset(L);
    int totalStates = 0;
    for (int p = 0; p < L; ++p) {
        offset[p] = 1 + totalStates;
        if (p == L - 1) totalStates += (bases[p] + 1) / 2;
        else totalStates += bases[p];
    }

    vector<vector<int>> s(totalStates + 1, vector<int>(N + 1, 0));

    auto state_id = [&](int p, int d) -> int {
        if (p == L - 1) return offset[p] + d / 2;
        return offset[p] + d;
    };

    auto declare_cmp = [&](int p, int cmp) -> int {
        if (p % 2 == 0) {
            return cmp < 0 ? -2 : -1;
        } else {
            return cmp < 0 ? -1 : -2;
        }
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = state_id(0, digit(j, 0));
    }

    for (int p = 0; p < L; ++p) {
        int cnt = (p == L - 1) ? (bases[p] + 1) / 2 : bases[p];

        for (int c = 0; c < cnt; ++c) {
            int id = offset[p] + c;
            s[id][0] = (p % 2 == 0) ? 1 : 0;

            for (int j = 1; j <= N; ++j) {
                int e = digit(j, p);

                if (p == L - 1) {
                    int lo = 2 * c;
                    int hi = min(lo + 1, bases[p] - 1);
                    int cmp = 0;

                    if (e < lo) cmp = -1;
                    else if (e > hi) cmp = 1;
                    else if (lo < hi && e == lo) cmp = -1;
                    else if (lo < hi && e == hi) cmp = 1;
                    else cmp = 1;

                    s[id][j] = declare_cmp(p, cmp);
                } else {
                    int d = c;
                    if (e < d) {
                        s[id][j] = declare_cmp(p, -1);
                    } else if (e > d) {
                        s[id][j] = declare_cmp(p, 1);
                    } else {
                        int nd = digit(j, p + 1);
                        s[id][j] = state_id(p + 1, nd);
                    }
                }
            }
        }
    }

    return s;
}
