#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int INF = 1e9;

    vector<int> dp(N + 1, INF), choice(N + 1, -1);
    dp[1] = 0;

    for (int v = 2; v <= N; ++v) {
        for (int k = 2; k <= v; ++k) {
            int nv = (v + k - 1) / k;
            if (k + dp[nv] < dp[v]) {
                dp[v] = k + dp[nv];
                choice[v] = k;
            }
        }
    }

    vector<int> rad;
    for (int v = N; v > 1; ) {
        int k = choice[v];
        rad.push_back(k);
        v = (v + k - 1) / k;
    }

    int L = (int)rad.size();
    vector<int> offset(L);
    int total = 0;
    for (int i = 0; i < L; ++i) {
        offset[i] = total + 1;
        total += rad[i];
    }

    vector<long long> suf(L + 1, 1);
    for (int i = L - 1; i >= 0; --i) {
        suf[i] = suf[i + 1] * rad[i];
    }

    auto digit = [&](int coins, int pos) -> int {
        long long v = coins - 1;
        return (int)((v / suf[pos + 1]) % rad[pos]);
    };

    vector<vector<int>> s(total + 1, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = offset[0] + digit(j, 0);
    }

    for (int p = 0; p < L; ++p) {
        for (int d = 0; d < rad[p]; ++d) {
            int id = offset[p] + d;
            bool inspectB = (p % 2 == 0);
            s[id][0] = inspectB ? 1 : 0;

            for (int j = 1; j <= N; ++j) {
                int cur = digit(j, p);

                if (cur < d) {
                    s[id][j] = inspectB ? -2 : -1;
                } else if (cur > d) {
                    s[id][j] = inspectB ? -1 : -2;
                } else {
                    if (p == L - 1) {
                        s[id][j] = -1;
                    } else {
                        s[id][j] = offset[p + 1] + digit(j, p + 1);
                    }
                }
            }
        }
    }

    return s;
}
