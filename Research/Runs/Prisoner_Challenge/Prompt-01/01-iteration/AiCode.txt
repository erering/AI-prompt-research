#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    vector<int> p3(1, 1);
    while (p3.back() < N) p3.push_back(p3.back() * 3);
    int D = (int)p3.size() - 1;
    if (D == 0) D = 1;

    int x = 3 * D;
    vector<vector<int>> s(x + 1, vector<int>(N + 1, 0));

    auto id = [](int level, int digit) {
        return 1 + level * 3 + digit;
    };

    auto digit_at = [&](int v, int level) {
        int z = v - 1;
        int div = p3[D - 1 - level];
        return (z / div) % 3;
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; j++) {
        int d = digit_at(j, 0);
        s[0][j] = id(0, d);
    }

    for (int level = 0; level < D; level++) {
        for (int stored = 0; stored < 3; stored++) {
            int st = id(level, stored);
            bool inspectB = (level % 2 == 0);
            s[st][0] = inspectB ? 1 : 0;

            for (int j = 1; j <= N; j++) {
                int cur = digit_at(j, level);

                if (inspectB) {
                    if (stored < cur) {
                        s[st][j] = -1;
                    } else if (stored > cur) {
                        s[st][j] = -2;
                    } else {
                        if (level == D - 1) {
                            s[st][j] = -1;
                        } else {
                            int nd = digit_at(j, level + 1);
                            s[st][j] = id(level + 1, nd);
                        }
                    }
                } else {
                    if (cur < stored) {
                        s[st][j] = -1;
                    } else if (cur > stored) {
                        s[st][j] = -2;
                    } else {
                        if (level == D - 1) {
                            s[st][j] = -1;
                        } else {
                            int nd = digit_at(j, level + 1);
                            s[st][j] = id(level + 1, nd);
                        }
                    }
                }
            }
        }
    }

    return s;
}
