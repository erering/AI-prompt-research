#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int INF = 1e9;

    vector<int> dp(N + 1, INF), parPrev(N + 1, -1), parBase(N + 1, -1);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;

    dp[1] = 0;
    pq.push({0, 1});

    while (!pq.empty()) {
        auto [cost, prod] = pq.top();
        pq.pop();
        if (cost != dp[prod]) continue;

        for (int base = 2; base <= N; base++) {
            int nprod;
            if (prod > N / base) nprod = N;
            else nprod = prod * base;

            if (cost + base < dp[nprod]) {
                dp[nprod] = cost + base;
                parPrev[nprod] = prod;
                parBase[nprod] = base;
                pq.push({dp[nprod], nprod});
            }
        }
    }

    vector<int> best_ge(N + 2, N);
    int best = N;
    for (int i = N; i >= 1; i--) {
        if (dp[i] < dp[best]) best = i;
        best_ge[i] = best;
    }

    int bestCost = INF;
    int bestLast = -1;
    int bestPrevTarget = -1;

    for (int last = 2; last <= N; last++) {
        int need = (N + last - 1) / last;
        int prevCost = (need <= 1 ? 0 : dp[best_ge[need]]);
        int totalCost = prevCost + last - 2;

        if (totalCost < bestCost) {
            bestCost = totalCost;
            bestLast = last;
            bestPrevTarget = need;
        }
    }

    vector<int> bases;
    if (bestPrevTarget > 1) {
        int v = best_ge[bestPrevTarget];
        vector<int> rev;
        while (v != 1) {
            rev.push_back(parBase[v]);
            v = parPrev[v];
        }
        reverse(rev.begin(), rev.end());
        bases = rev;
    }
    bases.push_back(bestLast);

    int L = (int)bases.size();

    vector<long long> weight(L, 1);
    for (int i = L - 2; i >= 0; i--) {
        weight[i] = weight[i + 1] * bases[i + 1];
    }

    auto digit = [&](int value, int level) -> int {
        return (int)(((long long)(value - 1) / weight[level]) % bases[level]);
    };

    vector<vector<int>> id(L);
    int nxt = 1;

    for (int level = 0; level < L; level++) {
        int base = bases[level];
        id[level].assign(base, -1);

        for (int d = 0; d < base; d++) {
            if (level == L - 1 && (d == 0 || d == base - 1)) continue;
            id[level][d] = nxt++;
        }
    }

    int x = nxt - 1;
    vector<vector<int>> s(x + 1, vector<int>(N + 1, 0));

    auto declare_fewer = [&](int bag) -> int {
        return bag == 0 ? -1 : -2;
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; j++) {
        int d = digit(j, 0);

        if (L == 1) {
            if (d == 0) s[0][j] = -1;
            else if (d == bases[0] - 1) s[0][j] = -2;
            else s[0][j] = id[0][d];
        } else {
            s[0][j] = id[0][d];
        }
    }

    for (int level = 0; level < L; level++) {
        int base = bases[level];

        for (int storedDigit = 0; storedDigit < base; storedDigit++) {
            int row = id[level][storedDigit];
            if (row == -1) continue;

            int storedBag = level % 2;
            int currentBag = 1 - storedBag;
            s[row][0] = currentBag;

            for (int j = 1; j <= N; j++) {
                int currentDigit = digit(j, level);

                if (currentDigit < storedDigit) {
                    s[row][j] = declare_fewer(currentBag);
                } else if (currentDigit > storedDigit) {
                    s[row][j] = declare_fewer(storedBag);
                } else {
                    if (level == L - 1) {
                        s[row][j] = -1;
                    } else {
                        int nextDigit = digit(j, level + 1);
                        int nextBase = bases[level + 1];

                        if (level + 1 == L - 1 && nextDigit == 0) {
                            s[row][j] = declare_fewer(currentBag);
                        } else if (level + 1 == L - 1 && nextDigit == nextBase - 1) {
                            s[row][j] = declare_fewer(storedBag);
                        } else {
                            s[row][j] = id[level + 1][nextDigit];
                        }
                    }
                }
            }
        }
    }

    return s;
}
