#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int L = 8;
    const int BASE = 3;

    vector<int> pw(L);
    pw[L - 1] = 1;
    for (int i = L - 2; i >= 0; --i) pw[i] = pw[i + 1] * BASE;

    auto digit = [&](int value, int pos) {
        int v = value - 1;
        return (v / pw[pos]) % BASE;
    };

    vector<vector<int>> id(L, vector<int>(BASE));
    int cur = 1;
    for (int p = 0; p < L; ++p) {
        for (int d = 0; d < BASE; ++d) {
            id[p][d] = cur++;
        }
    }

    vector<vector<int>> s(cur, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = id[0][digit(j, 0)];
    }

    for (int p = 0; p < L; ++p) {
        int inspect = (p % 2 == 0 ? 1 : 0);

        for (int d = 0; d < BASE; ++d) {
            int state = id[p][d];
            s[state][0] = inspect;

            for (int j = 1; j <= N; ++j) {
                int e = digit(j, p);

                if (e < d) {
                    s[state][j] = (inspect == 0 ? -1 : -2);
                } else if (e > d) {
                    s[state][j] = (inspect == 0 ? -2 : -1);
                } else {
                    if (p == L - 1) {
                        s[state][j] = -1;
                    } else {
                        s[state][j] = id[p + 1][digit(j, p + 1)];
                    }
                }
            }
        }
    }

    return s;
}
