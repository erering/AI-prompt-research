#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int LOW = -1;
    const int HIGH = -2;

    vector<long long> dp(61, -1);
    vector<int> pre(61, -1);
    dp[0] = 2;

    for (int s = 1; s <= 60; ++s) {
        for (int q = 2; q <= s; ++q) {
            if (dp[s - q] < 0) continue;
            long long val = 2 + 1LL * q * dp[s - q];
            if (val > dp[s]) {
                dp[s] = val;
                pre[s] = q;
            }
        }
    }

    int sum = 0;
    while (sum <= 60 && dp[sum] < N) ++sum;

    vector<int> radix;
    int curSum = sum;
    while (curSum > 0) {
        int q = pre[curSum];
        radix.push_back(q);
        curSum -= q;
    }

    int L = (int)radix.size();

    vector<long long> cap(L + 1);
    cap[L] = 2;
    for (int i = L - 1; i >= 0; --i) {
        cap[i] = 2 + 1LL * radix[i] * cap[i + 1];
    }

    vector<vector<int>> elem(L + 1, vector<int>(N + 1, LOW));

    function<void(int, int, int)> build = [&](int l, int r, int p) {
        if (l > r) return;

        if (p == L) {
            elem[p][l] = LOW;
            if (r > l) elem[p][r] = HIGH;
            return;
        }

        elem[p][l] = LOW;
        if (r > l) elem[p][r] = HIGH;
        if (r - l + 1 <= 2) return;

        int q = radix[p];
        int rem = r - l - 1;
        int cur = l + 1;
        int nxtCap = (int)cap[p + 1];

        for (int c = 0; c < q; ++c) {
            int sz = min(nxtCap, rem);
            if (sz > 0) {
                for (int v = cur; v < cur + sz; ++v) elem[p][v] = c;
                build(cur, cur + sz - 1, p + 1);
                cur += sz;
                rem -= sz;
            }
        }
    };

    build(1, N, 0);

    vector<vector<int>> id(L);
    int states = 0;
    for (int p = 0; p < L; ++p) {
        id[p].resize(radix[p]);
        for (int d = 0; d < radix[p]; ++d) {
            id[p][d] = ++states;
        }
    }

    vector<vector<int>> s(states + 1, vector<int>(N + 1, 0));

    auto declare_observed = [&](bool inspectA, bool observedFewer) {
        if (inspectA) return observedFewer ? -1 : -2;
        return observedFewer ? -2 : -1;
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        int e = elem[0][j];
        if (e == LOW) s[0][j] = -1;
        else if (e == HIGH) s[0][j] = -2;
        else s[0][j] = id[0][e];
    }

    for (int p = 0; p < L; ++p) {
        for (int d = 0; d < radix[p]; ++d) {
            int row = id[p][d];
            bool inspectA = (p % 2 == 1);
            s[row][0] = inspectA ? 0 : 1;

            for (int j = 1; j <= N; ++j) {
                int e = elem[p][j];

                if (e == LOW) {
                    s[row][j] = declare_observed(inspectA, true);
                } else if (e == HIGH) {
                    s[row][j] = declare_observed(inspectA, false);
                } else if (e < d) {
                    s[row][j] = declare_observed(inspectA, true);
                } else if (e > d) {
                    s[row][j] = declare_observed(inspectA, false);
                } else {
                    int ne = elem[p + 1][j];
                    if (ne == LOW) {
                        s[row][j] = declare_observed(inspectA, true);
                    } else if (ne == HIGH) {
                        s[row][j] = declare_observed(inspectA, false);
                    } else {
                        s[row][j] = id[p + 1][ne];
                    }
                }
            }
        }
    }

    return s;
}
