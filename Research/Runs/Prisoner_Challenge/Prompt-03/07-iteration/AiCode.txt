#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int B = 3;

    int L = 0;
    int pwr = 1;
    while (pwr <= N) {
        pwr *= B;
        ++L;
    }

    auto digit = [&](int v, int pos) {
        while (pos--) v /= B;
        return v % B;
    };

    auto id = [&](int pos, int d) {
        return 1 + pos * B + d;
    };

    int X = L * B;
    vector<vector<int>> s(X + 1, vector<int>(N + 1, -1));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = id(L - 1, digit(j, L - 1));
    }

    for (int pos = 0; pos < L; ++pos) {
        for (int d = 0; d < B; ++d) {
            int row = id(pos, d);
            int sel = ((L - 1 - pos) % 2 == 0) ? 1 : 0;
            s[row][0] = sel;

            for (int j = 1; j <= N; ++j) {
                int e = digit(j, pos);

                if (e < d) {
                    s[row][j] = (sel == 0 ? -1 : -2);
                } else if (e > d) {
                    s[row][j] = (sel == 0 ? -2 : -1);
                } else {
                    if (pos == 0) {
                        s[row][j] = -1;
                    } else {
                        s[row][j] = id(pos - 1, digit(j, pos - 1));
                    }
                }
            }
        }
    }

    return s;
}
