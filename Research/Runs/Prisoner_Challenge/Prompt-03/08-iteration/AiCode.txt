#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    int D = 0, pw = 1;
    while (pw < N) {
        pw *= 3;
        ++D;
    }

    auto digit = [](int v, int p) {
        while (p--) v /= 3;
        return v % 3;
    };

    auto id = [](int level, int d) {
        return 1 + level * 3 + d;
    };

    vector<vector<int>> s(1 + 3 * D, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int y = 1; y <= N; ++y) {
        s[0][y] = id(0, digit(y - 1, D - 1));
    }

    for (int level = 0; level < D; ++level) {
        int pos = D - 1 - level;
        bool prev_is_A = (level % 2 == 0);
        int selected = prev_is_A ? 1 : 0;

        for (int d = 0; d < 3; ++d) {
            int row = id(level, d);
            s[row][0] = selected;

            for (int y = 1; y <= N; ++y) {
                int c = digit(y - 1, pos);

                if (c != d) {
                    if (prev_is_A) {
                        s[row][y] = (d < c ? -1 : -2);
                    } else {
                        s[row][y] = (d < c ? -2 : -1);
                    }
                } else {
                    if (pos == 0) {
                        s[row][y] = -1;
                    } else {
                        s[row][y] = id(level + 1, digit(y - 1, pos - 1));
                    }
                }
            }
        }
    }

    return s;
}
