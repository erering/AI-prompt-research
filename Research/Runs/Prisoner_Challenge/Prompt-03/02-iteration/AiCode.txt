#include <vector>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    int L = 0, pw = 1;
    while (pw <= N) {
        pw *= 3;
        ++L;
    }

    vector<int> pow3(L);
    pow3[0] = 1;
    for (int i = 1; i < L; ++i) pow3[i] = pow3[i - 1] * 3;

    auto digit = [&](int v, int p) {
        return (v / pow3[p]) % 3;
    };

    auto id = [&](int p, int d) {
        int level = (L - 1) - p;
        return 1 + level * 3 + d;
    };

    vector<vector<int>> s(1 + 3 * L, vector<int>(N + 1, -1));

    s[0][0] = 0;
    for (int v = 1; v <= N; ++v) {
        s[0][v] = id(L - 1, digit(v, L - 1));
    }

    for (int p = L - 1; p >= 0; --p) {
        int level = (L - 1) - p;
        bool storedA = (level % 2 == 0);

        for (int d = 0; d < 3; ++d) {
            int row = id(p, d);
            s[row][0] = storedA ? 1 : 0;

            for (int v = 1; v <= N; ++v) {
                int e = digit(v, p);

                if (e < d) {
                    s[row][v] = storedA ? -2 : -1;
                } else if (e > d) {
                    s[row][v] = storedA ? -1 : -2;
                } else {
                    if (p == 0) {
                        s[row][v] = -1;
                    } else {
                        s[row][v] = id(p - 1, digit(v, p - 1));
                    }
                }
            }
        }
    }

    return s;
}
