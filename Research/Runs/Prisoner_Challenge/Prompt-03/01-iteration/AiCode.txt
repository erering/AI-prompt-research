#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    int p = 1, D = 0;
    while (p < N) {
        p *= 3;
        ++D;
    }

    int lower = p / 3;
    int first = (N + lower - 1) / lower;

    vector<int> radix;
    radix.push_back(first);
    for (int i = 1; i < D; ++i) radix.push_back(3);

    vector<int> place(D, 1);
    for (int i = D - 2; i >= 0; --i) {
        place[i] = place[i + 1] * radix[i + 1];
    }

    vector<vector<int>> id(D);
    int cnt = 0;
    for (int i = 0; i < D; ++i) {
        id[i].resize(radix[i]);
        for (int d = 0; d < radix[i]; ++d) {
            id[i][d] = ++cnt;
        }
    }

    vector<vector<int>> s(cnt + 1, vector<int>(N + 1, 0));

    auto digit = [&](int value, int pos) {
        int code = value - 1;
        return (code / place[pos]) % radix[pos];
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = id[0][digit(j, 0)];
    }

    for (int i = 0; i < D; ++i) {
        for (int d = 0; d < radix[i]; ++d) {
            int row = id[i][d];
            s[row][0] = (i % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; ++j) {
                int cur = digit(j, i);

                if (cur == d) {
                    if (i + 1 < D) {
                        s[row][j] = id[i + 1][digit(j, i + 1)];
                    } else {
                        s[row][j] = -1;
                    }
                } else {
                    if (i % 2 == 0) {
                        s[row][j] = (d < cur ? -1 : -2);
                    } else {
                        s[row][j] = (cur < d ? -1 : -2);
                    }
                }
            }
        }
    }

    return s;
}
