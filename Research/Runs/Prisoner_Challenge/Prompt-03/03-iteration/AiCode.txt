#include <vector>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int B = 3;

    vector<int> pw(1, 1);
    while (pw.back() < N) pw.push_back(pw.back() * B);
    int L = (int)pw.size() - 1;

    auto digit = [&](int v, int p) {
        int x = v - 1;
        return (x / pw[L - 1 - p]) % B;
    };

    auto id = [&](int p, int d) {
        return 1 + p * B + d;
    };

    vector<vector<int>> s(1 + L * B, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = id(0, digit(j, 0));
    }

    for (int p = 0; p < L; ++p) {
        for (int d = 0; d < B; ++d) {
            int row = id(p, d);
            s[row][0] = (p % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; ++j) {
                int c = digit(j, p);

                if (p % 2 == 0) {
                    if (d < c) s[row][j] = -1;
                    else if (d > c) s[row][j] = -2;
                    else if (p == L - 1) s[row][j] = -1;
                    else s[row][j] = id(p + 1, digit(j, p + 1));
                } else {
                    if (c < d) s[row][j] = -1;
                    else if (c > d) s[row][j] = -2;
                    else if (p == L - 1) s[row][j] = -1;
                    else s[row][j] = id(p + 1, digit(j, p + 1));
                }
            }
        }
    }

    return s;
}
