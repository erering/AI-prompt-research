#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int B = 3;

    int L = 0;
    int pw = 1;
    while (pw < N) {
        pw *= B;
        ++L;
    }

    vector<int> place(L);
    place[L - 1] = 1;
    for (int i = L - 2; i >= 0; --i) place[i] = place[i + 1] * B;

    auto digit = [&](int v, int p) {
        return ((v - 1) / place[p]) % B;
    };

    int rows = L * (B + 1);
    vector<vector<int>> s(rows, vector<int>(N + 1, 0));

    for (int p = 0; p < L; ++p) {
        int start = p * (B + 1);

        s[start][0] = 0;
        for (int y = 1; y <= N; ++y) {
            int d = digit(y, p);
            s[start][y] = start + 1 + d;
        }

        for (int d = 0; d < B; ++d) {
            int row = start + 1 + d;
            s[row][0] = 1;

            for (int y = 1; y <= N; ++y) {
                int e = digit(y, p);

                if (d < e) s[row][y] = -1;
                else if (d > e) s[row][y] = -2;
                else {
                    if (p + 1 < L) s[row][y] = (p + 1) * (B + 1);
                    else s[row][y] = -1;
                }
            }
        }
    }

    return s;
}
