#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    auto max_product = [](int s) {
        long long res = 1;
        if (s == 2) return 2LL;
        if (s == 3) return 3LL;
        if (s % 3 == 0) {
            for (int i = 0; i < s / 3; ++i) res *= 3;
        } else if (s % 3 == 1) {
            res = 4;
            for (int i = 0; i < (s - 4) / 3; ++i) res *= 3;
        } else {
            res = 2;
            for (int i = 0; i < (s - 2) / 3; ++i) res *= 3;
        }
        return res;
    };

    int S = 2;
    while (max_product(S) < N) ++S;

    vector<int> base;
    if (S == 2) {
        base.push_back(2);
    } else if (S == 3) {
        base.push_back(3);
    } else if (S % 3 == 0) {
        for (int i = 0; i < S / 3; ++i) base.push_back(3);
    } else if (S % 3 == 1) {
        base.push_back(4);
        for (int i = 0; i < (S - 4) / 3; ++i) base.push_back(3);
    } else {
        base.push_back(2);
        for (int i = 0; i < (S - 2) / 3; ++i) base.push_back(3);
    }

    int L = (int)base.size();

    vector<int> weight(L, 1);
    for (int i = L - 2; i >= 0; --i) {
        weight[i] = weight[i + 1] * base[i + 1];
    }

    vector<int> offset(L);
    offset[0] = 1;
    for (int i = 1; i < L; ++i) {
        offset[i] = offset[i - 1] + base[i - 1];
    }

    auto digit = [&](int v, int pos) {
        int z = v - 1;
        return (z / weight[pos]) % base[pos];
    };

    auto state = [&](int pos, int d) {
        return offset[pos] + d;
    };

    int total_states = 1;
    for (int b : base) total_states += b;

    vector<vector<int>> s(total_states, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = state(0, digit(j, 0));
    }

    for (int pos = 0; pos < L; ++pos) {
        for (int d = 0; d < base[pos]; ++d) {
            int id = state(pos, d);
            bool even = (pos % 2 == 0);

            s[id][0] = even ? 1 : 0;

            for (int j = 1; j <= N; ++j) {
                int e = digit(j, pos);

                if (even) {
                    if (e > d) {
                        s[id][j] = -1;
                    } else if (e < d) {
                        s[id][j] = -2;
                    } else {
                        if (pos == L - 1) s[id][j] = -1;
                        else s[id][j] = state(pos + 1, digit(j, pos + 1));
                    }
                } else {
                    if (e < d) {
                        s[id][j] = -1;
                    } else if (e > d) {
                        s[id][j] = -2;
                    } else {
                        if (pos == L - 1) s[id][j] = -1;
                        else s[id][j] = state(pos + 1, digit(j, pos + 1));
                    }
                }
            }
        }
    }

    return s;
}
