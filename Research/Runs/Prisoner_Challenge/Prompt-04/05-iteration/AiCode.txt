#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int MAXS = 60;

    vector<long long> dp(MAXS + 1, -1);
    vector<int> pre(MAXS + 1, -1);
    dp[0] = 1;

    for (int s = 1; s <= MAXS; ++s) {
        for (int r = 2; r <= s; ++r) {
            if (dp[s - r] < 0) continue;
            long long val = dp[s - r] * r;
            if (val > dp[s]) {
                dp[s] = val;
                pre[s] = r;
            }
        }
    }

    int sum = 0;
    while (dp[sum] < N) ++sum;

    vector<int> base;
    int curSum = sum;
    while (curSum > 0) {
        int r = pre[curSum];
        base.push_back(r);
        curSum -= r;
    }

    int D = (int)base.size();

    vector<int> suffix(D + 1, 1);
    for (int i = D - 1; i >= 0; --i) {
        suffix[i] = suffix[i + 1] * base[i];
    }

    vector<int> offset(D);
    int states = 1;
    for (int i = 0; i < D; ++i) {
        offset[i] = states;
        states += base[i];
    }

    vector<vector<int>> s(states, vector<int>(N + 1, 0));

    auto digit = [&](int value, int pos) -> int {
        int x = value - 1;
        return (x / suffix[pos + 1]) % base[pos];
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = offset[0] + digit(j, 0);
    }

    for (int pos = 0; pos < D; ++pos) {
        for (int v = 0; v < base[pos]; ++v) {
            int id = offset[pos] + v;

            s[id][0] = (pos % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; ++j) {
                int d = digit(j, pos);

                if (d < v) {
                    s[id][j] = (pos % 2 == 0 ? -2 : -1);
                } else if (d > v) {
                    s[id][j] = (pos % 2 == 0 ? -1 : -2);
                } else {
                    if (pos + 1 == D) {
                        s[id][j] = -1;
                    } else {
                        s[id][j] = offset[pos + 1] + digit(j, pos + 1);
                    }
                }
            }
        }
    }

    return s;
}
