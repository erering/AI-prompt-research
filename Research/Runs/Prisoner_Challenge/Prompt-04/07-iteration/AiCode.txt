#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    vector<int> radix;

    for (int sum = 2; sum <= 60; ++sum) {
        vector<int> cur;
        if (sum % 3 == 0) {
            for (int i = 0; i < sum / 3; ++i) cur.push_back(3);
        } else if (sum % 3 == 1) {
            cur.push_back(4);
            for (int i = 0; i < (sum - 4) / 3; ++i) cur.push_back(3);
        } else {
            cur.push_back(2);
            for (int i = 0; i < (sum - 2) / 3; ++i) cur.push_back(3);
        }

        long long prod = 1;
        for (int r : cur) prod *= r;

        if (prod >= N) {
            radix = cur;
            break;
        }
    }

    int D = (int)radix.size();

    vector<int> place(D, 1);
    for (int i = D - 2; i >= 0; --i) {
        place[i] = place[i + 1] * radix[i + 1];
    }

    vector<int> offset(D);
    offset[0] = 1;
    for (int i = 1; i < D; ++i) {
        offset[i] = offset[i - 1] + radix[i - 1];
    }

    int x = 0;
    for (int r : radix) x += r;

    vector<vector<int>> dig(D, vector<int>(N + 1));
    for (int p = 0; p < D; ++p) {
        for (int j = 1; j <= N; ++j) {
            dig[p][j] = ((j - 1) / place[p]) % radix[p];
        }
    }

    vector<vector<int>> s(x + 1, vector<int>(N + 1, -1));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = offset[0] + dig[0][j];
    }

    for (int p = 0; p < D; ++p) {
        for (int v = 0; v < radix[p]; ++v) {
            int id = offset[p] + v;
            s[id][0] = (p % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; ++j) {
                int w = dig[p][j];

                if (p % 2 == 0) {
                    if (v < w) {
                        s[id][j] = -1;
                    } else if (v > w) {
                        s[id][j] = -2;
                    } else {
                        if (p + 1 == D) s[id][j] = -1;
                        else s[id][j] = offset[p + 1] + dig[p + 1][j];
                    }
                } else {
                    if (w < v) {
                        s[id][j] = -1;
                    } else if (w > v) {
                        s[id][j] = -2;
                    } else {
                        if (p + 1 == D) s[id][j] = -1;
                        else s[id][j] = offset[p + 1] + dig[p + 1][j];
                    }
                }
            }
        }
    }

    return s;
}
