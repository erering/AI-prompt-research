#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    int M = (N + 1) / 2;

    vector<int> bases;
    if (M == 1) {
        bases.push_back(1);
    } else {
        const int INF = 1e9;
        vector<int> dist(M + 1, INF), par(M + 1, -1), parBase(M + 1, -1);
        priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;

        dist[1] = 0;
        pq.push({0, 1});

        while (!pq.empty()) {
            auto [cd, p] = pq.top();
            pq.pop();
            if (cd != dist[p]) continue;
            if (p == M) break;

            for (int b = 2; b <= M; ++b) {
                int np = (1LL * p * b >= M) ? M : p * b;
                int nd = cd + b;
                if (nd < dist[np]) {
                    dist[np] = nd;
                    par[np] = p;
                    parBase[np] = b;
                    pq.push({nd, np});
                }
            }
        }

        int cur = M;
        while (cur != 1) {
            bases.push_back(parBase[cur]);
            cur = par[cur];
        }
        reverse(bases.begin(), bases.end());
    }

    int L = (int)bases.size();

    vector<int> offset(L);
    int x = 0;
    for (int i = 0; i < L; ++i) {
        offset[i] = x + 1;
        x += bases[i];
    }

    vector<int> suffix(L + 1, 1);
    for (int i = L - 1; i >= 0; --i) {
        suffix[i] = suffix[i + 1] * bases[i];
    }

    auto digit = [&](int coins, int level) -> int {
        int q = (coins - 1) / 2;
        return (q / suffix[level + 1]) % bases[level];
    };

    auto smaller = [](int bag) -> int {
        return bag == 0 ? -1 : -2;
    };

    auto larger = [](int bag) -> int {
        return bag == 0 ? -2 : -1;
    };

    vector<vector<int>> s(x + 1, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = offset[0] + digit(j, 0);
    }

    for (int level = 0; level < L; ++level) {
        int bag = (level + 1) % 2;

        for (int d = 0; d < bases[level]; ++d) {
            int id = offset[level] + d;
            s[id][0] = bag;

            for (int j = 1; j <= N; ++j) {
                int curDigit = digit(j, level);

                if (curDigit < d) {
                    s[id][j] = smaller(bag);
                } else if (curDigit > d) {
                    s[id][j] = larger(bag);
                } else if (level + 1 < L) {
                    s[id][j] = offset[level + 1] + digit(j, level + 1);
                } else {
                    int r = (j - 1) & 1;
                    s[id][j] = (r == 0 ? smaller(bag) : larger(bag));
                }
            }
        }
    }

    return s;
}
