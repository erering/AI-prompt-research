#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    vector<int> dp(N + 1, 0), choice(N + 1, 0);

    dp[1] = 0;
    for (int n = 2; n <= N; ++n) {
        dp[n] = n;
        choice[n] = n;
        for (int r = 2; r <= n; ++r) {
            int nxt = (n + r - 1) / r;
            int cost = r + dp[nxt];
            if (cost < dp[n]) {
                dp[n] = cost;
                choice[n] = r;
            }
        }
    }

    vector<int> radix;
    for (int cur = N; cur > 1; ) {
        int r = choice[cur];
        radix.push_back(r);
        cur = (cur + r - 1) / r;
    }

    int L = (int)radix.size();

    vector<int> offset(L);
    int x = 0;
    for (int i = 0; i < L; ++i) {
        offset[i] = x + 1;
        x += radix[i];
    }

    vector<int> suffix(L + 1, 1);
    for (int i = L - 1; i >= 0; --i) {
        suffix[i] = suffix[i + 1] * radix[i];
    }

    auto digit = [&](int coins, int pos) {
        int v = coins - 1;
        return (v / suffix[pos + 1]) % radix[pos];
    };

    vector<vector<int>> s(x + 1, vector<int>(N + 1, -1));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = offset[0] + digit(j, 0);
    }

    for (int pos = 0; pos < L; ++pos) {
        for (int d = 0; d < radix[pos]; ++d) {
            int id = offset[pos] + d;
            int inspect = (pos % 2 == 0 ? 1 : 0);
            s[id][0] = inspect;

            for (int j = 1; j <= N; ++j) {
                int cur = digit(j, pos);

                if (cur == d) {
                    if (pos + 1 == L) {
                        s[id][j] = -1;
                    } else {
                        s[id][j] = offset[pos + 1] + digit(j, pos + 1);
                    }
                } else if (inspect == 1) {
                    s[id][j] = (d < cur ? -1 : -2);
                } else {
                    s[id][j] = (cur < d ? -1 : -2);
                }
            }
        }
    }

    return s;
}
