#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int INF = 1e9;

    vector<int> distSum(N + 1, INF), distLen(N + 1, INF);
    vector<int> pre(N + 1, -1), preR(N + 1, -1);

    using T = tuple<int, int, int>;
    priority_queue<T, vector<T>, greater<T>> pq;

    distSum[1] = 0;
    distLen[1] = 0;
    pq.emplace(0, 0, 1);

    while (!pq.empty()) {
        auto [sum, len, p] = pq.top();
        pq.pop();

        if (sum != distSum[p] || len != distLen[p]) continue;
        if (p == N) break;

        for (int r : {2, 3, 4}) {
            int np = min(N, p * r);
            int ns = sum + r;
            int nl = len + 1;

            if (ns < distSum[np] || (ns == distSum[np] && nl < distLen[np])) {
                distSum[np] = ns;
                distLen[np] = nl;
                pre[np] = p;
                preR[np] = r;
                pq.emplace(ns, nl, np);
            }
        }
    }

    vector<int> radix;
    for (int cur = N; cur != 1; cur = pre[cur]) {
        radix.push_back(preR[cur]);
    }
    reverse(radix.begin(), radix.end());

    int L = (int)radix.size();

    vector<int> start(L);
    int total = 1;
    for (int i = 0; i < L; i++) {
        start[i] = total;
        total += radix[i];
    }

    vector<long long> place(L);
    long long prod = 1;
    for (int i = L - 1; i >= 0; i--) {
        place[i] = prod;
        prod *= radix[i];
    }

    auto digit = [&](int value, int level) -> int {
        long long v = value - 1;
        return (int)((v / place[level]) % radix[level]);
    };

    auto id = [&](int level, int d) -> int {
        return start[level] + d;
    };

    vector<vector<int>> s(total, vector<int>(N + 1, -1));

    s[0][0] = 0;
    for (int j = 1; j <= N; j++) {
        s[0][j] = id(0, digit(j, 0));
    }

    for (int level = 0; level < L; level++) {
        for (int d = 0; d < radix[level]; d++) {
            int state = id(level, d);
            int inspect = (level % 2 == 0 ? 1 : 0);
            s[state][0] = inspect;

            for (int j = 1; j <= N; j++) {
                int cur = digit(j, level);

                if (cur == d) {
                    if (level == L - 1) {
                        s[state][j] = -1;
                    } else {
                        s[state][j] = id(level + 1, digit(j, level + 1));
                    }
                } else if (inspect == 1) {
                    s[state][j] = (d < cur ? -1 : -2);
                } else {
                    s[state][j] = (cur < d ? -1 : -2);
                }
            }
        }
    }

    return s;
}
