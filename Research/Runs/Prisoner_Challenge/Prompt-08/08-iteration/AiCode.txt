#include <vector>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    int L = 0, p = 1;
    while (p < N) {
        p *= 3;
        ++L;
    }

    vector<int> pow3(L);
    pow3[L - 1] = 1;
    for (int i = L - 2; i >= 0; --i) pow3[i] = pow3[i + 1] * 3;

    auto digit = [&](int v, int pos) {
        int z = v - 1;
        return (z / pow3[pos]) % 3;
    };

    auto id = [&](int pos, int d) {
        return 1 + pos * 3 + d;
    };

    int x = 3 * L;
    vector<vector<int>> s(x + 1, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = id(0, digit(j, 0));
    }

    for (int pos = 0; pos < L; ++pos) {
        for (int d = 0; d < 3; ++d) {
            int st = id(pos, d);
            bool storedA = (pos % 2 == 0);
            s[st][0] = storedA ? 1 : 0;

            for (int j = 1; j <= N; ++j) {
                int e = digit(j, pos);

                if (e < d) {
                    s[st][j] = storedA ? -2 : -1;
                } else if (e > d) {
                    s[st][j] = storedA ? -1 : -2;
                } else {
                    if (pos == L - 1) {
                        s[st][j] = -1;
                    } else {
                        s[st][j] = id(pos + 1, digit(j, pos + 1));
                    }
                }
            }
        }
    }

    return s;
}
