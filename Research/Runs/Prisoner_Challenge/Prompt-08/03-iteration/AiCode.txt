#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int INF = 1e9;

    vector<int> dp(N + 1, INF), pre(N + 1, -1), pr(N + 1, -1);
    dp[1] = 0;

    for (int p = 1; p < N; ++p) {
        if (dp[p] == INF) continue;
        int maxr = (N + p - 1) / p;
        for (int r = 2; r <= maxr; ++r) {
            int q = min(N, p * r);
            int nd = dp[p] + r + 1;
            if (nd < dp[q]) {
                dp[q] = nd;
                pre[q] = p;
                pr[q] = r;
            }
        }
    }

    vector<int> rad;
    for (int cur = N; cur > 1; cur = pre[cur]) rad.push_back(pr[cur]);
    reverse(rad.begin(), rad.end());

    int L = (int)rad.size();

    vector<int> w(L, 1);
    for (int i = L - 2; i >= 0; --i) w[i] = w[i + 1] * rad[i + 1];

    vector<int> A(L);
    vector<vector<int>> B(L);

    int id = 0;
    A[0] = 0;
    id = 1;

    for (int k = 0; k < L; ++k) {
        if (k > 0) A[k] = id++;
        B[k].resize(rad[k]);
        for (int d = 0; d < rad[k]; ++d) B[k][d] = id++;
    }

    vector<vector<int>> s(id, vector<int>(N + 1, 0));

    auto digit = [&](int val, int k) -> int {
        int z = val - 1;
        return (z / w[k]) % rad[k];
    };

    for (int k = 0; k < L; ++k) {
        s[A[k]][0] = 0;
        for (int j = 1; j <= N; ++j) {
            int d = digit(j, k);
            s[A[k]][j] = B[k][d];
        }

        for (int d = 0; d < rad[k]; ++d) {
            int st = B[k][d];
            s[st][0] = 1;
            for (int j = 1; j <= N; ++j) {
                int e = digit(j, k);
                if (e < d) s[st][j] = -2;
                else if (e > d) s[st][j] = -1;
                else {
                    if (k + 1 < L) s[st][j] = A[k + 1];
                    else s[st][j] = -1;
                }
            }
        }
    }

    return s;
}
