#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int INF = 1e9;

    vector<int> dp(N + 1, INF), pre(N + 1, -1), rad(N + 1, -1);
    dp[1] = 0;

    for (int p = 1; p < N; ++p) {
        if (dp[p] == INF) continue;
        int lim = (N + p - 1) / p;
        for (int r = 2; r <= lim; ++r) {
            int q = p * r;
            if (q > N) q = N;
            if (dp[q] > dp[p] + r) {
                dp[q] = dp[p] + r;
                pre[q] = p;
                rad[q] = r;
            }
        }
    }

    vector<int> base;
    for (int cur = N; cur != 1; cur = pre[cur]) base.push_back(rad[cur]);
    reverse(base.begin(), base.end());

    int L = (int)base.size();
    vector<int> place(L, 1);
    for (int i = L - 2; i >= 0; --i) place[i] = place[i + 1] * base[i + 1];

    vector<int> offset(L);
    int x = 0;
    for (int i = 0; i < L; ++i) {
        offset[i] = x + 1;
        x += base[i];
    }

    auto digit = [&](int value, int pos) {
        return (value / place[pos]) % base[pos];
    };

    vector<vector<int>> s(x + 1, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        int v = j - 1;
        s[0][j] = offset[0] + digit(v, 0);
    }

    for (int pos = 0; pos < L; ++pos) {
        for (int d = 0; d < base[pos]; ++d) {
            int id = offset[pos] + d;
            bool inspectB = (pos % 2 == 0);
            s[id][0] = inspectB ? 1 : 0;

            for (int j = 1; j <= N; ++j) {
                int v = j - 1;
                int e = digit(v, pos);

                if (e == d) {
                    if (pos + 1 == L) {
                        s[id][j] = 0;
                    } else {
                        s[id][j] = offset[pos + 1] + digit(v, pos + 1);
                    }
                } else if (inspectB) {
                    s[id][j] = (d < e) ? -1 : -2;
                } else {
                    s[id][j] = (e < d) ? -1 : -2;
                }
            }
        }
    }

    return s;
}
