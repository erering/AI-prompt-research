#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int INF = 1e9;

    vector<int> dist(N + 1, INF), pre(N + 1, -1), prer(N + 1, -1);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;

    dist[1] = 0;
    pq.push({0, 1});

    while (!pq.empty()) {
        auto [du, p] = pq.top();
        pq.pop();
        if (du != dist[p]) continue;
        if (p == N) break;

        for (int r = 2; r <= N; r++) {
            int np = (1LL * p * r >= N ? N : int(p * r));
            int nd = du + r;
            if (nd < dist[np]) {
                dist[np] = nd;
                pre[np] = p;
                prer[np] = r;
                pq.push({nd, np});
            }
        }
    }

    vector<int> base;
    for (int cur = N; cur != 1; cur = pre[cur]) {
        base.push_back(prer[cur]);
    }
    reverse(base.begin(), base.end());

    int L = (int)base.size();

    vector<long long> suf(L + 1, 1);
    for (int i = L - 1; i >= 0; i--) suf[i] = suf[i + 1] * base[i];

    vector<int> start(L);
    int x = 0;
    for (int i = 0; i < L; i++) {
        start[i] = x + 1;
        x += base[i];
    }

    vector<vector<int>> s(x + 1, vector<int>(N + 1, 0));

    auto digit = [&](int pos, int val) -> int {
        long long v = val - 1;
        return int((v / suf[pos + 1]) % base[pos]);
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; j++) {
        s[0][j] = start[0] + digit(0, j);
    }

    for (int pos = 0; pos < L; pos++) {
        for (int d = 0; d < base[pos]; d++) {
            int st = start[pos] + d;
            s[st][0] = (pos % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; j++) {
                int e = digit(pos, j);

                if (pos % 2 == 0) {
                    if (d < e) {
                        s[st][j] = -1;
                    } else if (d > e) {
                        s[st][j] = -2;
                    } else {
                        if (pos + 1 == L) s[st][j] = -1;
                        else s[st][j] = start[pos + 1] + digit(pos + 1, j);
                    }
                } else {
                    if (e < d) {
                        s[st][j] = -1;
                    } else if (e > d) {
                        s[st][j] = -2;
                    } else {
                        if (pos + 1 == L) s[st][j] = -1;
                        else s[st][j] = start[pos + 1] + digit(pos + 1, j);
                    }
                }
            }
        }
    }

    return s;
}
