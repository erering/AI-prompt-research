#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int INF = 1e9;

    vector<int> dp(N + 1, INF), preP(N + 1, -1), preR(N + 1, -1);
    dp[1] = 0;

    for (int p = 1; p < N; ++p) {
        if (dp[p] == INF) continue;
        for (int r = 2; r <= N; ++r) {
            long long qll = 1LL * p * r;
            int q = (qll >= N ? N : (int)qll);
            int cand = dp[p] + r;
            if (cand < dp[q]) {
                dp[q] = cand;
                preP[q] = p;
                preR[q] = r;
            }
        }
    }

    vector<int> base;
    for (int cur = N; cur != 1; cur = preP[cur]) {
        base.push_back(preR[cur]);
    }
    reverse(base.begin(), base.end());

    int L = (int)base.size();

    vector<long long> place(L + 1, 1);
    for (int i = L - 1; i >= 0; --i) place[i] = place[i + 1] * base[i];

    vector<int> used(L);
    for (int i = 0; i < L; ++i) {
        used[i] = min<long long>(base[i], (N - 1) / place[i + 1] + 1);
    }

    vector<int> off(L + 1, 1);
    for (int i = 0; i < L; ++i) off[i + 1] = off[i] + used[i];

    int rows = off[L];
    vector<vector<int>> s(rows, vector<int>(N + 1, 0));

    auto digit = [&](int value, int pos) -> int {
        long long v = value - 1;
        return (int)((v / place[pos + 1]) % base[pos]);
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = off[0] + digit(j, 0);
    }

    for (int pos = 0; pos < L; ++pos) {
        for (int d = 0; d < used[pos]; ++d) {
            int id = off[pos] + d;
            s[id][0] = (pos % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; ++j) {
                int e = digit(j, pos);

                if (e < d) {
                    s[id][j] = (pos % 2 == 0 ? -2 : -1);
                } else if (e > d) {
                    s[id][j] = (pos % 2 == 0 ? -1 : -2);
                } else {
                    if (pos + 1 == L) {
                        s[id][j] = -1;
                    } else {
                        s[id][j] = off[pos + 1] + digit(j, pos + 1);
                    }
                }
            }
        }
    }

    return s;
}
