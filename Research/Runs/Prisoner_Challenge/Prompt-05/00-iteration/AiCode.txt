#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    vector<int> pw(1, 1);
    while (pw.back() < N) pw.push_back(pw.back() * 3);
    int L = (int)pw.size() - 1;

    auto dig = [&](int j, int p) {
        int v = j - 1;
        return (v / pw[L - 1 - p]) % 3;
    };

    auto rid = [&](int p, int d) {
        return 1 + 3 * p + d;
    };

    int lastBase = 1 + 3 * (L - 1);
    int x = lastBase + 1;

    vector<vector<int>> s(x + 1, vector<int>(N + 1, -1));

    s[0][0] = 0;
    for (int j = 1; j <= N; j++) {
        int d = dig(j, 0);
        if (L == 1) s[0][j] = lastBase + (d == 2);
        else s[0][j] = rid(0, d);
    }

    for (int p = 0; p < L - 1; p++) {
        for (int d = 0; d < 3; d++) {
            int id = rid(p, d);
            int bag = (p % 2 == 0 ? 1 : 0);
            s[id][0] = bag;

            for (int j = 1; j <= N; j++) {
                int e = dig(j, p);

                if (e != d) {
                    if (bag == 1) {
                        s[id][j] = (e > d ? -1 : -2);
                    } else {
                        s[id][j] = (e > d ? -2 : -1);
                    }
                } else {
                    int nd = dig(j, p + 1);
                    if (p + 1 == L - 1) {
                        s[id][j] = lastBase + (nd == 2);
                    } else {
                        s[id][j] = rid(p + 1, nd);
                    }
                }
            }
        }
    }

    int lastBag = ((L - 1) % 2 == 0 ? 1 : 0);
    for (int g = 0; g < 2; g++) {
        int id = lastBase + g;
        s[id][0] = lastBag;

        for (int j = 1; j <= N; j++) {
            int e = dig(j, L - 1);

            if (lastBag == 1) {
                if (g == 0) {
                    s[id][j] = (e == 0 ? -2 : -1);
                } else {
                    s[id][j] = (e <= 1 ? -2 : -1);
                }
            } else {
                if (g == 0) {
                    s[id][j] = (e == 0 ? -1 : -2);
                } else {
                    s[id][j] = (e <= 1 ? -1 : -2);
                }
            }
        }
    }

    return s;
}
