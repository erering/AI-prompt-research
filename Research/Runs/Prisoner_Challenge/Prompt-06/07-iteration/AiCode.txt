#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int L = 7;
    int P = 1;
    for (int i = 0; i < L; i++) P *= 3;
    int C = (N + P - 1) / P;
    if (C < 1) C = 1;

    int resolver = (C == 3 ? 22 : -1);
    int X = (C == 3 ? 22 : 21);
    vector<vector<int>> s(X + 1, vector<int>(N + 1, -1));

    vector<int> pow3(L);
    pow3[L - 1] = 1;
    for (int i = L - 2; i >= 0; i--) pow3[i] = pow3[i + 1] * 3;

    auto code_of = [&](int v) {
        return (v - 1) / C;
    };

    auto digit = [&](int v, int p) {
        int c = code_of(v);
        return (c / pow3[p]) % 3;
    };

    auto state_id = [&](int p, int d) {
        return 1 + p * 3 + d;
    };

    auto smaller_action = [&](int bag) {
        return bag == 0 ? -1 : -2;
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; j++) {
        s[0][j] = state_id(0, digit(j, 0));
    }

    for (int p = 0; p < L; p++) {
        for (int d = 0; d < 3; d++) {
            int id = state_id(p, d);
            int cur = (p % 2 == 0) ? 1 : 0;
            int prev = 1 - cur;
            s[id][0] = cur;

            for (int j = 1; j <= N; j++) {
                int e = digit(j, p);

                if (e < d) {
                    s[id][j] = smaller_action(cur);
                } else if (e > d) {
                    s[id][j] = smaller_action(prev);
                } else {
                    if (p + 1 < L) {
                        s[id][j] = state_id(p + 1, digit(j, p + 1));
                    } else {
                        int c = code_of(j);
                        int start = c * C + 1;
                        int blockSize = min(C, N - start + 1);
                        int pos = j - start;

                        if (blockSize <= 1) {
                            s[id][j] = smaller_action(cur);
                        } else if (pos == 0) {
                            s[id][j] = smaller_action(cur);
                        } else if (pos == blockSize - 1) {
                            s[id][j] = smaller_action(prev);
                        } else {
                            s[id][j] = resolver;
                        }
                    }
                }
            }
        }
    }

    if (resolver != -1) {
        s[resolver][0] = 0;
        for (int j = 1; j <= N; j++) {
            int pos = (j - 1) % C;
            if (pos == 0) s[resolver][j] = -1;
            else if (pos == 2) s[resolver][j] = -2;
            else s[resolver][j] = -1;
        }
    }

    return s;
}
