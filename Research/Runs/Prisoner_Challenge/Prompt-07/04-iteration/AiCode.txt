#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    vector<int> rad = {3, 3, 3, 3, 4, 4, 4};
    int L = (int)rad.size();

    vector<int> w(L, 1);
    for (int i = L - 2; i >= 0; --i) w[i] = w[i + 1] * rad[i + 1];

    vector<int> pref(L + 1, 0);
    for (int i = 0; i < L; ++i) pref[i + 1] = pref[i] + rad[i];

    auto digit = [&](int value, int pos) {
        int v = value - 1;
        return (v / w[pos]) % rad[pos];
    };

    auto state = [&](int pos, int d) {
        return 1 + pref[pos] + d;
    };

    int S = 1 + pref[L];
    vector<vector<int>> s(S, vector<int>(N + 1, 0));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = state(0, digit(j, 0));
    }

    for (int pos = 0; pos < L; ++pos) {
        for (int d = 0; d < rad[pos]; ++d) {
            int row = state(pos, d);
            int inspect = (pos % 2 == 0 ? 1 : 0);
            s[row][0] = inspect;

            for (int j = 1; j <= N; ++j) {
                int cur = digit(j, pos);

                if (cur == d) {
                    if (pos == L - 1) {
                        s[row][j] = -1;
                    } else {
                        s[row][j] = state(pos + 1, digit(j, pos + 1));
                    }
                } else {
                    if (inspect == 1) {
                        s[row][j] = (d < cur ? -1 : -2);
                    } else {
                        s[row][j] = (cur < d ? -1 : -2);
                    }
                }
            }
        }
    }

    return s;
}
