#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    int L = 1;
    int pw = 3;
    while (pw < N) {
        pw *= 3;
        L++;
    }

    vector<int> pow3(L);
    pow3[0] = 1;
    for (int i = 1; i < L; i++) pow3[i] = pow3[i - 1] * 3;

    auto digit = [&](int value, int pos) {
        int x = value - 1;
        int place = pow3[L - 1 - pos];
        return (x / place) % 3;
    };

    auto Astate = [&](int pos) {
        return pos;
    };

    auto Bstate = [&](int pos, int d) {
        return L + pos * 3 + d;
    };

    int S = L + 3 * L;
    vector<vector<int>> s(S, vector<int>(N + 1, 0));

    for (int pos = 0; pos < L; pos++) {
        int arow = Astate(pos);
        s[arow][0] = 0;

        for (int j = 1; j <= N; j++) {
            int d = digit(j, pos);
            s[arow][j] = Bstate(pos, d);
        }

        for (int da = 0; da < 3; da++) {
            int brow = Bstate(pos, da);
            s[brow][0] = 1;

            for (int j = 1; j <= N; j++) {
                int db = digit(j, pos);

                if (da < db) {
                    s[brow][j] = -1;
                } else if (da > db) {
                    s[brow][j] = -2;
                } else {
                    if (pos + 1 < L) {
                        s[brow][j] = Astate(pos + 1);
                    } else {
                        s[brow][j] = -1;
                    }
                }
            }
        }
    }

    return s;
}
