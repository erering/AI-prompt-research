#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    const int B = 10;
    const int L = 4;

    auto digit = [&](int v, int p) {
        while (p--) v /= B;
        return v % B;
    };

    vector<vector<int>> bid(L, vector<int>(B));
    vector<int> aid(L, -1);

    int nxt = 1;
    for (int p = 0; p < L; ++p) {
        for (int d = 0; d < B; ++d) {
            bid[p][d] = nxt++;
        }
    }
    for (int p = 0; p < L - 1; ++p) {
        aid[p] = nxt++;
    }

    vector<vector<int>> s(nxt, vector<int>(N + 1, -1));

    s[0][0] = 0;
    for (int j = 1; j <= N; ++j) {
        s[0][j] = bid[L - 1][digit(j, L - 1)];
    }

    for (int p = 0; p < L - 1; ++p) {
        int id = aid[p];
        s[id][0] = 0;
        for (int j = 1; j <= N; ++j) {
            s[id][j] = bid[p][digit(j, p)];
        }
    }

    for (int p = 0; p < L; ++p) {
        for (int d = 0; d < B; ++d) {
            int id = bid[p][d];
            s[id][0] = 1;
            for (int j = 1; j <= N; ++j) {
                int e = digit(j, p);
                if (d < e) s[id][j] = -1;
                else if (d > e) s[id][j] = -2;
                else {
                    if (p == 0) s[id][j] = -1;
                    else s[id][j] = aid[p - 1];
                }
            }
        }
    }

    return s;
}
