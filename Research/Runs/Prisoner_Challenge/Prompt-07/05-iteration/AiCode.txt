#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> devise_strategy(int N) {
    vector<int> base(8, 3);
    int L = (int)base.size();

    vector<int> offset(L);
    int cur = 1;
    for (int i = 0; i < L; i++) {
        offset[i] = cur;
        cur += base[i];
    }

    vector<vector<int>> s(cur, vector<int>(N + 1, 0));

    auto digit = [&](int x, int pos) {
        x--;
        vector<int> d(L);
        for (int i = L - 1; i >= 0; i--) {
            d[i] = x % base[i];
            x /= base[i];
        }
        return d[pos];
    };

    s[0][0] = 0;
    for (int j = 1; j <= N; j++) {
        s[0][j] = offset[0] + digit(j, 0);
    }

    for (int p = 0; p < L; p++) {
        for (int v = 0; v < base[p]; v++) {
            int id = offset[p] + v;
            s[id][0] = (p % 2 == 0 ? 1 : 0);

            for (int j = 1; j <= N; j++) {
                int w = digit(j, p);

                if (p % 2 == 0) {
                    if (v < w) {
                        s[id][j] = -1;
                    } else if (v > w) {
                        s[id][j] = -2;
                    } else {
                        if (p + 1 == L) s[id][j] = 0;
                        else s[id][j] = offset[p + 1] + digit(j, p + 1);
                    }
                } else {
                    if (w < v) {
                        s[id][j] = -1;
                    } else if (w > v) {
                        s[id][j] = -2;
                    } else {
                        if (p + 1 == L) s[id][j] = 0;
                        else s[id][j] = offset[p + 1] + digit(j, p + 1);
                    }
                }
            }
        }
    }

    return s;
}
