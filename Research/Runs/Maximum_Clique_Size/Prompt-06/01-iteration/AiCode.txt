#include <bits/stdc++.h>
using namespace std;

#pragma GCC optimize("O3,unroll-loops")

struct FastScanner {
    static const int BUFSIZE = 1 << 20;
    int idx = 0, size = 0;
    char buf[BUFSIZE];

    inline char getChar() {
        if (idx >= size) {
            size = (int)fread(buf, 1, BUFSIZE, stdin);
            idx = 0;
            if (size == 0) return 0;
        }
        return buf[idx++];
    }

    int nextInt() {
        char c;
        do c = getChar(); while (c && (c < '0' || c > '9'));
        int x = 0;
        while (c >= '0' && c <= '9') {
            x = x * 10 + (c - '0');
            c = getChar();
        }
        return x;
    }

    int nextBit() {
        char c;
        do c = getChar(); while (c && c != '0' && c != '1');
        return c == '1';
    }
};

int N, W;
uint64_t LAST_MASK;
vector<uint64_t> adjBits;
vector<vector<int>> compAdj;
vector<uint64_t> tmpU, tmpQ, tmpMatched;
int bestAns = 0;

inline uint64_t bitOf(int v) { return 1ULL << (v & 63); }
inline bool getBitVec(const vector<uint64_t>& a, int v) { return (a[v >> 6] >> (v & 63)) & 1ULL; }
inline bool getBitPtr(const uint64_t* a, int v) { return (a[v >> 6] >> (v & 63)) & 1ULL; }
inline void clearBitVec(vector<uint64_t>& a, int v) { a[v >> 6] &= ~bitOf(v); }
inline void setBitFlat(int i, int j) { adjBits[(size_t)i * W + (j >> 6)] |= bitOf(j); }

inline int popcountSet(const vector<uint64_t>& a) {
    int s = 0;
    for (int i = 0; i < W; ++i) s += __builtin_popcountll(a[i]);
    return s;
}

inline int firstBit(const vector<uint64_t>& a) {
    for (int i = 0; i < W; ++i) {
        uint64_t x = a[i];
        if (x) return (i << 6) + __builtin_ctzll(x);
    }
    return -1;
}

int greedyInitialClique() {
    if (N == 0) return 0;
    vector<int> deg(N);
    vector<unsigned char> alive(N, 1);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;
    for (int i = 0; i < N; ++i) {
        deg[i] = (int)compAdj[i].size();
        pq.push({deg[i], i});
    }

    int rem = N, ans = 0;

    auto removeVertex = [&](int x) {
        if (!alive[x]) return;
        alive[x] = 0;
        --rem;
        for (int y : compAdj[x]) {
            if (alive[y]) {
                --deg[y];
                pq.push({deg[y], y});
            }
        }
    };

    vector<int> todel;
    while (rem > 0) {
        int v = -1;
        while (!pq.empty()) {
            auto [d, x] = pq.top();
            pq.pop();
            if (alive[x] && deg[x] == d) {
                v = x;
                break;
            }
        }
        if (v < 0) break;
        ++ans;
        todel.clear();
        todel.push_back(v);
        for (int u : compAdj[v]) if (alive[u]) todel.push_back(u);
        for (int x : todel) removeVertex(x);
    }
    return ans;
}

int greedyMatching(const vector<uint64_t>& P) {
    fill(tmpMatched.begin(), tmpMatched.end(), 0ULL);
    int m = 0;

    for (int wi = 0; wi < W; ++wi) {
        uint64_t word = P[wi];
        while (word) {
            int b = __builtin_ctzll(word);
            int v = (wi << 6) + b;
            word &= word - 1;
            if (v >= N) break;
            if (tmpMatched[v >> 6] & bitOf(v)) continue;

            for (int u : compAdj[v]) {
                if ((P[u >> 6] & bitOf(u)) && !(tmpMatched[u >> 6] & bitOf(u))) {
                    tmpMatched[v >> 6] |= bitOf(v);
                    tmpMatched[u >> 6] |= bitOf(u);
                    ++m;
                    break;
                }
            }
        }
    }
    return m;
}

void colorSort(const vector<uint64_t>& P, vector<int>& order, vector<int>& colors) {
    tmpU = P;
    order.clear();
    colors.clear();

    int color = 0;
    while (true) {
        int any = firstBit(tmpU);
        if (any < 0) break;

        ++color;
        tmpQ = tmpU;

        while (true) {
            int v = firstBit(tmpQ);
            if (v < 0) break;

            order.push_back(v);
            colors.push_back(color);

            uint64_t bv = bitOf(v);
            tmpU[v >> 6] &= ~bv;
            tmpQ[v >> 6] &= ~bv;

            const uint64_t* av = &adjBits[(size_t)v * W];
            for (int w = 0; w < W; ++w) tmpQ[w] &= ~av[w];
            tmpQ[W - 1] &= LAST_MASK;
        }
    }
}

void expand(vector<uint64_t> P, int cur) {
    int k = popcountSet(P);
    if (k == 0) {
        if (cur > bestAns) bestAns = cur;
        return;
    }

    if (cur + k <= bestAns) return;

    int match = greedyMatching(P);
    if (match == 0) {
        if (cur + k > bestAns) bestAns = cur + k;
        return;
    }
    if (cur + k - match <= bestAns) return;

    vector<int> order, colors;
    order.reserve(k);
    colors.reserve(k);
    colorSort(P, order, colors);

    for (int i = (int)order.size() - 1; i >= 0; --i) {
        if (cur + colors[i] <= bestAns) return;

        int v = order[i];
        if (!(P[v >> 6] & bitOf(v))) continue;

        const uint64_t* av = &adjBits[(size_t)v * W];
        vector<uint64_t> P2(W);
        for (int w = 0; w < W; ++w) P2[w] = P[w] & av[w];
        P2[W - 1] &= LAST_MASK;

        int pk = popcountSet(P2);
        if (cur + 1 + pk > bestAns) {
            expand(std::move(P2), cur + 1);
        } else if (cur + 1 > bestAns) {
            bestAns = cur + 1;
        }

        clearBitVec(P, v);
        if (cur + popcountSet(P) <= bestAns) return;
    }

    if (cur > bestAns) bestAns = cur;
}

int main() {
    FastScanner fs;
    N = fs.nextInt();
    if (N <= 0) {
        printf("0\n");
        return 0;
    }

    W = (N + 63) >> 6;
    LAST_MASK = (N & 63) ? ((1ULL << (N & 63)) - 1ULL) : ~0ULL;

    adjBits.assign((size_t)N * W, 0ULL);
    compAdj.assign(N, {});

    for (int i = 0; i < N; ++i) {
        for (int j = 0; j < N; ++j) {
            int b = fs.nextBit();
            if (i == j) continue;
            if (b) {
                setBitFlat(i, j);
            } else if (i < j) {
                compAdj[i].push_back(j);
                compAdj[j].push_back(i);
            }
        }
    }

    for (int i = 0; i < N; ++i) {
        adjBits[(size_t)i * W + (i >> 6)] &= ~bitOf(i);
        adjBits[(size_t)i * W + W - 1] &= LAST_MASK;
    }

    bestAns = greedyInitialClique();

    tmpU.assign(W, 0ULL);
    tmpQ.assign(W, 0ULL);
    tmpMatched.assign(W, 0ULL);

    vector<uint64_t> root(W, ~0ULL);
    root[W - 1] = LAST_MASK;

    expand(std::move(root), 0);

    printf("%d\n", bestAns);
    return 0;
}
