#include <bits/stdc++.h>
using namespace std;

class FastInput {
    static constexpr size_t BUFSIZE = 1 << 20;
    char buf[BUFSIZE];
    size_t idx = 0, size = 0;

    int getChar() {
        if (idx >= size) {
            size = fread(buf, 1, BUFSIZE, stdin);
            idx = 0;
            if (size == 0) return EOF;
        }
        return buf[idx++];
    }

public:
    int nextInt() {
        int c;
        do {
            c = getChar();
        } while (c != EOF && isspace(c));

        int sign = 1;
        if (c == '-') {
            sign = -1;
            c = getChar();
        }

        int x = 0;
        while (c != EOF && isdigit(c)) {
            x = x * 10 + (c - '0');
            c = getChar();
        }
        return x * sign;
    }

    int nextBit() {
        int c;
        do {
            c = getChar();
        } while (c != EOF && c != '0' && c != '1');
        if (c == EOF) return 0;
        return c - '0';
    }
};

class MaxCliqueSolver {
    int n, W;
    uint64_t lastMask;
    vector<vector<uint64_t>> comp;
    vector<vector<int>> adj;
    int best = 0;

    static int popcnt64(uint64_t x) {
        return __builtin_popcountll(x);
    }

    int popcountBits(const vector<uint64_t>& b) const {
        int s = 0;
        for (uint64_t x : b) s += popcnt64(x);
        return s;
    }

    bool emptyBits(const vector<uint64_t>& b) const {
        for (uint64_t x : b) if (x) return false;
        return true;
    }

    static int firstBit(uint64_t x) {
        return __builtin_ctzll(x);
    }

    bool testBit(const vector<uint64_t>& b, int v) const {
        return (b[v >> 6] >> (v & 63)) & 1ULL;
    }

    void clearBit(vector<uint64_t>& b, int v) const {
        b[v >> 6] &= ~(1ULL << (v & 63));
    }

    int greedyInitial() const {
        vector<int> deg(n);
        vector<char> active(n, 1);
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;

        for (int i = 0; i < n; ++i) {
            deg[i] = (int)adj[i].size();
            pq.push({deg[i], i});
        }

        int res = 0;
        while (!pq.empty()) {
            auto [d, v] = pq.top();
            pq.pop();

            if (!active[v] || d != deg[v]) continue;

            ++res;

            vector<int> rem;
            rem.push_back(v);
            for (int u : adj[v]) {
                if (active[u]) rem.push_back(u);
            }

            for (int x : rem) {
                if (!active[x]) continue;
                active[x] = 0;
                for (int y : adj[x]) {
                    if (active[y]) {
                        --deg[y];
                        pq.push({deg[y], y});
                    }
                }
            }
        }

        return res;
    }

    void colorSort(const vector<uint64_t>& P, vector<int>& order, vector<int>& colors) const {
        order.clear();
        colors.clear();

        vector<uint64_t> U = P;
        vector<uint64_t> Q(W);

        int color = 0;
        while (!emptyBits(U)) {
            ++color;
            Q = U;

            while (!emptyBits(Q)) {
                int wi = 0;
                while (wi < W && Q[wi] == 0) ++wi;
                int v = (wi << 6) + firstBit(Q[wi]);

                order.push_back(v);
                colors.push_back(color);

                clearBit(U, v);
                clearBit(Q, v);

                for (int k = 0; k < W; ++k) {
                    Q[k] &= comp[v][k] & U[k];
                }
            }
        }
    }

    void expand(vector<uint64_t> P, int depth) {
        int pc = popcountBits(P);

        if (pc == 0) {
            if (depth > best) best = depth;
            return;
        }

        if (depth + pc <= best) return;

        vector<int> order;
        vector<int> colors;
        colorSort(P, order, colors);

        for (int i = (int)order.size() - 1; i >= 0; --i) {
            if (depth + colors[i] <= best) return;

            int v = order[i];
            if (!testBit(P, v)) continue;

            vector<uint64_t> NP(W);
            for (int k = 0; k < W; ++k) {
                NP[k] = P[k] & ~comp[v][k];
            }
            NP[W - 1] &= lastMask;

            int npc = popcountBits(NP);
            if (npc == 0) {
                if (depth + 1 > best) best = depth + 1;
            } else if (depth + 1 + npc > best) {
                expand(std::move(NP), depth + 1);
            }

            clearBit(P, v);
        }
    }

public:
    explicit MaxCliqueSolver(vector<vector<int>> localAdj) : adj(std::move(localAdj)) {
        n = (int)adj.size();
        W = (n + 63) >> 6;

        int rem = n & 63;
        lastMask = rem == 0 ? ~0ULL : ((1ULL << rem) - 1ULL);

        comp.assign(n, vector<uint64_t>(W, 0));

        for (int i = 0; i < n; ++i) {
            comp[i][i >> 6] |= 1ULL << (i & 63);
        }

        for (int i = 0; i < n; ++i) {
            for (int j : adj[i]) {
                comp[i][j >> 6] |= 1ULL << (j & 63);
            }
        }
    }

    int solve() {
        if (n == 0) return 0;
        if (n == 1) return 1;

        best = greedyInitial();

        vector<uint64_t> P(W, ~0ULL);
        P[W - 1] &= lastMask;

        expand(std::move(P), 0);
        return best;
    }
};

int hopcroftKarpAlpha(const vector<vector<int>>& adj) {
    int n = (int)adj.size();
    vector<int> color(n, -1);

    for (int s = 0; s < n; ++s) {
        if (color[s] != -1) continue;

        queue<int> q;
        color[s] = 0;
        q.push(s);

        while (!q.empty()) {
            int v = q.front();
            q.pop();

            for (int u : adj[v]) {
                if (color[u] == -1) {
                    color[u] = color[v] ^ 1;
                    q.push(u);
                } else if (color[u] == color[v]) {
                    return -1;
                }
            }
        }
    }

    vector<int> pairU(n, -1), pairV(n, -1), dist(n);
    const int INF = 1e9;

    auto bfs = [&]() {
        queue<int> q;
        bool found = false;

        for (int u = 0; u < n; ++u) {
            if (color[u] == 0 && pairU[u] == -1) {
                dist[u] = 0;
                q.push(u);
            } else {
                dist[u] = INF;
            }
        }

        while (!q.empty()) {
            int u = q.front();
            q.pop();

            for (int v : adj[u]) {
                if (color[v] != 1) continue;

                int pu = pairV[v];
                if (pu == -1) {
                    found = true;
                } else if (dist[pu] == INF) {
                    dist[pu] = dist[u] + 1;
                    q.push(pu);
                }
            }
        }

        return found;
    };

    function<bool(int)> dfs = [&](int u) {
        for (int v : adj[u]) {
            if (color[v] != 1) continue;

            int pu = pairV[v];
            if (pu == -1 || (dist[pu] == dist[u] + 1 && dfs(pu))) {
                pairU[u] = v;
                pairV[v] = u;
                return true;
            }
        }

        dist[u] = INF;
        return false;
    };

    int matching = 0;
    while (bfs()) {
        for (int u = 0; u < n; ++u) {
            if (color[u] == 0 && pairU[u] == -1 && dfs(u)) {
                ++matching;
            }
        }
    }

    return n - matching;
}

int main() {
    FastInput in;

    int N = in.nextInt();
    vector<vector<int>> graph(N);

    for (int i = 0; i < N; ++i) {
        for (int j = 0; j < N; ++j) {
            int x = in.nextBit();
            if (i < j && x == 0) {
                graph[i].push_back(j);
                graph[j].push_back(i);
            }
        }
    }

    vector<char> active(N, 1);
    vector<int> deg(N);
    queue<int> q;

    for (int i = 0; i < N; ++i) {
        deg[i] = (int)graph[i].size();
        if (deg[i] <= 1) q.push(i);
    }

    int answer = 0;

    auto removeVertex = [&](int v) {
        if (!active[v]) return;
        active[v] = 0;

        for (int u : graph[v]) {
            if (active[u]) {
                --deg[u];
                if (deg[u] <= 1) q.push(u);
            }
        }
    };

    while (!q.empty()) {
        int v = q.front();
        q.pop();

        if (!active[v] || deg[v] > 1) continue;

        ++answer;

        if (deg[v] == 0) {
            removeVertex(v);
        } else {
            int u = -1;
            for (int x : graph[v]) {
                if (active[x]) {
                    u = x;
                    break;
                }
            }

            removeVertex(v);
            if (u != -1) removeVertex(u);
        }
    }

    vector<char> visited(N, 0);
    vector<int> pos(N, -1);

    for (int s = 0; s < N; ++s) {
        if (!active[s] || visited[s]) continue;

        vector<int> verts;
        queue<int> bfs;
        visited[s] = 1;
        bfs.push(s);

        while (!bfs.empty()) {
            int v = bfs.front();
            bfs.pop();
            verts.push_back(v);

            for (int u : graph[v]) {
                if (active[u] && !visited[u]) {
                    visited[u] = 1;
                    bfs.push(u);
                }
            }
        }

        int m = (int)verts.size();
        if (m == 1) {
            ++answer;
            continue;
        }

        sort(verts.begin(), verts.end(), [&](int a, int b) {
            if (deg[a] != deg[b]) return deg[a] < deg[b];
            return a < b;
        });

        for (int i = 0; i < m; ++i) pos[verts[i]] = i;

        vector<vector<int>> localAdj(m);

        for (int i = 0; i < m; ++i) {
            int gv = verts[i];
            for (int gu : graph[gv]) {
                int j = pos[gu];
                if (j >= 0 && j > i) {
                    localAdj[i].push_back(j);
                    localAdj[j].push_back(i);
                }
            }
        }

        for (int v : verts) pos[v] = -1;

        int bipAlpha = hopcroftKarpAlpha(localAdj);
        if (bipAlpha >= 0) {
            answer += bipAlpha;
        } else {
            MaxCliqueSolver solver(std::move(localAdj));
            answer += solver.solve();
        }
    }

    printf("%d\n", answer);
    return 0;
}
