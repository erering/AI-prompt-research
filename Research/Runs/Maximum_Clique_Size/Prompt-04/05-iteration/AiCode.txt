#include <bits/stdc++.h>
using namespace std;

using ull = uint64_t;

struct FastScanner {
    static const int BUFSIZE = 1 << 20;
    int idx = 0, size = 0;
    char buf[BUFSIZE];

    inline int getChar() {
        if (idx >= size) {
            size = (int)fread(buf, 1, BUFSIZE, stdin);
            idx = 0;
            if (size == 0) return EOF;
        }
        return buf[idx++];
    }

    int readInt() {
        int c, x = 0;
        do {
            c = getChar();
            if (c == EOF) return 0;
        } while (c < '0' || c > '9');
        while (c >= '0' && c <= '9') {
            x = x * 10 + (c - '0');
            c = getChar();
        }
        return x;
    }

    int readBit() {
        int c;
        do {
            c = getChar();
            if (c == EOF) return 0;
        } while (c != '0' && c != '1');
        return c - '0';
    }
};

static int M, W;
static ull LAST_MASK;
static vector<vector<ull>> adj;
static int bestAns = 0;

inline bool bitTest(const vector<ull>& b, int v) {
    return (b[v >> 6] >> (v & 63)) & 1ULL;
}

inline void bitSet(vector<ull>& b, int v) {
    b[v >> 6] |= 1ULL << (v & 63);
}

inline void bitClear(vector<ull>& b, int v) {
    b[v >> 6] &= ~(1ULL << (v & 63));
}

inline int firstBit(const vector<ull>& b) {
    for (int i = 0; i < W; ++i) {
        ull x = b[i];
        if (x) return (i << 6) + __builtin_ctzll(x);
    }
    return -1;
}

inline bool anyBit(const vector<ull>& b) {
    for (ull x : b) if (x) return true;
    return false;
}

inline int popcountBits(const vector<ull>& b) {
    int s = 0;
    for (ull x : b) s += __builtin_popcountll(x);
    return s;
}

inline int popcountIntersection(const vector<ull>& a, const vector<ull>& b) {
    int s = 0;
    for (int i = 0; i < W; ++i) s += __builtin_popcountll(a[i] & b[i]);
    return s;
}

void colorSort(const vector<ull>& P, vector<int>& order, vector<int>& color) {
    order.clear();
    color.clear();

    vector<ull> U = P;
    vector<ull> Q(W);

    int c = 0;
    while (anyBit(U)) {
        ++c;
        Q = U;
        while (true) {
            int v = firstBit(Q);
            if (v < 0 || v >= M) break;

            order.push_back(v);
            color.push_back(c);

            ull m = 1ULL << (v & 63);
            U[v >> 6] &= ~m;
            Q[v >> 6] &= ~m;

            const vector<ull>& av = adj[v];
            for (int i = 0; i < W; ++i) Q[i] &= ~av[i];
            Q[W - 1] &= LAST_MASK;
        }
    }
}

int greedyFrom(int start) {
    vector<ull> P(W);
    int sz = 0;

    if (start >= 0) {
        ++sz;
        P = adj[start];
    } else {
        for (int i = 0; i < W; ++i) P[i] = ~0ULL;
        P[W - 1] = LAST_MASK;
    }

    while (anyBit(P)) {
        int bv = -1, bs = -1;

        for (int wi = 0; wi < W; ++wi) {
            ull x = P[wi];
            while (x) {
                int b = __builtin_ctzll(x);
                int v = (wi << 6) + b;
                if (v < M) {
                    int sc = popcountIntersection(P, adj[v]);
                    if (sc > bs) {
                        bs = sc;
                        bv = v;
                    }
                }
                x &= x - 1;
            }
        }

        if (bv < 0) break;
        ++sz;
        for (int i = 0; i < W; ++i) P[i] &= adj[bv][i];
    }

    return sz;
}

int greedyInitial() {
    if (M == 0) return 0;

    int ans = greedyFrom(-1);

    vector<pair<int,int>> degs;
    degs.reserve(M);
    for (int v = 0; v < M; ++v) degs.push_back({popcountBits(adj[v]), v});
    sort(degs.rbegin(), degs.rend());

    int lim = min(M, 24);
    for (int i = 0; i < lim; ++i) {
        ans = max(ans, greedyFrom(degs[i].second));
    }
    return ans;
}

void expand(vector<ull> P, int cur) {
    int cnt = popcountBits(P);
    if (cur + cnt <= bestAns) return;
    if (cnt == 0) {
        if (cur > bestAns) bestAns = cur;
        return;
    }

    vector<int> order;
    vector<int> color;
    order.reserve(cnt);
    color.reserve(cnt);
    colorSort(P, order, color);

    for (int i = (int)order.size() - 1; i >= 0; --i) {
        if (cur + color[i] <= bestAns) return;

        int v = order[i];
        if (!bitTest(P, v)) continue;

        vector<ull> NP(W);
        bool nonempty = false;
        for (int j = 0; j < W; ++j) {
            NP[j] = P[j] & adj[v][j];
            nonempty |= NP[j] != 0;
        }

        if (nonempty) {
            expand(std::move(NP), cur + 1);
        } else {
            if (cur + 1 > bestAns) bestAns = cur + 1;
        }

        bitClear(P, v);
    }
}

int main() {
    FastScanner fs;
    int N = fs.readInt();

    if (N <= 0) {
        printf("0\n");
        return 0;
    }

    int W0 = (N + 63) >> 6;
    vector<vector<ull>> input(N, vector<ull>(W0, 0));

    for (int i = 0; i < N; ++i) {
        for (int j = 0; j < N; ++j) {
            int b = fs.readBit();
            if (b && i != j) input[i][j >> 6] |= 1ULL << (j & 63);
        }
    }

    vector<vector<int>> comp(N);
    vector<int> deg(N, 0);

    for (int i = 0; i < N; ++i) {
        for (int j = i + 1; j < N; ++j) {
            bool e = ((input[i][j >> 6] >> (j & 63)) & 1ULL) ||
                     ((input[j][i >> 6] >> (i & 63)) & 1ULL);
            if (!e) {
                comp[i].push_back(j);
                comp[j].push_back(i);
                ++deg[i];
                ++deg[j];
            }
        }
    }

    vector<vector<ull>>().swap(input);

    vector<char> active(N, 1);
    vector<int> q;
    q.reserve(N);
    for (int i = 0; i < N; ++i) {
        if (deg[i] <= 1) q.push_back(i);
    }

    int fixed = 0;
    size_t head = 0;

    while (head < q.size()) {
        int v = q[head++];
        if (!active[v]) continue;

        int d = 0, lone = -1;
        for (int u : comp[v]) {
            if (active[u]) {
                ++d;
                lone = u;
                if (d > 1) break;
            }
        }
        deg[v] = d;

        if (d > 1) continue;

        if (d == 0) {
            active[v] = 0;
            ++fixed;
        } else {
            int u = lone;
            if (!active[u]) {
                q.push_back(v);
                continue;
            }

            active[v] = 0;
            active[u] = 0;
            ++fixed;

            for (int x : comp[u]) {
                if (active[x]) {
                    --deg[x];
                    if (deg[x] <= 1) q.push_back(x);
                }
            }
        }
    }

    vector<int> residual;
    residual.reserve(N);
    vector<int> cdeg(N, 0);

    for (int v = 0; v < N; ++v) {
        if (active[v]) {
            int d = 0;
            for (int u : comp[v]) if (active[u]) ++d;
            cdeg[v] = d;
            residual.push_back(v);
        }
    }

    if (residual.empty()) {
        printf("%d\n", fixed);
        return 0;
    }

    sort(residual.begin(), residual.end(), [&](int a, int b) {
        if (cdeg[a] != cdeg[b]) return cdeg[a] > cdeg[b];
        return a < b;
    });

    M = (int)residual.size();
    W = (M + 63) >> 6;
    int rem = M & 63;
    LAST_MASK = rem ? ((1ULL << rem) - 1ULL) : ~0ULL;

    vector<int> pos(N, -1);
    for (int i = 0; i < M; ++i) pos[residual[i]] = i;

    adj.assign(M, vector<ull>(W, 0));

    for (int i = 0; i < M; ++i) {
        for (int w = 0; w < W; ++w) adj[i][w] = ~0ULL;
        adj[i][W - 1] = LAST_MASK;
        bitClear(adj[i], i);

        int old = residual[i];
        for (int nb : comp[old]) {
            if (active[nb]) {
                int p = pos[nb];
                if (p >= 0) bitClear(adj[i], p);
            }
        }
    }

    vector<vector<int>>().swap(comp);

    bestAns = greedyInitial();

    vector<ull> all(W, ~0ULL);
    all[W - 1] = LAST_MASK;

    expand(std::move(all), 0);

    printf("%d\n", fixed + bestAns);
    return 0;
}
