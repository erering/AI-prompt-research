#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <queue>
#include <string>
#include <limits>
#include <cstdint>

using ull = unsigned long long;

std::string toString(__int128 x) {
    if (x == 0) return "0";
    bool neg = x < 0;
    if (neg) x = -x;
    std::string s;
    while (x > 0) {
        s.push_back(char('0' + x % 10));
        x /= 10;
    }
    if (neg) s.push_back('-');
    std::reverse(s.begin(), s.end());
    return s;
}

bool solveDP(const std::vector<ull>& vals, __int128 target128, __int128& best) {
    const __int128 MAX_TARGET = 200000000;
    if (target128 < 0 || target128 > MAX_TARGET) return false;

    size_t target = (size_t)target128;
    size_t words = target / 64 + 1;
    std::vector<uint64_t> bits(words, 0);
    bits[0] = 1;

    size_t reach = 0;
    size_t targetWord = target / 64;
    int targetBit = int(target % 64);

    uint64_t lastMask = ~0ULL;
    int usedBits = int((target + 1) % 64);
    if (usedBits != 0) lastMask = (1ULL << usedBits) - 1;

    for (ull av : vals) {
        if (av > (ull)target) continue;
        size_t shift = (size_t)av;
        size_t oldReach = reach;
        reach = std::min(target, reach + shift);

        size_t wshift = shift / 64;
        int bshift = int(shift % 64);
        size_t highWord = reach / 64;

        if (bshift == 0) {
            for (size_t i = highWord + 1; i-- > wshift;) {
                bits[i] |= bits[i - wshift];
            }
        } else {
            for (size_t i = highWord + 1; i-- > wshift + 1;) {
                bits[i] |= (bits[i - wshift] << bshift) | (bits[i - wshift - 1] >> (64 - bshift));
            }
            if (wshift <= highWord) {
                bits[wshift] |= bits[0] << bshift;
            }
        }

        (void)oldReach;
        bits.back() &= lastMask;

        if ((bits[targetWord] >> targetBit) & 1ULL) {
            best = target128;
            return true;
        }
    }

    for (size_t i = targetWord + 1; i-- > 0;) {
        uint64_t w = bits[i];
        if (i == targetWord) {
            int keep = int(target % 64) + 1;
            if (keep < 64) w &= ((1ULL << keep) - 1);
        }
        if (w) {
            int pos = 63 - __builtin_clzll(w);
            best = (__int128)i * 64 + pos;
            return true;
        }
    }

    best = 0;
    return true;
}

bool solveMITM(const std::vector<ull>& vals, __int128 target128, __int128& best) {
    if (vals.size() > 47) return false;
    if (target128 > (__int128)std::numeric_limits<ull>::max()) return false;

    ull target = (ull)target128;
    int n = (int)vals.size();
    int n1 = n / 2;
    int n2 = n - n1;

    std::vector<ull> sums;
    if (n1 <= 24) sums.reserve(size_t(1) << n1);
    sums.push_back(0);

    for (int i = 0; i < n1; ++i) {
        ull x = vals[i];
        size_t sz = sums.size();
        for (size_t j = 0; j < sz; ++j) {
            if (sums[j] <= target - x) sums.push_back(sums[j] + x);
        }
    }

    std::sort(sums.begin(), sums.end());

    ull bestU = 0;

    auto dfs = [&](auto&& self, int idx, ull cur) -> void {
        if (bestU == target) return;
        if (idx == n2) {
            ull cap = target - cur;
            auto it = std::upper_bound(sums.begin(), sums.end(), cap);
            if (it != sums.begin()) {
                --it;
                ull cand = cur + *it;
                if (cand > bestU) bestU = cand;
            }
            return;
        }

        self(self, idx + 1, cur);

        ull x = vals[n1 + idx];
        if (cur <= target - x) self(self, idx + 1, cur + x);
    };

    dfs(dfs, 0, 0);
    best = bestU;
    return true;
}

struct BranchSolver {
    std::vector<ull> v;
    std::vector<__int128> suffix;
    __int128 target;
    __int128 best;
    int n;

    void dfs(int i, __int128 cur) {
        if (cur > best) {
            best = cur;
            if (best == target) return;
        }

        if (i >= n) return;
        if (cur + suffix[i] <= best) return;
        if (best == target) return;

        __int128 with = cur + (__int128)v[i];
        if (with <= target) dfs(i + 1, with);

        if (best == target) return;
        if (cur + suffix[i + 1] > best) dfs(i + 1, cur);
    }

    __int128 solve(const std::vector<ull>& vals, __int128 t, __int128 initialBest) {
        v = vals;
        target = t;
        best = initialBest;
        n = (int)v.size();

        suffix.assign(n + 1, 0);
        for (int i = n - 1; i >= 0; --i) suffix[i] = suffix[i + 1] + (__int128)v[i];

        dfs(0, 0);
        return best;
    }
};

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int N;
    if (!(std::cin >> N)) return 0;

    std::vector<ull> a(N);
    ull g = 0;

    for (int i = 0; i < N; ++i) {
        std::cin >> a[i];
        g = std::gcd(g, a[i]);
    }

    if (N == 0) {
        std::cout << 0 << '\n';
        return 0;
    }

    for (ull& x : a) x /= g;

    __int128 total = 0;
    for (ull x : a) total += (__int128)x;

    __int128 target = total / 2;

    std::vector<ull> vals;
    vals.reserve(a.size());
    __int128 candidateSum = 0;

    for (ull x : a) {
        if ((__int128)x <= target) {
            vals.push_back(x);
            candidateSum += (__int128)x;
        }
    }

    __int128 best = 0;

    if (candidateSum <= target) {
        best = candidateSum;
        __int128 ans = (total - 2 * best) * (__int128)g;
        std::cout << toString(ans) << '\n';
        return 0;
    }

    std::sort(vals.begin(), vals.end(), std::greater<ull>());

    __int128 greedy = 0;
    for (ull x : vals) {
        if (greedy + (__int128)x <= target) greedy += (__int128)x;
    }
    best = std::max(best, greedy);

    {
        __int128 s1 = 0, s2 = 0;
        std::vector<ull> sortedAll = a;
        std::sort(sortedAll.begin(), sortedAll.end(), std::greater<ull>());
        for (ull x : sortedAll) {
            if (s1 <= s2) s1 += (__int128)x;
            else s2 += (__int128)x;
        }
        __int128 diff = s1 >= s2 ? s1 - s2 : s2 - s1;
        __int128 b = (total - diff) / 2;
        if (b > best && b <= target) best = b;
    }

    {
        std::priority_queue<ull> pq;
        for (ull x : a) pq.push(x);
        while (pq.size() > 1) {
            ull x = pq.top(); pq.pop();
            ull y = pq.top(); pq.pop();
            pq.push(x >= y ? x - y : y - x);
        }
        __int128 diff = pq.empty() ? 0 : pq.top();
        __int128 b = (total - diff) / 2;
        if (b > best && b <= target) best = b;
    }

    if (best == target) {
        __int128 ans = (total - 2 * best) * (__int128)g;
        std::cout << toString(ans) << '\n';
        return 0;
    }

    __int128 exactBest = 0;

    if (solveDP(vals, target, exactBest)) {
        best = exactBest;
    } else if (solveMITM(vals, target, exactBest)) {
        best = exactBest;
    } else {
        BranchSolver solver;
        best = solver.solve(vals, target, best);
    }

    __int128 ans = (total - 2 * best) * (__int128)g;
    std::cout << toString(ans) << '\n';

    return 0;
}
