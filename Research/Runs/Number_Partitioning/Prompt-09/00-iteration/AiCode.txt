#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <queue>
#include <cstdint>

using ull = unsigned long long;

static ull solve_dp(std::vector<ull> a, ull total) {
    ull target64 = total / 2;
    std::sort(a.begin(), a.end());

    size_t target = static_cast<size_t>(target64);
    size_t words = target / 64 + 1;
    std::vector<ull> dp(words, 0);
    dp[0] = 1ULL;

    size_t currentMax = 0;
    size_t targetWord = target / 64;
    unsigned targetBit = static_cast<unsigned>(target & 63);

    ull lastMask;
    if ((target & 63) == 63) lastMask = ~0ULL;
    else lastMask = (1ULL << ((target & 63) + 1)) - 1ULL;

    for (ull x64 : a) {
        if (x64 > target64) continue;

        size_t x = static_cast<size_t>(x64);
        size_t wordShift = x >> 6;
        unsigned bitShift = static_cast<unsigned>(x & 63);

        size_t newMax = (x > target - currentMax) ? target : currentMax + x;
        size_t hi = newMax >> 6;

        if (bitShift == 0) {
            for (size_t i = hi + 1; i-- > wordShift;) {
                dp[i] |= dp[i - wordShift];
            }
        } else {
            for (size_t i = hi + 1; i-- > wordShift;) {
                ull v = dp[i - wordShift] << bitShift;
                if (i > wordShift) {
                    v |= dp[i - wordShift - 1] >> (64 - bitShift);
                }
                dp[i] |= v;
            }
        }

        dp.back() &= lastMask;
        currentMax = newMax;

        if ((dp[targetWord] >> targetBit) & 1ULL) {
            return total - 2ULL * target64;
        }
    }

    for (size_t s = target + 1; s-- > 0;) {
        if ((dp[s >> 6] >> (s & 63)) & 1ULL) {
            return total - 2ULL * static_cast<ull>(s);
        }
    }

    return total;
}

static ull solve_mitm(const std::vector<ull>& a, ull total) {
    int n = static_cast<int>(a.size());
    int mid = n / 2;
    ull target = total / 2;

    std::vector<ull> right;
    right.reserve(1ULL << (n - mid));
    right.push_back(0);

    for (int i = mid; i < n; ++i) {
        ull x = a[i];
        size_t sz = right.size();
        if (x > target) continue;
        for (size_t j = 0; j < sz; ++j) {
            if (right[j] <= target - x) {
                right.push_back(right[j] + x);
            }
        }
    }

    std::sort(right.begin(), right.end());
    right.erase(std::unique(right.begin(), right.end()), right.end());

    ull best = 0;

    auto query = [&](ull s) {
        if (s > target) return;
        ull rem = target - s;
        auto it = std::upper_bound(right.begin(), right.end(), rem);
        if (it == right.begin()) return;
        --it;
        ull cand = s + *it;
        if (cand > best) best = cand;
    };

    auto dfs = [&](auto&& self, int idx, ull sum) -> void {
        if (best == target) return;
        if (idx == mid) {
            query(sum);
            return;
        }

        self(self, idx + 1, sum);

        ull x = a[idx];
        if (x <= target - sum) {
            self(self, idx + 1, sum + x);
        }
    };

    dfs(dfs, 0, 0);

    return total - 2ULL * best;
}

struct CKK {
    ull best;
    ull parity;

    static std::vector<ull> child_with(const std::vector<ull>& a, ull val) {
        std::vector<ull> b;
        b.reserve(a.size() - 1);

        bool inserted = (val == 0);
        for (size_t i = 2; i < a.size(); ++i) {
            if (!inserted && val >= a[i]) {
                b.push_back(val);
                inserted = true;
            }
            b.push_back(a[i]);
        }

        if (!inserted) b.push_back(val);
        return b;
    }

    void search(const std::vector<ull>& a, ull sum) {
        if (best == parity) return;

        size_t n = a.size();

        if (n == 0) {
            best = 0;
            return;
        }

        if (n == 1) {
            if (a[0] < best) best = a[0];
            return;
        }

        ull largest = a[0];
        ull rest = sum - largest;

        if (largest >= rest) {
            ull residue = largest - rest;
            if (residue < best) best = residue;
            return;
        }

        ull x = a[0];
        ull y = a[1];

        std::vector<ull> diffChild = child_with(a, x - y);
        search(diffChild, sum - 2ULL * y);

        if (best == parity) return;

        std::vector<ull> sumChild = child_with(a, x + y);
        search(sumChild, sum);
    }
};

static ull kk_heuristic(const std::vector<ull>& a) {
    std::priority_queue<ull> pq;
    for (ull x : a) {
        if (x) pq.push(x);
    }

    while (pq.size() > 1) {
        ull x = pq.top(); pq.pop();
        ull y = pq.top(); pq.pop();
        if (x > y) pq.push(x - y);
    }

    return pq.empty() ? 0 : pq.top();
}

static ull greedy_heuristic(std::vector<ull> a) {
    std::sort(a.begin(), a.end(), std::greater<ull>());
    ull s1 = 0, s2 = 0;

    for (ull x : a) {
        if (s1 <= s2) s1 += x;
        else s2 += x;
    }

    return s1 > s2 ? s1 - s2 : s2 - s1;
}

static ull solve_ckk(std::vector<ull> a, ull total) {
    a.erase(std::remove(a.begin(), a.end(), 0), a.end());
    std::sort(a.begin(), a.end(), std::greater<ull>());

    ull parity = total & 1ULL;
    ull best = std::min(kk_heuristic(a), greedy_heuristic(a));

    if (best == parity) return best;

    CKK solver;
    solver.best = best;
    solver.parity = parity;
    solver.search(a, total);

    return solver.best;
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int N;
    if (!(std::cin >> N)) return 0;

    std::vector<ull> a;
    a.reserve(N);

    ull total = 0;
    ull g = 0;

    for (int i = 0; i < N; ++i) {
        ull x;
        std::cin >> x;
        a.push_back(x);
        total += x;
        g = std::gcd(g, x);
    }

    if (N == 0 || total == 0) {
        std::cout << 0 << '\n';
        return 0;
    }

    if (g > 1) {
        for (ull& x : a) x /= g;
        total /= g;
    } else {
        g = 1;
    }

    ull diff;
    ull target = total / 2;

    const ull DP_MAX_TARGET = 200000000ULL;

    if (target <= DP_MAX_TARGET) {
        diff = solve_dp(a, total);
    } else if (N <= 46) {
        diff = solve_mitm(a, total);
    } else {
        diff = solve_ckk(a, total);
    }

    std::cout << diff * g << '\n';
    return 0;
}
