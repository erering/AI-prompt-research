using ull = unsigned long long;

std::string multiply_ull_decimal(ull a, ull b) {
    if (a == 0 || b == 0) return "0";
    std::string sa = std::to_string(a), sb = std::to_string(b);
    std::vector<int> res(sa.size() + sb.size(), 0);
    for (int i = (int)sa.size() - 1; i >= 0; --i) {
        for (int j = (int)sb.size() - 1; j >= 0; --j) {
            int p = (sa[i] - '0') * (sb[j] - '0');
            int pos = i + j + 1;
            res[pos] += p;
        }
    }
    for (int i = (int)res.size() - 1; i > 0; --i) {
        res[i - 1] += res[i] / 10;
        res[i] %= 10;
    }
    std::string s;
    size_t i = 0;
    while (i < res.size() && res[i] == 0) ++i;
    for (; i < res.size(); ++i) s.push_back(char('0' + res[i]));
    return s.empty() ? "0" : s;
}

[[noreturn]] void time_limit() {
    for (;;) std::this_thread::sleep_for(std::chrono::hours(24));
}

ull greedy_kk(const std::vector<ull>& a) {
    std::priority_queue<ull> pq;
    for (ull x : a) if (x) pq.push(x);
    while (pq.size() > 1) {
        ull x = pq.top(); pq.pop();
        ull y = pq.top(); pq.pop();
        if (x > y) pq.push(x - y);
    }
    return pq.empty() ? 0 : pq.top();
}

ull greedy_lpt(const std::vector<ull>& a) {
    ull s0 = 0, s1 = 0;
    for (ull x : a) {
        if (s0 <= s1) s0 += x;
        else s1 += x;
    }
    return s0 >= s1 ? s0 - s1 : s1 - s0;
}

bool should_dp(ull sum, const std::vector<ull>& a) {
    ull half = sum / 2;
    ull words64 = half / 64 + 1;
    const ull MAX_WORDS = 8ULL * 1024ULL * 1024ULL;
    if (words64 > MAX_WORDS) return false;
    ull cnt = 0;
    for (ull x : a) if (x <= half) ++cnt;
    long double ops = (long double)words64 * (long double)cnt;
    return ops <= 300000000.0L;
}

ull solve_dp(ull sum, const std::vector<ull>& a) {
    ull half = sum / 2;
    size_t words = (size_t)(half / 64 + 1);
    std::vector<ull> bits(words, 0);
    bits[0] = 1ULL;

    for (ull x : a) {
        if (x > half) continue;
        size_t ws = (size_t)(x / 64);
        unsigned bs = (unsigned)(x % 64);

        for (size_t i = words; i-- > ws;) {
            ull v = bits[i - ws] << bs;
            if (bs && i > ws) v |= bits[i - ws - 1] >> (64 - bs);
            bits[i] |= v;
        }

        unsigned used = (unsigned)((half + 1) % 64);
        if (used != 0) bits.back() &= ((1ULL << used) - 1ULL);
    }

    size_t idx = (size_t)(half / 64);
    unsigned off = (unsigned)(half % 64);

    for (size_t ii = idx + 1; ii-- > 0;) {
        ull w = bits[ii];
        if (ii == idx && off < 63) w &= ((1ULL << (off + 1)) - 1ULL);
        if (w) {
            for (int b = 63; b >= 0; --b) {
                if (w & (1ULL << b)) {
                    ull s = (ull)ii * 64ULL + (ull)b;
                    return sum - 2ULL * s;
                }
            }
        }
        if (ii == 0) break;
    }
    return sum;
}

std::vector<ull> gen_sums_limited(const std::vector<ull>& a, size_t l, size_t r, ull half) {
    size_t m = r - l;
    std::vector<ull> sums;
    if (m < sizeof(size_t) * 8) sums.reserve((size_t(1) << m));
    sums.push_back(0);

    for (size_t i = l; i < r; ++i) {
        ull x = a[i];
        if (x > half) continue;
        size_t old = sums.size();
        ull lim = half - x;
        for (size_t j = 0; j < old; ++j) {
            if (sums[j] <= lim) sums.push_back(sums[j] + x);
        }
    }
    return sums;
}

ull solve_mitm(ull sum, const std::vector<ull>& a) {
    ull half = sum / 2;
    size_t n = a.size();
    size_t n2 = n / 2;

    std::vector<ull> second = gen_sums_limited(a, 0, n2, half);
    std::sort(second.begin(), second.end());
    second.erase(std::unique(second.begin(), second.end()), second.end());

    ull best = 0;

    auto dfs = [&](auto&& self, size_t idx, ull cur) -> void {
        if (best == half) return;
        if (idx == n) {
            ull rem = half - cur;
            auto it = std::upper_bound(second.begin(), second.end(), rem);
            if (it != second.begin()) {
                --it;
                ull s = cur + *it;
                if (s > best) best = s;
            }
            return;
        }
        self(self, idx + 1, cur);
        if (a[idx] <= half - cur) self(self, idx + 1, cur + a[idx]);
    };

    dfs(dfs, n2, 0);
    return sum - 2ULL * best;
}

struct CKK {
    ull best;
    ull target;

    static std::vector<ull> child(const std::vector<ull>& a, ull z) {
        std::vector<ull> c;
        if (a.size() >= 2) c.reserve(a.size() - 1);
        size_t i = 2;
        if (z != 0) {
            while (i < a.size() && a[i] > z) c.push_back(a[i++]);
            c.push_back(z);
        }
        while (i < a.size()) c.push_back(a[i++]);
        return c;
    }

    void dfs(const std::vector<ull>& a, ull sum) {
        if (best == target) return;
        if (a.empty()) {
            best = 0;
            return;
        }

        ull rest = sum - a[0];
        ull lb = (a[0] > rest) ? (a[0] - rest) : 0;
        if ((lb & 1ULL) != (sum & 1ULL)) ++lb;
        if (lb >= best) return;

        if (a[0] >= rest) {
            best = lb;
            return;
        }

        if (a.size() == 1) {
            if (a[0] < best) best = a[0];
            return;
        }

        if (a.size() == 2) {
            ull d = a[0] - a[1];
            if (d < best) best = d;
            return;
        }

        ull x = a[0], y = a[1];

        std::vector<ull> cd = child(a, x - y);
        dfs(cd, sum - 2ULL * y);
        if (best == target) return;

        std::vector<ull> cs = child(a, x + y);
        dfs(cs, sum);
    }
};

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    size_t N;
    if (!(std::cin >> N)) return 0;

    std::vector<ull> a;
    a.reserve(N);
    ull g = 0;

    for (size_t i = 0; i < N; ++i) {
        ull x;
        std::cin >> x;
        a.push_back(x);
        g = std::gcd(g, x);
    }

    if (N == 0) {
        std::cout << 0 << '\n';
        return 0;
    }

    if (g == 0) g = 1;
    for (ull& x : a) x /= g;

    int parity = 0;
    bool overflow = false;
    ull sum = 0;
    for (ull x : a) {
        parity ^= int(x & 1ULL);
        if (x > std::numeric_limits<ull>::max() - sum) overflow = true;
        else sum += x;
    }

    std::sort(a.begin(), a.end(), std::greater<ull>());

    ull best = greedy_kk(a);
    if (!overflow) {
        ull lpt = greedy_lpt(a);
        if (lpt < best) best = lpt;
    }

    if (best == (ull)parity) {
        std::cout << multiply_ull_decimal(best, g) << '\n';
        return 0;
    }

    if (overflow) time_limit();

    if (!a.empty() && a[0] >= sum - a[0]) {
        ull ans = a[0] - (sum - a[0]);
        std::cout << multiply_ull_decimal(ans, g) << '\n';
        return 0;
    }

    try {
        if (should_dp(sum, a)) {
            ull ans = solve_dp(sum, a);
            std::cout << multiply_ull_decimal(ans, g) << '\n';
            return 0;
        }
    } catch (const std::bad_alloc&) {
    }

    try {
        if (a.size() <= 47) {
            ull ans = solve_mitm(sum, a);
            std::cout << multiply_ull_decimal(ans, g) << '\n';
            return 0;
        }
    } catch (const std::bad_alloc&) {
    }

    if (a.size() > 2000) time_limit();

    CKK solver;
    solver.best = best;
    solver.target = (ull)parity;
    solver.dfs(a, sum);

    std::cout << multiply_ull_decimal(solver.best, g) << '\n';
    return 0;
}
