#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <queue>
#include <string>
#include <limits>
#include <cstdint>
#include <bit>

using ull = std::uint64_t;
using u128 = unsigned __int128;
using i128 = __int128_t;

static void print_u128(u128 x) {
    if (x == 0) {
        std::cout << '0' << '\n';
        return;
    }
    std::string s;
    while (x > 0) {
        s.push_back(char('0' + x % 10));
        x /= 10;
    }
    std::reverse(s.begin(), s.end());
    std::cout << s << '\n';
}

static u128 abs_i128(i128 x) {
    return x < 0 ? (u128)(-x) : (u128)x;
}

static u128 solve_dp(const std::vector<ull>& a, std::size_t target, u128 total, u128 parity) {
    const std::size_t words = target / 64 + 1;
    std::vector<ull> dp(words, 0);
    dp[0] = 1;

    const std::size_t target_word = target / 64;
    const unsigned target_bit = unsigned(target & 63);

    unsigned last_bits = unsigned(target & 63);
    ull last_mask = (last_bits == 63) ? ~0ULL : ((1ULL << (last_bits + 1)) - 1ULL);

    for (ull x : a) {
        if (x <= target) {
            std::size_t w = std::size_t(x / 64);
            unsigned b = unsigned(x & 63);

            for (std::size_t i = words; i-- > w;) {
                ull v = dp[i - w] << b;
                if (b && i > w) v |= dp[i - w - 1] >> (64 - b);
                dp[i] |= v;
            }
            dp.back() &= last_mask;

            if ((dp[target_word] >> target_bit) & 1ULL) return parity;
        }
    }

    for (std::size_t wi = words; wi-- > 0;) {
        ull word = dp[wi];
        if (wi == words - 1) word &= last_mask;
        if (word) {
            unsigned pos = 63U - std::countl_zero(word);
            u128 best = (u128)wi * 64 + pos;
            return total - 2 * best;
        }
    }
    return total;
}

static bool solve_mitm(const std::vector<ull>& a, ull total, u128& ans) {
    int n = (int)a.size();
    int mR = std::min(23, n / 2);
    int mL = n - mR;
    if (mL > 24) return false;

    ull target = total / 2;

    std::size_t cntR = 1ULL << mR;
    std::vector<ull> right(cntR, 0);
    for (int i = 0; i < mR; ++i) {
        std::size_t base = 1ULL << i;
        ull v = a[mL + i];
        for (std::size_t j = 0; j < base; ++j) right[base + j] = right[j] + v;
    }
    std::sort(right.begin(), right.end());

    ull best = 0;
    std::size_t cntL = 1ULL << mL;
    ull sum = 0, prev_gray = 0;

    for (std::size_t mask = 0; mask < cntL; ++mask) {
        ull gray = ull(mask ^ (mask >> 1));
        if (mask) {
            ull changed = gray ^ prev_gray;
            unsigned bit = std::countr_zero(changed);
            if ((gray >> bit) & 1ULL) sum += a[bit];
            else sum -= a[bit];
        }
        prev_gray = gray;

        if (sum <= target) {
            ull rem = target - sum;
            auto it = std::upper_bound(right.begin(), right.end(), rem);
            if (it != right.begin()) {
                --it;
                ull cand = sum + *it;
                if (cand > best) {
                    best = cand;
                    if (best == target) {
                        ans = (u128)total - 2 * best;
                        return true;
                    }
                }
            }
        }
    }

    ans = (u128)total - 2 * best;
    return true;
}

static u128 heuristic_bound(const std::vector<ull>& a) {
    u128 s1 = 0, s2 = 0;
    for (ull x : a) {
        if (s1 <= s2) s1 += x;
        else s2 += x;
    }
    u128 best = s1 > s2 ? s1 - s2 : s2 - s1;

    std::priority_queue<ull> pq(a.begin(), a.end());
    while (pq.size() > 1) {
        ull x = pq.top(); pq.pop();
        ull y = pq.top(); pq.pop();
        pq.push(x - y);
    }
    if (!pq.empty() && (u128)pq.top() < best) best = pq.top();
    return best;
}

static u128 solve_dfs(const std::vector<ull>& a, u128 total, u128 best, u128 parity) {
    std::size_t n = a.size();
    if (n == 0) return 0;

    std::vector<u128> suffix(n + 1, 0);
    for (std::size_t i = n; i-- > 0;) suffix[i] = suffix[i + 1] + a[i];

    struct Node {
        std::size_t i;
        i128 diff;
    };

    std::vector<Node> st;
    st.reserve(std::min<std::size_t>(n + 1, 100000));
    st.push_back({1, (i128)a[0]});

    while (!st.empty()) {
        if (best == parity) break;

        Node cur = st.back();
        st.pop_back();

        u128 ad = abs_i128(cur.diff);
        u128 rem = suffix[cur.i];
        u128 lb = ad > rem ? ad - rem : 0;
        if (lb >= best) continue;

        if (cur.i == n) {
            best = ad;
            continue;
        }

        i128 v = (i128)a[cur.i];
        i128 d1 = cur.diff + v;
        i128 d2 = cur.diff - v;

        u128 a1 = abs_i128(d1);
        u128 a2 = abs_i128(d2);

        auto push_if = [&](i128 d) {
            u128 ad2 = abs_i128(d);
            u128 rem2 = suffix[cur.i + 1];
            u128 lb2 = ad2 > rem2 ? ad2 - rem2 : 0;
            if (lb2 < best) st.push_back({cur.i + 1, d});
        };

        if (a1 < a2) {
            push_if(d2);
            push_if(d1);
        } else {
            push_if(d1);
            push_if(d2);
        }
    }

    return best;
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    std::size_t N;
    if (!(std::cin >> N)) return 0;

    std::vector<ull> a;
    a.reserve(N);

    ull g = 0;
    u128 total = 0;

    for (std::size_t i = 0; i < N; ++i) {
        ull x;
        std::cin >> x;
        a.push_back(x);
        g = std::gcd(g, x);
        total += x;
    }

    if (N == 0 || total == 0) {
        std::cout << 0 << '\n';
        return 0;
    }

    if (g == 0) g = 1;
    if (g != 1) {
        for (ull& x : a) x /= g;
        total /= g;
    }

    u128 parity = total & 1;

    constexpr u128 DP_BYTES_CAP = (u128)128 * 1024 * 1024;
    constexpr u128 QUICK_OPS_CAP = (u128)200 * 1000 * 1000;

    u128 target128 = total / 2;
    bool dp_possible = false;
    bool quick_dp = false;
    std::size_t target = 0;

    if (target128 <= (u128)std::numeric_limits<std::size_t>::max()) {
        target = (std::size_t)target128;
        u128 words = (u128)(target / 64 + 1);
        u128 bytes = words * 8;

        if (bytes <= DP_BYTES_CAP) {
            dp_possible = true;
            std::size_t cnt = 0;
            for (ull x : a) if ((u128)x <= target128) ++cnt;
            u128 ops = words * cnt;
            if (ops <= QUICK_OPS_CAP) quick_dp = true;
        }
    }

    if (quick_dp) {
        u128 ans = solve_dp(a, target, total, parity);
        print_u128(ans * g);
        return 0;
    }

    std::sort(a.begin(), a.end(), std::greater<ull>());

    if ((u128)a[0] >= total - a[0]) {
        print_u128(((u128)a[0] - (total - a[0])) * g);
        return 0;
    }

    u128 best = heuristic_bound(a);
    if (best == parity) {
        print_u128(best * g);
        return 0;
    }

    if (a.size() <= 47 && total <= (u128)std::numeric_limits<ull>::max()) {
        u128 ans;
        if (solve_mitm(a, (ull)total, ans)) {
            print_u128(ans * g);
            return 0;
        }
    }

    if (dp_possible) {
        u128 ans = solve_dp(a, target, total, parity);
        print_u128(ans * g);
        return 0;
    }

    best = solve_dfs(a, total, best, parity);
    print_u128(best * g);
    return 0;
}
