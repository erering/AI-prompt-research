#include <vector>
#include <cstdint>

std::vector<bool> send_packet(std::vector<bool> A);

static unsigned long long comb_table[32][32];
static bool comb_ready = false;

static void init_comb() {
    if (comb_ready) return;
    for (int n = 0; n <= 31; ++n) {
        comb_table[n][0] = comb_table[n][n] = 1;
        for (int k = 1; k < n; ++k) {
            comb_table[n][k] = comb_table[n - 1][k - 1] + comb_table[n - 1][k];
        }
    }
    comb_ready = true;
}

static unsigned long long rank_subset_31_15(const std::vector<bool>& C) {
    init_comb();
    unsigned long long rank = 0;
    int prev = -1;
    int rem = 15;
    for (int pos = 0; pos < 31; ++pos) {
        if (!C[pos]) continue;
        for (int x = prev + 1; x < pos; ++x) {
            if (31 - 1 - x >= rem - 1) {
                rank += comb_table[31 - 1 - x][rem - 1];
            }
        }
        prev = pos;
        --rem;
    }
    return rank;
}

static std::vector<bool> unrank_subset_31_15(unsigned long long rank) {
    init_comb();
    std::vector<bool> C(31, false);
    int rem = 15;
    for (int pos = 0; pos < 31 && rem > 0; ++pos) {
        unsigned long long cnt = comb_table[31 - 1 - pos][rem - 1];
        if (rank < cnt) {
            C[pos] = true;
            --rem;
        } else {
            rank -= cnt;
        }
    }
    return C;
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    unsigned long long r = rank_subset_31_15(C);

    for (int t = 0; t < 29; ++t) {
        bool bit = ((r >> t) & 1ULL) != 0;
        std::vector<bool> A(31, bit);
        send_packet(A);
    }

    std::vector<int> P;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) P.push_back(i);
    }

    std::vector<bool> stream;
    int len = (int)M.size() - 1;
    for (int b = 0; b < 10; ++b) {
        stream.push_back(((len >> b) & 1) != 0);
    }
    for (bool x : M) stream.push_back(x);

    int ptr = 0;
    while (ptr < (int)stream.size()) {
        std::vector<bool> A(31, false);
        for (int j = 0; j < 16; ++j) {
            if (ptr < (int)stream.size()) {
                A[P[j]] = stream[ptr++];
            }
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    unsigned long long vals[31];

    for (int i = 0; i < 31; ++i) {
        vals[i] = 0;
        for (int t = 0; t < 29; ++t) {
            if (R[t][i]) vals[i] |= (1ULL << t);
        }
    }

    unsigned long long best = vals[0];
    int best_count = 0;
    for (int i = 0; i < 31; ++i) {
        int cnt = 0;
        for (int j = 0; j < 31; ++j) {
            if (vals[j] == vals[i]) ++cnt;
        }
        if (cnt > best_count) {
            best_count = cnt;
            best = vals[i];
        }
    }

    std::vector<bool> C = unrank_subset_31_15(best);

    std::vector<int> P;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) P.push_back(i);
    }

    std::vector<bool> stream;
    for (int t = 29; t < (int)R.size(); ++t) {
        for (int j = 0; j < 16; ++j) {
            stream.push_back(R[t][P[j]]);
        }
    }

    int len = 0;
    for (int b = 0; b < 10; ++b) {
        if (stream[b]) len |= (1 << b);
    }
    int S = len + 1;

    std::vector<bool> M;
    for (int i = 0; i < S; ++i) {
        M.push_back(stream[10 + i]);
    }
    return M;
}
