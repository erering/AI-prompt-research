#include <vector>

std::vector<bool> send_packet(std::vector<bool> A);

void send_message(std::vector<bool> M, std::vector<bool> C) {
    std::vector<int> protected_indices;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) protected_indices.push_back(i);
    }

    for (int j = 0; j < 31; ++j) {
        bool bit = !C[j];
        std::vector<bool> A(31, false);
        for (int idx : protected_indices) A[idx] = bit;
        send_packet(A);
    }

    int S = (int)M.size();
    int encoded_len = S - 1;

    std::vector<bool> payload;
    for (int b = 0; b < 10; ++b) {
        payload.push_back((encoded_len >> b) & 1);
    }
    for (bool x : M) payload.push_back(x);

    int ptr = 0;
    while (ptr < (int)payload.size()) {
        std::vector<bool> A(31, false);
        for (int k = 0; k < 16; ++k) {
            if (ptr < (int)payload.size()) A[protected_indices[k]] = payload[ptr++];
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    std::vector<int> protected_indices;
    std::vector<bool> good(31, false);

    for (int j = 0; j < 31; ++j) {
        int ones = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[j][i]) ++ones;
        }
        good[j] = (ones >= 16);
        if (good[j]) protected_indices.push_back(j);
    }

    std::vector<bool> payload;
    for (int p = 31; p < (int)R.size(); ++p) {
        for (int idx : protected_indices) {
            payload.push_back(R[p][idx]);
        }
    }

    int encoded_len = 0;
    for (int b = 0; b < 10; ++b) {
        if (payload[b]) encoded_len |= (1 << b);
    }
    int S = encoded_len + 1;

    std::vector<bool> M;
    for (int i = 0; i < S; ++i) {
        M.push_back(payload[10 + i]);
    }
    return M;
}
