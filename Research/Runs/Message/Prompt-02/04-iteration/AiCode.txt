#include <bits/stdc++.h>
using namespace std;

vector<bool> send_packet(vector<bool> A);

void send_message(vector<bool> M, vector<bool> C) {
    vector<int> G;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) G.push_back(i);
    }

    int S = (int)M.size();
    vector<bool> payload;
    int x = S - 1;
    for (int b = 0; b < 10; ++b) payload.push_back((x >> b) & 1);
    for (bool v : M) payload.push_back(v);

    int data_packets = ((int)payload.size() + 15) / 16;

    int ptr = 0;
    for (int t = 0; t < data_packets; ++t) {
        vector<bool> A(31, false);
        for (int k = 0; k < 16; ++k) {
            if (ptr < (int)payload.size()) A[G[k]] = payload[ptr++];
            else A[G[k]] = false;
        }
        send_packet(A);
    }

    uint32_t gmask = 0;
    for (int i : G) gmask |= (1u << i);

    for (int h = 0; h < 30; ++h) {
        vector<bool> A(31, false);
        for (int i : G) {
            int phys = (h < i ? h : h + 1);
            A[i] = ((gmask >> phys) & 1u);
        }
        send_packet(A);
    }
}

vector<bool> receive_message(vector<vector<bool>> R) {
    int Q = (int)R.size();
    int data_packets = Q - 30;

    unordered_map<uint32_t, int> cnt;
    vector<uint32_t> masks(31, 0);

    for (int i = 0; i < 31; ++i) {
        uint32_t m = (1u << i);
        for (int h = 0; h < 30; ++h) {
            if (R[data_packets + h][i]) {
                int phys = (h < i ? h : h + 1);
                m |= (1u << phys);
            }
        }
        masks[i] = m;
        if (__builtin_popcount(m) == 16) cnt[m]++;
    }

    uint32_t gmask = 0;
    int best = -1;
    for (auto &p : cnt) {
        if (p.second > best) {
            best = p.second;
            gmask = p.first;
        }
    }

    vector<int> G;
    for (int i = 0; i < 31; ++i) {
        if ((gmask >> i) & 1u) G.push_back(i);
    }

    vector<bool> payload;
    for (int t = 0; t < data_packets; ++t) {
        for (int i : G) payload.push_back(R[t][i]);
    }

    int S = 1;
    if ((int)payload.size() >= 10) {
        int x = 0;
        for (int b = 0; b < 10; ++b) {
            if (payload[b]) x |= (1 << b);
        }
        S = x + 1;
    }

    vector<bool> M;
    for (int i = 0; i < S; ++i) {
        M.push_back(payload[10 + i]);
    }
    return M;
}
