#include <bits/stdc++.h>
using namespace std;

vector<bool> send_packet(vector<bool> A);

static unsigned long long COMB[32][32];
static bool comb_ready = false;

static void init_comb() {
    if (comb_ready) return;
    comb_ready = true;
    for (int n = 0; n <= 31; ++n) {
        COMB[n][0] = COMB[n][n] = 1;
        for (int k = 1; k < n; ++k) COMB[n][k] = COMB[n - 1][k - 1] + COMB[n - 1][k];
    }
}

static unsigned long long choose_val(int n, int k) {
    if (n < 0 || k < 0 || k > n) return 0;
    return COMB[n][k];
}

static unsigned long long rank_mask(const vector<bool>& C) {
    init_comb();
    unsigned long long rank = 0;
    int rem = 15;
    for (int i = 0; i < 31; ++i) {
        if (C[i]) {
            rank += choose_val(30 - i, rem);
            --rem;
        }
    }
    return rank;
}

static vector<bool> unrank_mask(unsigned long long rank) {
    init_comb();
    vector<bool> C(31, false);
    int rem = 15;
    for (int i = 0; i < 31; ++i) {
        if (rem == 0) break;
        unsigned long long cnt0 = choose_val(30 - i, rem);
        if (rank < cnt0) {
            C[i] = false;
        } else {
            C[i] = true;
            rank -= cnt0;
            --rem;
        }
    }
    return C;
}

void send_message(vector<bool> M, vector<bool> C) {
    unsigned long long r = rank_mask(C);

    for (int p = 0; p < 29; ++p) {
        bool b = (r >> p) & 1ULL;
        vector<bool> A(31, b);
        send_packet(A);
    }

    int L = (int)M.size();
    int total = L + 10;
    int ptr = 0;

    while (ptr < total) {
        vector<bool> A(31, false);
        for (int i = 0; i < 31; ++i) {
            if (!C[i]) {
                bool b = false;
                if (ptr < 10) {
                    b = ((L - 1) >> ptr) & 1;
                } else if (ptr < total) {
                    b = M[ptr - 10];
                }
                A[i] = b;
                ++ptr;
            }
        }
        send_packet(A);
    }
}

vector<bool> receive_message(vector<vector<bool>> R) {
    unsigned long long r = 0;

    for (int p = 0; p < 29; ++p) {
        int ones = 0;
        for (int i = 0; i < 31; ++i) if (R[p][i]) ++ones;
        if (ones >= 16) r |= (1ULL << p);
    }

    vector<bool> C = unrank_mask(r);

    vector<bool> bits;
    for (int p = 29; p < (int)R.size(); ++p) {
        for (int i = 0; i < 31; ++i) {
            if (!C[i]) bits.push_back(R[p][i]);
        }
    }

    int Lm1 = 0;
    for (int i = 0; i < 10; ++i) {
        if (bits[i]) Lm1 |= (1 << i);
    }
    int L = Lm1 + 1;

    vector<bool> M;
    M.reserve(L);
    for (int i = 0; i < L; ++i) {
        M.push_back(bits[10 + i]);
    }
    return M;
}
