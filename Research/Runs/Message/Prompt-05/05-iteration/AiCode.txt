#include <vector>

std::vector<bool> send_packet(std::vector<bool> A);

static unsigned long long CHOOSE_TBL[32][32];
static bool CHOOSE_READY = false;

static void init_choose_tbl() {
    if (CHOOSE_READY) return;
    for (int n = 0; n <= 31; ++n) {
        CHOOSE_TBL[n][0] = CHOOSE_TBL[n][n] = 1;
        for (int k = 1; k < n; ++k) {
            CHOOSE_TBL[n][k] = CHOOSE_TBL[n - 1][k - 1] + CHOOSE_TBL[n - 1][k];
        }
    }
    CHOOSE_READY = true;
}

static unsigned long long rank_good_subset(const std::vector<bool>& C) {
    init_choose_tbl();
    std::vector<int> v;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) v.push_back(i);
    }

    unsigned long long rank = 0;
    int prev = -1;
    for (int i = 0; i < 16; ++i) {
        for (int x = prev + 1; x < v[i]; ++x) {
            rank += CHOOSE_TBL[31 - 1 - x][16 - 1 - i];
        }
        prev = v[i];
    }
    return rank;
}

static std::vector<bool> unrank_good_subset(unsigned long long rank) {
    init_choose_tbl();
    std::vector<bool> good(31, false);
    int prev = -1;

    for (int i = 0; i < 16; ++i) {
        for (int x = prev + 1; x < 31; ++x) {
            unsigned long long cnt = CHOOSE_TBL[31 - 1 - x][16 - 1 - i];
            if (rank >= cnt) {
                rank -= cnt;
            } else {
                good[x] = true;
                prev = x;
                break;
            }
        }
    }

    return good;
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    unsigned long long mask_rank = rank_good_subset(C);

    for (int t = 0; t < 29; ++t) {
        bool bit = ((mask_rank >> t) & 1ULL) != 0;
        std::vector<bool> A(31, false);
        for (int i = 0; i < 31; ++i) {
            if (!C[i]) A[i] = bit;
        }
        send_packet(A);
    }

    std::vector<bool> payload;
    int len_code = (int)M.size() - 1;
    for (int i = 0; i < 10; ++i) {
        payload.push_back(((len_code >> i) & 1) != 0);
    }
    for (bool b : M) payload.push_back(b);

    int ptr = 0;
    while (ptr < (int)payload.size()) {
        std::vector<bool> A(31, false);
        for (int i = 0; i < 31; ++i) {
            if (!C[i]) {
                if (ptr < (int)payload.size()) A[i] = payload[ptr++];
            }
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    unsigned long long mask_rank = 0;

    for (int t = 0; t < 29; ++t) {
        int ones = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[t][i]) ++ones;
        }
        if (ones >= 16) mask_rank |= (1ULL << t);
    }

    std::vector<bool> good = unrank_good_subset(mask_rank);

    std::vector<bool> payload;
    for (int p = 29; p < (int)R.size(); ++p) {
        for (int i = 0; i < 31; ++i) {
            if (good[i]) payload.push_back(R[p][i]);
        }
    }

    int len_code = 0;
    for (int i = 0; i < 10; ++i) {
        if (payload[i]) len_code |= (1 << i);
    }

    int S = len_code + 1;
    std::vector<bool> M;
    M.reserve(S);
    for (int i = 0; i < S; ++i) {
        M.push_back(payload[10 + i]);
    }

    return M;
}
