#include <bits/stdc++.h>
using namespace std;

vector<bool> send_packet(vector<bool> A);

namespace {
    constexpr int N = 31;
    constexpr int Q = 91;
    constexpr int HEADER = 19;
    constexpr int DATA_PER_COL = Q - HEADER;
    constexpr int GOOD = 16;
    constexpr int HASH_BITS = 128;

    uint64_t mix64(uint64_t x) {
        x += 0x9e3779b97f4a7c15ULL;
        x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
        x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
        return x ^ (x >> 31);
    }

    pair<uint64_t, uint64_t> calc_hash(const vector<bool>& M) {
        uint64_t h1 = 0x243f6a8885a308d3ULL ^ (uint64_t)M.size();
        uint64_t h2 = 0x13198a2e03707344ULL + ((uint64_t)M.size() << 32);

        for (size_t i = 0; i < M.size(); ++i) {
            uint64_t v = ((uint64_t)M[i] << 63) ^ (uint64_t)i;
            h1 = mix64(h1 ^ v);
            h2 = mix64(h2 + v + (h1 << 1));
        }

        h1 = mix64(h1 ^ (uint64_t)M.size());
        h2 = mix64(h2 ^ h1 ^ ((uint64_t)M.size() * 0x9e3779b97f4a7c15ULL));
        return {h1, h2};
    }

    bool bit_at(const vector<vector<bool>>& R, int r, int c) {
        if (r < 0 || r >= (int)R.size()) return false;
        if (c < 0 || c >= (int)R[r].size()) return false;
        return R[r][c];
    }

    int read_bits_col(const vector<vector<bool>>& R, int col, int from, int len) {
        int x = 0;
        for (int i = 0; i < len; ++i) {
            if (bit_at(R, from + i, col)) x |= (1 << i);
        }
        return x;
    }

    vector<bool> decode_from_choice(const vector<vector<bool>>& R, const array<int, GOOD>& chosen, int S) {
        vector<bool> data;
        data.reserve(GOOD * DATA_PER_COL);

        for (int rnk = 0; rnk < GOOD; ++rnk) {
            int col = chosen[rnk];
            for (int row = HEADER; row < Q; ++row) {
                data.push_back(bit_at(R, row, col));
            }
        }

        vector<bool> M(S);
        for (int i = 0; i < S; ++i) M[i] = data[i];

        uint64_t h1 = 0, h2 = 0;
        for (int i = 0; i < 64; ++i) {
            if (data[S + i]) h1 |= (1ULL << i);
            if (data[S + 64 + i]) h2 |= (1ULL << i);
        }

        auto hh = calc_hash(M);
        if (hh.first == h1 && hh.second == h2) return M;
        return {};
    }
}

void send_message(vector<bool> M, vector<bool> C) {
    vector<int> good;
    for (int i = 0; i < N; ++i) {
        if (!C[i]) good.push_back(i);
    }

    sort(good.begin(), good.end());

    int S = (int)M.size();
    int len_code = S - 1;

    vector<bool> payload(GOOD * DATA_PER_COL, false);

    for (int i = 0; i < S; ++i) payload[i] = M[i];

    auto hh = calc_hash(M);
    for (int i = 0; i < 64; ++i) {
        payload[S + i] = (hh.first >> i) & 1ULL;
        payload[S + 64 + i] = (hh.second >> i) & 1ULL;
    }

    vector<vector<bool>> packets(Q, vector<bool>(N, false));

    for (int rnk = 0; rnk < GOOD; ++rnk) {
        int col = good[rnk];

        for (int b = 0; b < 5; ++b) {
            packets[b][col] = (col >> b) & 1;
        }

        for (int b = 0; b < 10; ++b) {
            packets[5 + b][col] = (len_code >> b) & 1;
        }

        for (int b = 0; b < 4; ++b) {
            packets[15 + b][col] = (rnk >> b) & 1;
        }

        for (int b = 0; b < DATA_PER_COL; ++b) {
            packets[HEADER + b][col] = payload[rnk * DATA_PER_COL + b];
        }
    }

    for (int row = 0; row < Q; ++row) {
        send_packet(packets[row]);
    }
}

vector<bool> receive_message(vector<vector<bool>> R) {
    map<int, array<vector<int>, GOOD>> groups;

    for (int col = 0; col < N; ++col) {
        int idx = read_bits_col(R, col, 0, 5);
        int len_code = read_bits_col(R, col, 5, 10);
        int rnk = read_bits_col(R, col, 15, 4);

        if (idx == col && rnk >= 0 && rnk < GOOD && len_code >= 0 && len_code < 1024) {
            groups[len_code][rnk].push_back(col);
        }
    }

    vector<pair<int, int>> order;
    for (auto& kv : groups) {
        int total = 0;
        bool ok = true;
        for (int r = 0; r < GOOD; ++r) {
            sort(kv.second[r].begin(), kv.second[r].end());
            kv.second[r].erase(unique(kv.second[r].begin(), kv.second[r].end()), kv.second[r].end());
            if (kv.second[r].empty()) ok = false;
            total += (int)kv.second[r].size();
        }
        if (ok) order.push_back({-total, kv.first});
    }

    sort(order.begin(), order.end());

    for (auto [neg_total, len_code] : order) {
        auto& lists = groups[len_code];
        int S = len_code + 1;
        array<int, GOOD> chosen{};

        function<vector<bool>(int, int)> dfs = [&](int rnk, int prev_col) -> vector<bool> {
            if (rnk == GOOD) {
                return decode_from_choice(R, chosen, S);
            }

            for (int col : lists[rnk]) {
                if (col <= prev_col) continue;
                chosen[rnk] = col;
                vector<bool> res = dfs(rnk + 1, col);
                if (!res.empty()) return res;
            }

            return {};
        };

        vector<bool> res = dfs(0, -1);
        if (!res.empty()) return res;
    }

    for (auto [neg_total, len_code] : order) {
        auto& lists = groups[len_code];
        int S = len_code + 1;
        array<int, GOOD> chosen{};
        int prev = -1;
        bool ok = true;

        for (int r = 0; r < GOOD; ++r) {
            int pick = -1;
            for (int col : lists[r]) {
                if (col > prev) {
                    pick = col;
                    break;
                }
            }
            if (pick == -1) {
                ok = false;
                break;
            }
            chosen[r] = pick;
            prev = pick;
        }

        if (ok) {
            vector<bool> data;
            data.reserve(GOOD * DATA_PER_COL);
            for (int rnk = 0; rnk < GOOD; ++rnk) {
                int col = chosen[rnk];
                for (int row = HEADER; row < Q; ++row) {
                    data.push_back(bit_at(R, row, col));
                }
            }

            vector<bool> M(S);
            for (int i = 0; i < S; ++i) M[i] = data[i];
            return M;
        }
    }

    return vector<bool>(1, false);
}
