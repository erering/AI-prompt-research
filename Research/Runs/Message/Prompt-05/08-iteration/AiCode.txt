#include <vector>
#include <cstdint>
#include <algorithm>

std::vector<bool> send_packet(std::vector<bool> A);

namespace {
    constexpr int N = 31;
    constexpr int K = 16;
    constexpr int RANK_BITS = 29;
    constexpr int DATA_PACKETS = 65;
    constexpr int TOTAL_PACKETS = RANK_BITS + DATA_PACKETS;

    std::uint64_t comb[32][32];

    void init_comb() {
        static bool done = false;
        if (done) return;
        done = true;
        for (int n = 0; n <= 31; ++n) {
            comb[n][0] = comb[n][n] = 1;
            for (int k = 1; k < n; ++k) {
                comb[n][k] = comb[n - 1][k - 1] + comb[n - 1][k];
            }
        }
    }

    std::uint64_t rank_good_mask(const std::vector<bool>& good) {
        init_comb();
        std::uint64_t rank = 0;
        int rem = K;
        for (int i = 0; i < N; ++i) {
            if (good[i]) {
                rank += comb[N - i - 1][rem];
                --rem;
            }
        }
        return rank;
    }

    std::vector<bool> unrank_good_mask(std::uint64_t rank) {
        init_comb();
        std::vector<bool> good(N, false);
        int rem = K;
        for (int i = 0; i < N; ++i) {
            if (rem == 0) break;
            std::uint64_t cnt0 = comb[N - i - 1][rem];
            if (rank < cnt0) {
                good[i] = false;
            } else {
                good[i] = true;
                rank -= cnt0;
                --rem;
            }
        }
        return good;
    }
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    std::vector<bool> good(N, false);
    for (int i = 0; i < N; ++i) good[i] = !C[i];

    std::uint64_t rank = rank_good_mask(good);

    for (int b = 0; b < RANK_BITS; ++b) {
        bool bit = (rank >> b) & 1ULL;
        std::vector<bool> A(N, bit);
        send_packet(A);
    }

    std::vector<bool> payload;
    payload.reserve(DATA_PACKETS * K);

    int len = (int)M.size() - 1;
    for (int b = 0; b < 10; ++b) payload.push_back((len >> b) & 1);

    for (bool x : M) payload.push_back(x);

    while ((int)payload.size() < DATA_PACKETS * K) payload.push_back(false);

    int ptr = 0;
    for (int p = 0; p < DATA_PACKETS; ++p) {
        std::vector<bool> A(N, false);
        for (int i = 0; i < N; ++i) {
            if (good[i]) A[i] = payload[ptr++];
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    std::uint64_t rank = 0;

    for (int b = 0; b < RANK_BITS; ++b) {
        int ones = 0;
        for (int i = 0; i < N; ++i) {
            if (R[b][i]) ++ones;
        }
        if (ones >= K) rank |= (1ULL << b);
    }

    std::vector<bool> good = unrank_good_mask(rank);

    std::vector<bool> payload;
    payload.reserve(DATA_PACKETS * K);

    for (int p = 0; p < DATA_PACKETS; ++p) {
        int row = RANK_BITS + p;
        for (int i = 0; i < N; ++i) {
            if (good[i]) payload.push_back(R[row][i]);
        }
    }

    int len = 0;
    for (int b = 0; b < 10; ++b) {
        if (payload[b]) len |= (1 << b);
    }
    int S = len + 1;

    std::vector<bool> M;
    M.reserve(S);
    for (int i = 0; i < S; ++i) {
        M.push_back(payload[10 + i]);
    }
    return M;
}
