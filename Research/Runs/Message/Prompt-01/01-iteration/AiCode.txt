#include <vector>
#include <cstdint>
#include <algorithm>

std::vector<bool> send_packet(std::vector<bool> A);

namespace {
    long long binom_table[32][32];
    bool binom_ready = false;

    void init_binom() {
        if (binom_ready) return;
        for (int n = 0; n <= 31; ++n) {
            binom_table[n][0] = binom_table[n][n] = 1;
            for (int k = 1; k < n; ++k) {
                binom_table[n][k] = binom_table[n - 1][k - 1] + binom_table[n - 1][k];
            }
        }
        binom_ready = true;
    }

    uint64_t rank_combination(const std::vector<int>& a) {
        init_binom();
        const int n = 31, k = 16;
        uint64_t rank = 0;
        int last = -1;
        for (int j = 0; j < k; ++j) {
            for (int x = last + 1; x < a[j]; ++x) {
                rank += (uint64_t)binom_table[n - 1 - x][k - 1 - j];
            }
            last = a[j];
        }
        return rank;
    }

    std::vector<int> unrank_combination(uint64_t rank) {
        init_binom();
        const int n = 31, k = 16;
        std::vector<int> a;
        int start = 0;
        for (int j = 0; j < k; ++j) {
            for (int x = start; x < n; ++x) {
                uint64_t cnt = (uint64_t)binom_table[n - 1 - x][k - 1 - j];
                if (rank >= cnt) {
                    rank -= cnt;
                } else {
                    a.push_back(x);
                    start = x + 1;
                    break;
                }
            }
        }
        return a;
    }
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    std::vector<int> protected_pos;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) protected_pos.push_back(i);
    }

    uint64_t r = rank_combination(protected_pos);

    for (int t = 0; t < 29; ++t) {
        bool b = (r >> t) & 1ULL;
        std::vector<bool> A(31, b);
        send_packet(A);
    }

    std::vector<bool> payload;
    int s = (int)M.size();
    int len_code = s - 1;
    for (int i = 0; i < 10; ++i) {
        payload.push_back((len_code >> i) & 1);
    }
    for (bool b : M) payload.push_back(b);

    int ptr = 0;
    while (ptr < (int)payload.size()) {
        std::vector<bool> A(31, false);
        for (int idx : protected_pos) {
            if (ptr < (int)payload.size()) {
                A[idx] = payload[ptr++];
            }
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    uint64_t r = 0;
    for (int t = 0; t < 29; ++t) {
        int ones = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[t][i]) ++ones;
        }
        if (ones >= 16) r |= (1ULL << t);
    }

    std::vector<int> protected_pos = unrank_combination(r);

    std::vector<bool> payload;
    for (int t = 29; t < (int)R.size(); ++t) {
        for (int idx : protected_pos) {
            payload.push_back(R[t][idx]);
        }
    }

    int len_code = 0;
    for (int i = 0; i < 10; ++i) {
        if (payload[i]) len_code |= (1 << i);
    }
    int s = len_code + 1;

    std::vector<bool> M;
    M.reserve(s);
    for (int i = 0; i < s; ++i) {
        M.push_back(payload[10 + i]);
    }
    return M;
}
