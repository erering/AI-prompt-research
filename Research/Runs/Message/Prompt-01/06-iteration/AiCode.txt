#include <vector>
#include <cstdint>

std::vector<bool> send_packet(std::vector<bool> A);

namespace {
    std::uint64_t comb_table[32][32];
    bool comb_ready = false;

    void init_comb() {
        if (comb_ready) return;
        for (int n = 0; n <= 31; ++n) {
            comb_table[n][0] = 1;
            comb_table[n][n] = 1;
            for (int k = 1; k < n; ++k) {
                comb_table[n][k] = comb_table[n - 1][k - 1] + comb_table[n - 1][k];
            }
        }
        comb_ready = true;
    }

    std::uint64_t rank_control_set(const std::vector<bool>& C) {
        init_comb();
        std::uint64_t r = 0;
        int k = 1;
        for (int i = 0; i < 31; ++i) {
            if (C[i]) {
                r += comb_table[i][k];
                ++k;
            }
        }
        return r;
    }

    std::vector<bool> unrank_control_set(std::uint64_t r) {
        init_comb();
        std::vector<bool> C(31, false);
        int k = 15;
        for (int x = 30; x >= 0 && k > 0; --x) {
            if (comb_table[x][k] <= r) {
                C[x] = true;
                r -= comb_table[x][k];
                --k;
            }
        }
        return C;
    }
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    std::uint64_t rank = rank_control_set(C);

    for (int b = 0; b < 29; ++b) {
        bool bit = ((rank >> b) & 1ULL) != 0;
        std::vector<bool> A(31, false);
        for (int i = 0; i < 31; ++i) {
            if (!C[i]) A[i] = bit;
        }
        (void)send_packet(A);
    }

    std::vector<bool> stream;
    int S = (int)M.size();
    int len_code = S - 1;

    for (int b = 0; b < 10; ++b) {
        stream.push_back(((len_code >> b) & 1) != 0);
    }
    for (bool x : M) stream.push_back(x);

    std::vector<int> safe;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) safe.push_back(i);
    }

    int ptr = 0;
    while (ptr < (int)stream.size()) {
        std::vector<bool> A(31, false);
        for (int j = 0; j < 16; ++j) {
            if (ptr < (int)stream.size()) A[safe[j]] = stream[ptr++];
        }
        (void)send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    std::uint64_t rank = 0;

    for (int p = 0; p < 29; ++p) {
        int cnt = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[p][i]) ++cnt;
        }
        if (cnt >= 16) rank |= (1ULL << p);
    }

    std::vector<bool> C = unrank_control_set(rank);

    std::vector<int> safe;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) safe.push_back(i);
    }

    std::vector<bool> stream;
    for (int p = 29; p < (int)R.size(); ++p) {
        for (int j = 0; j < 16; ++j) {
            stream.push_back(R[p][safe[j]]);
        }
    }

    int len_code = 0;
    for (int b = 0; b < 10; ++b) {
        if (stream[b]) len_code |= (1 << b);
    }

    int S = len_code + 1;
    std::vector<bool> M;
    M.reserve(S);

    for (int i = 0; i < S; ++i) {
        M.push_back(stream[10 + i]);
    }

    return M;
}
