#include <vector>
#include <cstdint>

std::vector<bool> send_packet(std::vector<bool> A);

namespace {
    bool inited = false;
    uint64_t Cmb[32][32];

    void init_comb() {
        if (inited) return;
        inited = true;
        for (int n = 0; n <= 31; ++n) {
            Cmb[n][0] = Cmb[n][n] = 1;
            for (int k = 1; k < n; ++k) Cmb[n][k] = Cmb[n - 1][k - 1] + Cmb[n - 1][k];
        }
    }

    uint64_t rank_mask(const std::vector<bool>& C) {
        init_comb();
        uint64_t r = 0;
        int seen = 0;
        for (int i = 0; i < 31; ++i) {
            if (C[i]) {
                ++seen;
                r += Cmb[i][seen];
            }
        }
        return r;
    }

    std::vector<bool> unrank_mask(uint64_t r) {
        init_comb();
        std::vector<bool> C(31, false);
        int x = 30;
        for (int k = 15; k >= 1; --k) {
            while (x >= 0 && Cmb[x][k] > r) --x;
            if (x >= 0) {
                C[x] = true;
                r -= Cmb[x][k];
                --x;
            }
        }
        return C;
    }
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    uint64_t r = rank_mask(C);

    for (int i = 0; i < 29; ++i) {
        bool b = ((r >> i) & 1ULL) != 0;
        std::vector<bool> A(31, b);
        send_packet(A);
    }

    std::vector<bool> bits;
    int S = (int)M.size();
    int len = S - 1;
    for (int i = 0; i < 10; ++i) bits.push_back(((len >> i) & 1) != 0);
    for (bool b : M) bits.push_back(b);

    int D = ((int)bits.size() + 15) / 16;
    bits.resize(D * 16, false);

    int p = 0;
    for (int q = 0; q < D; ++q) {
        std::vector<bool> A(31, false);
        for (int j = 0; j < 31; ++j) {
            if (!C[j]) A[j] = bits[p++];
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    uint64_t r = 0;
    for (int i = 0; i < 29; ++i) {
        int cnt = 0;
        for (int j = 0; j < 31; ++j) if (R[i][j]) ++cnt;
        if (cnt >= 16) r |= (1ULL << i);
    }

    std::vector<bool> C = unrank_mask(r);

    std::vector<bool> bits;
    for (int i = 29; i < (int)R.size(); ++i) {
        for (int j = 0; j < 31; ++j) {
            if (!C[j]) bits.push_back(R[i][j]);
        }
    }

    int len = 0;
    for (int i = 0; i < 10 && i < (int)bits.size(); ++i) {
        if (bits[i]) len |= (1 << i);
    }

    int S = len + 1;
    std::vector<bool> M;
    M.reserve(S);
    for (int i = 0; i < S; ++i) {
        int idx = 10 + i;
        M.push_back(idx < (int)bits.size() ? bits[idx] : false);
    }
    return M;
}
