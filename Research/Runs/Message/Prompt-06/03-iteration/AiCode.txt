#include <bits/stdc++.h>
using namespace std;

vector<bool> send_packet(vector<bool> A);

static unsigned long long COMB[32][32];
static bool COMB_INIT = false;

static void init_comb() {
    if (COMB_INIT) return;
    COMB_INIT = true;
    for (int n = 0; n <= 31; ++n) {
        COMB[n][0] = COMB[n][n] = 1;
        for (int k = 1; k < n; ++k) COMB[n][k] = COMB[n - 1][k - 1] + COMB[n - 1][k];
    }
}

static unsigned long long rank_mask(const vector<bool>& C) {
    init_comb();
    unsigned long long r = 0;
    int rem = 15;
    for (int i = 0; i < 31; ++i) {
        if (C[i]) {
            r += COMB[30 - i][rem];
            --rem;
        }
    }
    return r;
}

static vector<bool> unrank_mask(unsigned long long r) {
    init_comb();
    vector<bool> C(31, false);
    int rem = 15;
    for (int i = 0; i < 31; ++i) {
        if (rem == 0) {
            C[i] = false;
            continue;
        }
        unsigned long long cnt0 = COMB[30 - i][rem];
        if (r < cnt0) {
            C[i] = false;
        } else {
            C[i] = true;
            r -= cnt0;
            --rem;
        }
    }
    return C;
}

void send_message(vector<bool> M, vector<bool> C) {
    init_comb();

    unsigned long long r = rank_mask(C);

    for (int b = 0; b < 29; ++b) {
        bool x = ((r >> b) & 1ULL);
        vector<bool> A(31, x);
        send_packet(A);
    }

    vector<bool> bits;
    int S = (int)M.size();
    int len_code = S - 1;
    for (int i = 0; i < 10; ++i) bits.push_back((len_code >> i) & 1);
    for (bool x : M) bits.push_back(x);

    size_t ptr = 0;
    while (ptr < bits.size()) {
        vector<bool> A(31, false);
        for (int j = 0; j < 31; ++j) {
            if (!C[j]) {
                if (ptr < bits.size()) A[j] = bits[ptr++];
                else A[j] = false;
            }
        }
        send_packet(A);
    }
}

vector<bool> receive_message(vector<vector<bool>> R) {
    init_comb();

    unsigned long long r = 0;
    for (int b = 0; b < 29; ++b) {
        int ones = 0;
        for (int j = 0; j < 31; ++j) if (R[b][j]) ++ones;
        if (ones >= 16) r |= (1ULL << b);
    }

    vector<bool> C = unrank_mask(r);

    vector<bool> bits;
    for (int p = 29; p < (int)R.size(); ++p) {
        for (int j = 0; j < 31; ++j) {
            if (!C[j]) bits.push_back(R[p][j]);
        }
    }

    int len_code = 0;
    for (int i = 0; i < 10; ++i) {
        if (i < (int)bits.size() && bits[i]) len_code |= (1 << i);
    }
    int S = len_code + 1;

    vector<bool> M;
    M.reserve(S);
    for (int i = 0; i < S; ++i) {
        int idx = 10 + i;
        M.push_back(idx < (int)bits.size() ? bits[idx] : false);
    }
    return M;
}
