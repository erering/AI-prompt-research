#include <vector>
#include <unordered_map>
#include <cstdint>
#include <algorithm>

std::vector<bool> send_packet(std::vector<bool> A);

static uint32_t COMB_TBL[32][32];
static bool COMB_INIT = false;

static void init_comb() {
    if (COMB_INIT) return;
    for (int n = 0; n <= 31; ++n) {
        COMB_TBL[n][0] = COMB_TBL[n][n] = 1;
        for (int k = 1; k < n; ++k) {
            COMB_TBL[n][k] = COMB_TBL[n - 1][k - 1] + COMB_TBL[n - 1][k];
        }
    }
    COMB_INIT = true;
}

static uint32_t rank_comb_30_15(const std::vector<int>& a) {
    uint32_t rank = 0;
    int prev = -1;
    for (int i = 0; i < 15; ++i) {
        for (int v = prev + 1; v < a[i]; ++v) {
            rank += COMB_TBL[30 - 1 - v][15 - 1 - i];
        }
        prev = a[i];
    }
    return rank;
}

static std::vector<int> unrank_comb_30_15(uint32_t rank) {
    std::vector<int> a;
    int start = 0;
    for (int i = 0; i < 15; ++i) {
        for (int v = start; v < 30; ++v) {
            uint32_t cnt = COMB_TBL[30 - 1 - v][15 - 1 - i];
            if (rank >= cnt) {
                rank -= cnt;
            } else {
                a.push_back(v);
                start = v + 1;
                break;
            }
        }
    }
    return a;
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    init_comb();

    uint32_t good_mask = 0;
    for (int j = 0; j < 31; ++j) {
        if (!C[j]) good_mask |= (1u << j);
    }

    uint32_t rank_for_pos[31]{};

    for (int j = 0; j < 31; ++j) {
        if (C[j]) continue;
        std::vector<int> local;
        for (int x = 0; x < 31; ++x) {
            if (x == j) continue;
            if (!C[x]) local.push_back(x < j ? x : x - 1);
        }
        rank_for_pos[j] = rank_comb_30_15(local);
    }

    for (int t = 0; t < 28; ++t) {
        std::vector<bool> A(31, false);
        for (int j = 0; j < 31; ++j) {
            if (!C[j]) A[j] = ((rank_for_pos[j] >> t) & 1u);
        }
        send_packet(A);
    }

    std::vector<bool> stream = M;
    stream.push_back(true);

    int ptr = 0;
    while (ptr < (int)stream.size()) {
        std::vector<bool> A(31, false);
        for (int j = 0; j < 31; ++j) {
            if (!C[j]) {
                if (ptr < (int)stream.size()) A[j] = stream[ptr++];
                else A[j] = false;
            }
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    init_comb();

    std::unordered_map<uint32_t, int> cnt;
    uint32_t best_mask = 0;
    int best_cnt = -1;

    for (int j = 0; j < 31; ++j) {
        uint32_t x = 0;
        for (int t = 0; t < 28 && t < (int)R.size(); ++t) {
            if (R[t][j]) x |= (1u << t);
        }

        if (x >= COMB_TBL[30][15]) continue;

        std::vector<int> local = unrank_comb_30_15(x);
        uint32_t mask = (1u << j);
        for (int v : local) {
            int g = (v < j ? v : v + 1);
            mask |= (1u << g);
        }

        int c = ++cnt[mask];
        if (c > best_cnt) {
            best_cnt = c;
            best_mask = mask;
        }
    }

    std::vector<bool> bits;
    for (int t = 28; t < (int)R.size(); ++t) {
        for (int j = 0; j < 31; ++j) {
            if (best_mask & (1u << j)) {
                bits.push_back(R[t][j]);
            }
        }
    }

    int last_one = -1;
    for (int i = 0; i < (int)bits.size(); ++i) {
        if (bits[i]) last_one = i;
    }

    if (last_one < 0) return {};

    std::vector<bool> M;
    for (int i = 0; i < last_one; ++i) M.push_back(bits[i]);
    return M;
}
