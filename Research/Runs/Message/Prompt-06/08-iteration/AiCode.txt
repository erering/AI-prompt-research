#include <bits/stdc++.h>
using namespace std;

vector<bool> send_packet(vector<bool> A);

static unsigned long long COMB[32][32];
static bool COMB_INIT = false;

static void init_comb() {
    if (COMB_INIT) return;
    COMB_INIT = true;
    for (int n = 0; n <= 31; ++n) {
        COMB[n][0] = COMB[n][n] = 1;
        for (int k = 1; k < n; ++k) COMB[n][k] = COMB[n - 1][k - 1] + COMB[n - 1][k];
    }
}

static unsigned long long rank_subset(const vector<int>& a) {
    init_comb();
    const int n = 31, k = 16;
    unsigned long long r = 0;
    int prev = -1;
    for (int i = 0; i < k; ++i) {
        for (int v = prev + 1; v < a[i]; ++v) {
            r += COMB[n - 1 - v][k - 1 - i];
        }
        prev = a[i];
    }
    return r;
}

static vector<int> unrank_subset(unsigned long long r) {
    init_comb();
    const int n = 31, k = 16;
    unsigned long long total = COMB[n][k];
    if (r >= total) r %= total;

    vector<int> a;
    int prev = -1;
    for (int i = 0; i < k; ++i) {
        for (int v = prev + 1; v <= n - (k - i); ++v) {
            unsigned long long cnt = COMB[n - 1 - v][k - 1 - i];
            if (r >= cnt) {
                r -= cnt;
            } else {
                a.push_back(v);
                prev = v;
                break;
            }
        }
    }
    return a;
}

void send_message(vector<bool> M, vector<bool> C) {
    init_comb();

    vector<int> good;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) good.push_back(i);
    }

    unsigned long long r = rank_subset(good);

    for (int t = 0; t < 29; ++t) {
        bool b = (r >> t) & 1ULL;
        vector<bool> A(31, false);
        for (int x : good) A[x] = b;
        send_packet(A);
    }

    vector<bool> payload;
    int S = (int)M.size();
    int L = S - 1;

    for (int i = 0; i < 10; ++i) payload.push_back((L >> i) & 1);
    for (bool b : M) payload.push_back(b);

    int packets = ((int)payload.size() + 15) / 16;
    for (int p = 0; p < packets; ++p) {
        vector<bool> A(31, false);
        for (int j = 0; j < 16; ++j) {
            int idx = p * 16 + j;
            if (idx < (int)payload.size()) A[good[j]] = payload[idx];
        }
        send_packet(A);
    }
}

vector<bool> receive_message(vector<vector<bool>> R) {
    init_comb();

    if ((int)R.size() < 29) return {};

    unsigned long long r = 0;
    for (int t = 0; t < 29; ++t) {
        int ones = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[t][i]) ++ones;
        }
        if (ones >= 16) r |= (1ULL << t);
    }

    vector<int> good = unrank_subset(r);

    vector<bool> bits;
    for (int p = 29; p < (int)R.size(); ++p) {
        for (int j = 0; j < 16; ++j) {
            bits.push_back(R[p][good[j]]);
        }
    }

    if ((int)bits.size() < 10) return {};

    int L = 0;
    for (int i = 0; i < 10; ++i) {
        if (bits[i]) L |= (1 << i);
    }

    int S = L + 1;
    if (S < 1) S = 1;
    if (S > 1024) S = 1024;
    if (10 + S > (int)bits.size()) S = max(0, (int)bits.size() - 10);

    vector<bool> M;
    for (int i = 0; i < S; ++i) {
        M.push_back(bits[10 + i]);
    }
    return M;
}
