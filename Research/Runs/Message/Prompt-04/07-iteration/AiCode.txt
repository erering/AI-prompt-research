#include <vector>
#include <algorithm>

std::vector<bool> send_packet(std::vector<bool> A);

void send_message(std::vector<bool> M, std::vector<bool> C) {
    const int N = 31;
    const int L = 10;
    int S = (int)M.size();

    for (int k = 0; k < N; ++k) {
        std::vector<bool> A(N, false);
        for (int i = 0; i < N; ++i) {
            if (!C[i]) A[i] = C[k];
        }
        send_packet(A);
    }

    std::vector<bool> bits;
    bits.reserve(L + S);

    int x = S - 1;
    for (int b = 0; b < L; ++b) {
        bits.push_back((x >> b) & 1);
    }

    for (bool v : M) bits.push_back(v);

    std::vector<int> safe;
    safe.reserve(16);
    for (int i = 0; i < N; ++i) {
        if (!C[i]) safe.push_back(i);
    }

    int ptr = 0;
    while (ptr < (int)bits.size()) {
        std::vector<bool> A(N, false);
        for (int idx : safe) {
            if (ptr < (int)bits.size()) {
                A[idx] = bits[ptr++];
            }
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    const int N = 31;
    const int L = 10;

    std::vector<bool> C(N, false);

    for (int k = 0; k < N; ++k) {
        int cnt = 0;
        for (int i = 0; i < N; ++i) {
            if (R[k][i]) ++cnt;
        }
        C[k] = (cnt >= 16);
    }

    std::vector<int> safe;
    safe.reserve(16);
    for (int i = 0; i < N; ++i) {
        if (!C[i]) safe.push_back(i);
    }

    std::vector<bool> bits;
    for (int p = N; p < (int)R.size(); ++p) {
        for (int idx : safe) {
            bits.push_back(R[p][idx]);
        }
    }

    int S = 1;
    if ((int)bits.size() >= L) {
        int x = 0;
        for (int b = 0; b < L; ++b) {
            if (bits[b]) x |= (1 << b);
        }
        S = x + 1;
    }

    std::vector<bool> M;
    M.reserve(S);
    for (int i = 0; i < S; ++i) {
        M.push_back(bits[L + i]);
    }

    return M;
}
