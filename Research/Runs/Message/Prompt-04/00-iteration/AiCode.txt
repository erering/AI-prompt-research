#include <vector>
#include <cstdint>
#include <algorithm>

std::vector<bool> send_packet(std::vector<bool> A);

namespace {
    long long comb_table[32][32];
    bool comb_ready = false;

    void init_comb() {
        if (comb_ready) return;
        for (int n = 0; n <= 31; ++n) {
            comb_table[n][0] = comb_table[n][n] = 1;
            for (int k = 1; k < n; ++k) {
                comb_table[n][k] = comb_table[n - 1][k - 1] + comb_table[n - 1][k];
            }
        }
        comb_ready = true;
    }

    long long rank_combination(const std::vector<int>& subset) {
        init_comb();
        const int n = 31;
        const int k = 16;
        long long rank = 0;
        int prev = -1;
        int rem = k;
        for (int x : subset) {
            for (int v = prev + 1; v < x; ++v) {
                rank += comb_table[n - v - 1][rem - 1];
            }
            prev = x;
            --rem;
        }
        return rank;
    }

    std::vector<int> unrank_combination(long long rank) {
        init_comb();
        const int n = 31;
        const int k = 16;
        std::vector<int> subset;
        int prev = -1;
        int rem = k;
        for (int pos = 0; pos < k; ++pos) {
            for (int v = prev + 1; v <= n - rem; ++v) {
                long long cnt = comb_table[n - v - 1][rem - 1];
                if (rank >= cnt) {
                    rank -= cnt;
                } else {
                    subset.push_back(v);
                    prev = v;
                    --rem;
                    break;
                }
            }
        }
        return subset;
    }
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    std::vector<int> good;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) good.push_back(i);
    }

    long long r = rank_combination(good);

    for (int b = 0; b < 29; ++b) {
        std::vector<bool> A(31, false);
        bool bit = ((r >> b) & 1LL) != 0;
        for (int idx : good) A[idx] = bit;
        send_packet(A);
    }

    std::vector<bool> data;
    int s = (int)M.size();
    int len = s - 1;
    for (int b = 0; b < 10; ++b) {
        data.push_back(((len >> b) & 1) != 0);
    }
    for (bool bit : M) data.push_back(bit);

    int packets = ((int)data.size() + 15) / 16;
    for (int p = 0; p < packets; ++p) {
        std::vector<bool> A(31, false);
        for (int j = 0; j < 16; ++j) {
            int at = p * 16 + j;
            if (at < (int)data.size()) A[good[j]] = data[at];
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    long long r = 0;
    for (int b = 0; b < 29; ++b) {
        int cnt = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[b][i]) ++cnt;
        }
        if (cnt >= 16) r |= (1LL << b);
    }

    std::vector<int> good = unrank_combination(r);

    std::vector<bool> data;
    for (int p = 29; p < (int)R.size(); ++p) {
        for (int j = 0; j < 16; ++j) {
            data.push_back(R[p][good[j]]);
        }
    }

    int len = 0;
    for (int b = 0; b < 10 && b < (int)data.size(); ++b) {
        if (data[b]) len |= (1 << b);
    }
    int s = len + 1;

    std::vector<bool> M;
    for (int i = 0; i < s; ++i) {
        int at = 10 + i;
        M.push_back(at < (int)data.size() ? data[at] : false);
    }
    return M;
}
