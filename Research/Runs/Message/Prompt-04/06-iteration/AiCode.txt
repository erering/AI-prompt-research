#include <vector>
#include <array>
#include <cstdint>
#include <algorithm>

std::vector<bool> send_packet(std::vector<bool> A);

static std::array<std::array<unsigned long long, 32>, 32> build_choose_table() {
    std::array<std::array<unsigned long long, 32>, 32> ch{};
    for (int n = 0; n <= 31; ++n) {
        ch[n][0] = 1;
        ch[n][n] = 1;
        for (int k = 1; k < n; ++k) {
            ch[n][k] = ch[n - 1][k - 1] + ch[n - 1][k];
        }
    }
    return ch;
}

static unsigned long long rank_combination_31_16(const std::vector<int>& comb) {
    static const auto ch = build_choose_table();
    unsigned long long rank = 0;
    int prev = -1;
    int rem = 16;
    for (int pos = 0; pos < 16; ++pos) {
        for (int x = prev + 1; x < comb[pos]; ++x) {
            rank += ch[31 - 1 - x][rem - 1];
        }
        prev = comb[pos];
        --rem;
    }
    return rank;
}

static std::vector<int> unrank_combination_31_16(unsigned long long rank) {
    static const auto ch = build_choose_table();
    std::vector<int> comb;
    comb.reserve(16);
    int start = 0;
    int rem = 16;
    while (rem > 0) {
        for (int x = start; x <= 31 - rem; ++x) {
            unsigned long long cnt = ch[31 - 1 - x][rem - 1];
            if (rank >= cnt) {
                rank -= cnt;
            } else {
                comb.push_back(x);
                start = x + 1;
                --rem;
                break;
            }
        }
    }
    return comb;
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    std::vector<int> good;
    good.reserve(16);
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) good.push_back(i);
    }

    unsigned long long r = rank_combination_31_16(good);

    for (int t = 0; t < 29; ++t) {
        bool bit = ((r >> t) & 1ULL) != 0;
        std::vector<bool> A(31, bit);
        send_packet(A);
    }

    std::vector<bool> data;
    data.reserve(1040);

    int len_code = static_cast<int>(M.size()) - 1;
    for (int b = 0; b < 10; ++b) {
        data.push_back(((len_code >> b) & 1) != 0);
    }
    for (bool bit : M) data.push_back(bit);
    while (data.size() < 1040) data.push_back(false);

    for (int p = 0; p < 65; ++p) {
        std::vector<bool> A(31, false);
        for (int j = 0; j < 16; ++j) {
            A[good[j]] = data[p * 16 + j];
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    unsigned long long r = 0;
    for (int t = 0; t < 29; ++t) {
        int ones = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[t][i]) ++ones;
        }
        if (ones >= 16) r |= (1ULL << t);
    }

    std::vector<int> good = unrank_combination_31_16(r);

    std::vector<bool> data;
    data.reserve(1040);
    for (int p = 0; p < 65; ++p) {
        for (int j = 0; j < 16; ++j) {
            data.push_back(R[29 + p][good[j]]);
        }
    }

    int len_code = 0;
    for (int b = 0; b < 10; ++b) {
        if (data[b]) len_code |= (1 << b);
    }
    int S = len_code + 1;

    std::vector<bool> M;
    M.reserve(S);
    for (int i = 0; i < S; ++i) {
        M.push_back(data[10 + i]);
    }
    return M;
}
