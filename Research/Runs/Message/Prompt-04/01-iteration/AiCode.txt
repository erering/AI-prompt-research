#include <bits/stdc++.h>

std::vector<bool> send_packet(std::vector<bool> A);

namespace {
long long Cmb[32][32];
bool inited = false;

void init_cmb() {
    if (inited) return;
    inited = true;
    for (int n = 0; n <= 31; ++n) {
        Cmb[n][0] = Cmb[n][n] = 1;
        for (int k = 1; k < n; ++k) Cmb[n][k] = Cmb[n - 1][k - 1] + Cmb[n - 1][k];
    }
}

unsigned long long rank_subset(const std::vector<int>& v) {
    init_cmb();
    unsigned long long r = 0;
    int prev = -1;
    int k = (int)v.size();
    for (int j = 0; j < k; ++j) {
        for (int x = prev + 1; x < v[j]; ++x) {
            r += (unsigned long long)Cmb[31 - 1 - x][k - 1 - j];
        }
        prev = v[j];
    }
    return r;
}

std::vector<int> unrank_subset(unsigned long long r) {
    init_cmb();
    std::vector<int> v;
    int prev = -1;
    int k = 16;
    for (int j = 0; j < k; ++j) {
        for (int x = prev + 1; x < 31; ++x) {
            unsigned long long cnt = (unsigned long long)Cmb[31 - 1 - x][k - 1 - j];
            if (r >= cnt) {
                r -= cnt;
            } else {
                v.push_back(x);
                prev = x;
                break;
            }
        }
    }
    return v;
}
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    std::vector<int> good;
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) good.push_back(i);
    }

    int S = (int)M.size();
    int bits = S + 10;
    int D = (bits + 15) / 16;

    std::vector<bool> payload(D * 16, false);
    int len = S - 1;
    for (int b = 0; b < 10; ++b) payload[b] = ((len >> b) & 1) != 0;
    for (int i = 0; i < S; ++i) payload[10 + i] = M[i];

    for (int t = 0; t < D; ++t) {
        std::vector<bool> A(31, false);
        for (int k = 0; k < 16; ++k) {
            A[good[k]] = payload[t * 16 + k];
        }
        send_packet(A);
    }

    unsigned long long r = rank_subset(good);
    for (int b = 0; b < 29; ++b) {
        bool bit = ((r >> b) & 1ULL) != 0;
        std::vector<bool> A(31, bit);
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    int Q = (int)R.size();
    int D = Q - 29;

    unsigned long long r = 0;
    for (int b = 0; b < 29; ++b) {
        int ones = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[D + b][i]) ++ones;
        }
        if (ones >= 16) r |= (1ULL << b);
    }

    std::vector<int> good = unrank_subset(r);
    std::vector<bool> payload(D * 16, false);

    for (int t = 0; t < D; ++t) {
        for (int k = 0; k < 16; ++k) {
            payload[t * 16 + k] = R[t][good[k]];
        }
    }

    int len = 0;
    for (int b = 0; b < 10; ++b) {
        if (payload[b]) len |= (1 << b);
    }
    int S = len + 1;

    std::vector<bool> M(S, false);
    for (int i = 0; i < S; ++i) {
        M[i] = payload[10 + i];
    }
    return M;
}
