#include <vector>
#include <cstdint>
#include <algorithm>

std::vector<bool> send_packet(std::vector<bool> A);

namespace {
    unsigned long long COMB[32][32];
    bool comb_inited = false;

    void init_comb() {
        if (comb_inited) return;
        comb_inited = true;
        for (int n = 0; n <= 31; ++n) {
            COMB[n][0] = COMB[n][n] = 1;
            for (int k = 1; k < n; ++k) {
                COMB[n][k] = COMB[n - 1][k - 1] + COMB[n - 1][k];
            }
        }
    }

    unsigned long long rank_subset(const std::vector<int>& subset) {
        init_comb();
        const int n = 31;
        const int k = 16;
        unsigned long long rank = 0;
        int prev = -1;
        for (int i = 0; i < k; ++i) {
            for (int x = prev + 1; x < subset[i]; ++x) {
                rank += COMB[n - 1 - x][k - 1 - i];
            }
            prev = subset[i];
        }
        return rank;
    }

    std::vector<int> unrank_subset(unsigned long long rank) {
        init_comb();
        const int n = 31;
        const int k = 16;
        std::vector<int> subset;
        subset.reserve(k);
        int prev = -1;
        for (int i = 0; i < k; ++i) {
            for (int x = prev + 1; x < n; ++x) {
                unsigned long long cnt = COMB[n - 1 - x][k - 1 - i];
                if (rank >= cnt) {
                    rank -= cnt;
                } else {
                    subset.push_back(x);
                    prev = x;
                    break;
                }
            }
        }
        return subset;
    }
}

void send_message(std::vector<bool> M, std::vector<bool> C) {
    init_comb();

    std::vector<int> good;
    good.reserve(16);
    for (int i = 0; i < 31; ++i) {
        if (!C[i]) good.push_back(i);
    }

    unsigned long long r = rank_subset(good);

    for (int b = 0; b < 29; ++b) {
        bool bit = ((r >> b) & 1ULL) != 0;
        std::vector<bool> A(31, bit);
        send_packet(A);
    }

    std::vector<bool> stream;
    stream.reserve(M.size() + 1);
    for (bool bit : M) stream.push_back(bit);
    stream.push_back(true);

    int idx = 0;
    int total = static_cast<int>(stream.size());
    int packets = (total + 15) / 16;

    for (int p = 0; p < packets; ++p) {
        std::vector<bool> A(31, false);
        for (int j = 0; j < 16; ++j) {
            if (idx < total) A[good[j]] = stream[idx++];
        }
        send_packet(A);
    }
}

std::vector<bool> receive_message(std::vector<std::vector<bool>> R) {
    init_comb();

    unsigned long long r = 0;
    for (int b = 0; b < 29; ++b) {
        int ones = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[b][i]) ++ones;
        }
        if (ones >= 16) r |= (1ULL << b);
    }

    std::vector<int> good = unrank_subset(r);

    std::vector<bool> stream;
    for (int p = 29; p < static_cast<int>(R.size()); ++p) {
        for (int j = 0; j < 16; ++j) {
            stream.push_back(R[p][good[j]]);
        }
    }

    int last = -1;
    for (int i = 0; i < static_cast<int>(stream.size()); ++i) {
        if (stream[i]) last = i;
    }

    std::vector<bool> M;
    if (last > 0) {
        M.reserve(last);
        for (int i = 0; i < last; ++i) M.push_back(stream[i]);
    }
    return M;
}
