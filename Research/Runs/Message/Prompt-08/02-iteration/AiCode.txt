#include <bits/stdc++.h>
using namespace std;

vector<bool> send_packet(vector<bool> A);

namespace {
    unsigned long long COMB[32][32];
    bool comb_ready = false;

    void init_comb() {
        if (comb_ready) return;
        memset(COMB, 0, sizeof(COMB));
        for (int n = 0; n <= 31; ++n) {
            COMB[n][0] = COMB[n][n] = 1;
            for (int k = 1; k < n; ++k) {
                COMB[n][k] = COMB[n - 1][k - 1] + COMB[n - 1][k];
            }
        }
        comb_ready = true;
    }

    unsigned long long rank_combination(const vector<int>& a, int n, int k) {
        init_comb();
        unsigned long long rank = 0;
        int prev = -1;
        for (int j = 0; j < k; ++j) {
            for (int x = prev + 1; x < a[j]; ++x) {
                rank += COMB[n - 1 - x][k - 1 - j];
            }
            prev = a[j];
        }
        return rank;
    }

    vector<int> unrank_combination(unsigned long long rank, int n, int k) {
        init_comb();
        vector<int> a;
        int prev = -1;
        for (int j = 0; j < k; ++j) {
            for (int x = prev + 1; x < n; ++x) {
                unsigned long long cnt = COMB[n - 1 - x][k - 1 - j];
                if (rank >= cnt) {
                    rank -= cnt;
                } else {
                    a.push_back(x);
                    prev = x;
                    break;
                }
            }
        }
        return a;
    }
}

void send_message(vector<bool> M, vector<bool> C) {
    vector<int> bad_positions;
    for (int i = 0; i < 31; ++i) {
        if (C[i]) bad_positions.push_back(i);
    }

    unsigned long long r = rank_combination(bad_positions, 31, 15);

    for (int b = 0; b < 29; ++b) {
        bool bit = ((r >> b) & 1ULL);
        vector<bool> A(31, bit);
        send_packet(A);
    }

    vector<bool> payload;
    int S = (int)M.size();
    int len_code = S - 1;

    for (int b = 0; b < 10; ++b) {
        payload.push_back((len_code >> b) & 1);
    }
    for (bool bit : M) {
        payload.push_back(bit);
    }

    int ptr = 0;
    while (ptr < (int)payload.size()) {
        vector<bool> A(31, false);
        for (int i = 0; i < 31; ++i) {
            if (!C[i]) {
                if (ptr < (int)payload.size()) {
                    A[i] = payload[ptr++];
                } else {
                    A[i] = false;
                }
            }
        }
        send_packet(A);
    }
}

vector<bool> receive_message(vector<vector<bool>> R) {
    if ((int)R.size() < 29) return {};

    unsigned long long r = 0;
    for (int b = 0; b < 29; ++b) {
        int ones = 0;
        for (int i = 0; i < 31; ++i) {
            if (R[b][i]) ++ones;
        }
        if (ones >= 16) {
            r |= (1ULL << b);
        }
    }

    vector<int> bad_positions = unrank_combination(r, 31, 15);
    vector<bool> bad(31, false);
    for (int x : bad_positions) {
        if (0 <= x && x < 31) bad[x] = true;
    }

    vector<bool> payload;
    for (int p = 29; p < (int)R.size(); ++p) {
        for (int i = 0; i < 31; ++i) {
            if (!bad[i]) {
                payload.push_back(R[p][i]);
            }
        }
    }

    if ((int)payload.size() < 10) return {};

    int len_code = 0;
    for (int b = 0; b < 10; ++b) {
        if (payload[b]) len_code |= (1 << b);
    }

    int S = len_code + 1;
    vector<bool> M;
    M.reserve(S);

    for (int i = 0; i < S; ++i) {
        int idx = 10 + i;
        M.push_back(idx < (int)payload.size() ? payload[idx] : false);
    }

    return M;
}
