#include <bits/stdc++.h>
using namespace std;

namespace {
    constexpr int NN = 10000;
    constexpr int LOG = 15;
    constexpr int T = 9615;
    constexpr int L = 384;
    constexpr int B = 8;
    constexpr int BLOCKS = 48;

    bool initialized = false;
    int up[LOG][NN];
    int depthArr[NN];

    int curA = 0, curB = 0, curD = 0;
    int snapA = 0, snapB = 0;
    int trits[L];

    void init_state() {
        initialized = true;
        for (int k = 0; k < LOG; ++k) up[k][0] = 0;
        depthArr[0] = 0;
        curA = curB = curD = 0;
        snapA = snapB = 0;
        for (int &x : trits) x = 0;
    }

    int lca(int a, int b) {
        if (depthArr[a] < depthArr[b]) swap(a, b);
        int diff = depthArr[a] - depthArr[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up[k][a];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up[k][a] != up[k][b]) {
                a = up[k][a];
                b = up[k][b];
            }
        }
        return up[0][a];
    }

    int dist_tree(int a, int b) {
        int c = lca(a, b);
        return depthArr[a] + depthArr[b] - 2 * depthArr[c];
    }

    int add_vertex_and_update(int i, int p) {
        up[0][i] = p;
        for (int k = 1; k < LOG; ++k) up[k][i] = up[k - 1][up[k - 1][i]];
        depthArr[i] = depthArr[p] + 1;

        int oldA = curA, oldB = curB, oldD = curD;
        int dA = dist_tree(i, oldA);
        int dB = dist_tree(i, oldB);

        if (dA > oldD) {
            curA = i;
            curB = oldA;
            curD = dA;
            return 1;
        }
        if (dB > oldD) {
            curA = i;
            curB = oldB;
            curD = dB;
            return 2;
        }
        return 0;
    }
}

int send_message(int N, int i, int Pi) {
    (void)N;
    if (!initialized || i == 1) init_state();

    int tr = add_vertex_and_update(i, Pi);

    if (i == T) {
        snapA = curA;
        snapB = curB;
        return snapA + 1;
    }

    if (i > T && i <= T + L) {
        trits[i - (T + 1)] = tr;
    }

    if (i == T + 1) {
        return snapB + 1;
    }

    if (i >= T + B && i <= T + L && (i - T) % B == 0) {
        int block = (i - T) / B - 1;
        int code = 0;
        int mul = 1;
        for (int j = 0; j < B; ++j) {
            code += trits[block * B + j] * mul;
            mul *= 3;
        }
        return code + 1;
    }

    return 0;
}

std::pair<int,int> longest_path(std::vector<int> S) {
    int a = S[T] - 1;
    int b = S[T + 1] - 1;

    for (int block = 0; block < BLOCKS; ++block) {
        int pos = T + B * (block + 1);
        int code = S[pos] - 1;

        for (int j = 0; j < B; ++j) {
            int tr = code % 3;
            code /= 3;
            int v = T + 1 + block * B + j;

            if (tr == 1) {
                int oldA = a;
                a = v;
                b = oldA;
            } else if (tr == 2) {
                int oldB = b;
                a = v;
                b = oldB;
            }
        }
    }

    return {a, b};
}
