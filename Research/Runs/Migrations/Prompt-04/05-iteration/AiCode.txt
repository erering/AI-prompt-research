#include <bits/stdc++.h>
using namespace std;

namespace {
    constexpr int NN = 10000;
    constexpr int LOG = 15;
    constexpr int L = NN - 50;

    int up[LOG][NN];
    int dep[NN];

    int diamA = 0, diamB = 0, diamD = 0;
    int baseA = 0, baseB = 0, snapX = 0;
    int mode = 0;

    void reset_state() {
        for (int k = 0; k < LOG; ++k) up[k][0] = 0;
        dep[0] = 0;
        diamA = diamB = diamD = 0;
        baseA = baseB = snapX = 0;
        mode = 0;
    }

    int lca_node(int a, int b) {
        if (dep[a] < dep[b]) swap(a, b);
        int diff = dep[a] - dep[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up[k][a];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up[k][a] != up[k][b]) {
                a = up[k][a];
                b = up[k][b];
            }
        }
        return up[0][a];
    }

    int dist_node(int a, int b) {
        int c = lca_node(a, b);
        return dep[a] + dep[b] - 2 * dep[c];
    }
}

int send_message(int N, int i, int Pi) {
    (void)N;
    if (i == 1) reset_state();

    dep[i] = dep[Pi] + 1;
    up[0][i] = Pi;
    for (int k = 1; k < LOG; ++k) up[k][i] = up[k - 1][up[k - 1][i]];

    int oldA = diamA, oldB = diamB, oldD = diamD;
    int da = dist_node(i, oldA);
    int db = dist_node(i, oldB);

    bool changed = false;
    int retained = -1;

    if (da > oldD || db > oldD) {
        changed = true;
        if (da >= db) {
            diamA = i;
            diamB = oldA;
            diamD = da;
            retained = oldA;
        } else {
            diamA = i;
            diamB = oldB;
            diamD = db;
            retained = oldB;
        }
    }

    if (i < L) return 0;

    if (i == L) {
        baseA = diamA;
        baseB = diamB;
        return baseA + 1;
    }

    if (i == L + 1) {
        if (changed) {
            snapX = retained;
            mode = 2;
            return snapX + 1;
        } else {
            snapX = baseB;
            mode = 1;
            return snapX + 1;
        }
    }

    if (mode == 1) {
        if (!changed) return 0;
        mode = 3;
        return (retained == baseA) ? 1 : 2;
    }

    if (mode == 2) {
        if (!changed) {
            mode = 3;
            return 4;
        }
        mode = 3;
        return (retained == snapX) ? 2 : 3;
    }

    if (changed) {
        return (retained == oldA) ? 1 : 2;
    }

    return 0;
}

std::pair<int,int> longest_path(std::vector<int> S) {
    int a = S[L] - 1;
    int x = S[L + 1] - 1;

    pair<int,int> cur = {a, x};
    bool resolved = false;

    for (int i = L + 2; i < NN; ++i) {
        int v = S[i];
        if (v == 0) continue;

        if (!resolved) {
            if (v == 1) cur = {i, a};
            else if (v == 2) cur = {i, x};
            else if (v == 3) cur = {i, L + 1};
            else if (v == 4) cur = {L + 1, x};
            resolved = true;
        } else {
            if (v == 1) cur = {i, cur.first};
            else if (v == 2) cur = {i, cur.second};
        }
    }

    return cur;
}
