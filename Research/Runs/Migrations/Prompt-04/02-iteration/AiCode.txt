#include <bits/stdc++.h>
using namespace std;

namespace {
    constexpr int FIXED_N = 10000;
    constexpr int LOG = 15;
    constexpr int T0 = 9950;

    int par_[FIXED_N];
    int dep_[FIXED_N];
    int up_[FIXED_N][LOG];

    int diaA = 0, diaB = 0, diaD = 0;
    bool initialized = false;
    long long savedQ = 0;

    int lca_(int a, int b) {
        if (dep_[a] < dep_[b]) swap(a, b);
        int diff = dep_[a] - dep_[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up_[a][k];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up_[a][k] != up_[b][k]) {
                a = up_[a][k];
                b = up_[b][k];
            }
        }
        return up_[a][0];
    }

    int dist_(int a, int b) {
        int c = lca_(a, b);
        return dep_[a] + dep_[b] - 2 * dep_[c];
    }

    long long rank_pair_(int a, int b) {
        if (a > b) swap(a, b);
        return 1LL * a * (FIXED_N - 1) - 1LL * a * (a - 1) / 2 + (b - a - 1);
    }

    pair<int,int> unrank_pair_(long long r) {
        for (int a = 0; a < FIXED_N - 1; ++a) {
            int cnt = FIXED_N - 1 - a;
            if (r < cnt) return {a, a + 1 + (int)r};
            r -= cnt;
        }
        return {0, 1};
    }

    void init_() {
        initialized = true;
        par_[0] = 0;
        dep_[0] = 0;
        for (int k = 0; k < LOG; ++k) up_[0][k] = 0;
        diaA = diaB = 0;
        diaD = 0;
        savedQ = 0;
    }
}

int send_message(int N, int i, int Pi) {
    (void)N;
    if (!initialized) init_();

    par_[i] = Pi;
    dep_[i] = dep_[Pi] + 1;
    up_[i][0] = Pi;
    for (int k = 1; k < LOG; ++k) up_[i][k] = up_[up_[i][k - 1]][k - 1];

    int state = 0;

    int da = dist_(i, diaA);
    int db = dist_(i, diaB);

    if (da >= db && da > diaD) {
        int oldA = diaA;
        diaA = i;
        diaB = oldA;
        diaD = da;
        state = 1;
    } else if (db > diaD) {
        int oldB = diaB;
        diaA = i;
        diaB = oldB;
        diaD = db;
        state = 2;
    }

    if (i < T0) return 0;

    int t = i - T0;

    if (t == 0) {
        if (diaA > diaB) swap(diaA, diaB);
        long long r = rank_pair_(diaA, diaB);
        int d0 = (int)(r % 6);
        savedQ = r / 6;
        return d0 + 1;
    }

    if (t <= 23) {
        int bit = (int)((savedQ >> (t - 1)) & 1LL);
        return bit * 3 + state + 1;
    }

    return state + 1;
}

std::pair<int,int> longest_path(std::vector<int> S) {
    long long r = 0;

    int d0 = S[T0] - 1;
    if (d0 < 0) d0 = 0;
    if (d0 > 5) d0 = 5;

    long long q = 0;
    vector<int> states(50, 0);

    for (int t = 1; t <= 23; ++t) {
        int x = S[T0 + t] - 1;
        if (x < 0) x = 0;
        if (x > 5) x = 5;
        int bit = x / 3;
        int st = x % 3;
        if (bit) q |= (1LL << (t - 1));
        states[t] = st;
    }

    for (int t = 24; t <= 49; ++t) {
        int st = S[T0 + t] - 1;
        if (st < 0) st = 0;
        if (st > 2) st = 2;
        states[t] = st;
    }

    r = d0 + 6LL * q;
    auto p = unrank_pair_(r);
    int A = p.first;
    int B = p.second;

    for (int t = 1; t <= 49; ++t) {
        int i = T0 + t;
        if (states[t] == 1) {
            int oldA = A;
            A = i;
            B = oldA;
        } else if (states[t] == 2) {
            int oldB = B;
            A = i;
            B = oldB;
        }
    }

    return {A, B};
}
