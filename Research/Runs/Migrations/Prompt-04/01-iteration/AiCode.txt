#include <bits/stdc++.h>
using namespace std;

namespace {
    constexpr int MAXN = 10000;
    constexpr int LOG = 15;

    bool initialized = false;
    int up[LOG][MAXN];
    int dep[MAXN];

    int e1 = 0, e2 = 0;
    int diam = 0;

    void init_state() {
        initialized = true;
        for (int k = 0; k < LOG; ++k) up[k][0] = 0;
        dep[0] = 0;
        e1 = e2 = 0;
        diam = 0;
    }

    int lca(int a, int b) {
        if (dep[a] < dep[b]) swap(a, b);
        int d = dep[a] - dep[b];
        for (int k = 0; k < LOG; ++k) {
            if (d & (1 << k)) a = up[k][a];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up[k][a] != up[k][b]) {
                a = up[k][a];
                b = up[k][b];
            }
        }
        return up[0][a];
    }

    int dist(int a, int b) {
        int c = lca(a, b);
        return dep[a] + dep[b] - 2 * dep[c];
    }
}

int send_message(int N, int i, int Pi) {
    (void)N;

    if (!initialized || i == 1) init_state();

    up[0][i] = Pi;
    dep[i] = dep[Pi] + 1;
    for (int k = 1; k < LOG; ++k) up[k][i] = up[k - 1][up[k - 1][i]];

    if (i == MAXN - 1) {
        int d1 = dist(i, e1);
        int d2 = dist(i, e2);

        if (d2 >= d1 && d2 >= diam) return e2 + 1;
        if (d1 >= diam) return e1 + 1;

        return e2 + 10001;
    }

    int old1 = e1, old2 = e2;
    int d1 = dist(i, old1);
    int d2 = dist(i, old2);

    if (d1 >= diam || d2 >= diam) {
        if (d2 >= d1) {
            e1 = i;
            e2 = old2;
            diam = d2;
        } else {
            e1 = i;
            e2 = old1;
            diam = d1;
        }
    }

    if (i == MAXN - 2) {
        if (e1 < e2) swap(e1, e2);
        return e1 + 1;
    }

    return 0;
}

std::pair<int,int> longest_path(std::vector<int> S) {
    int a = S[MAXN - 2] - 1;
    int v = S[MAXN - 1];

    if (v <= 10000) {
        return {MAXN - 1, v - 1};
    } else {
        return {a, v - 10001};
    }
}
