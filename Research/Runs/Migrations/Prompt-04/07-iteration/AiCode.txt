#include <bits/stdc++.h>
using namespace std;

namespace {
    constexpr int FIXED_N = 10000;
    constexpr int LOG = 15;

    int up[LOG][FIXED_N];
    int dep[FIXED_N];
    int diam_u = 0, diam_v = 0, diam_len = 0;
    bool initialized = false;

    void init_state() {
        for (int k = 0; k < LOG; ++k) up[k][0] = 0;
        dep[0] = 0;
        diam_u = diam_v = 0;
        diam_len = 0;
        initialized = true;
    }

    int lca_node(int a, int b) {
        if (dep[a] < dep[b]) swap(a, b);
        int diff = dep[a] - dep[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up[k][a];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up[k][a] != up[k][b]) {
                a = up[k][a];
                b = up[k][b];
            }
        }
        return up[0][a];
    }

    int dist_nodes(int a, int b) {
        int c = lca_node(a, b);
        return dep[a] + dep[b] - 2 * dep[c];
    }
}

int send_message(int N, int i, int Pi) {
    if (!initialized || i == 1) init_state();

    up[0][i] = Pi;
    dep[i] = dep[Pi] + 1;
    for (int k = 1; k < LOG; ++k) {
        up[k][i] = up[k - 1][up[k - 1][i]];
    }

    int old_u = diam_u;
    int old_v = diam_v;
    int old_len = diam_len;

    int du = dist_nodes(i, old_u);
    int dv = dist_nodes(i, old_v);

    if (du > old_len) {
        diam_v = i;
        diam_len = du;
    } else if (dv > old_len) {
        diam_u = i;
        diam_len = dv;
    }

    if (i == N - 2) {
        return diam_u + 1;
    }

    if (i == N - 1) {
        if (diam_u == i) {
            return 10001 + diam_v;
        }
        if (diam_v == i) {
            return 10001 + diam_u;
        }
        return diam_v + 1;
    }

    return 0;
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = (int)S.size();
    int a = S[N - 2] - 1;
    int x = S[N - 1];

    if (x > 10000) {
        int other = x - 10001;
        return {N - 1, other};
    } else {
        int b = x - 1;
        return {a, b};
    }
}
