#include <bits/stdc++.h>
using namespace std;

namespace {
    constexpr int LOG = 20;

    int encN = 0;
    vector<array<int, LOG>> upTable;
    vector<int> depthArr;
    int diamA = 0, diamB = 0, diamD = 0;

    void init_encoder(int N) {
        encN = N;
        upTable.assign(max(1, N), {});
        depthArr.assign(max(1, N), 0);
        for (int k = 0; k < LOG; ++k) upTable[0][k] = 0;
        diamA = diamB = 0;
        diamD = 0;
    }

    void add_vertex(int v, int p) {
        upTable[v][0] = p;
        for (int k = 1; k < LOG; ++k) {
            upTable[v][k] = upTable[upTable[v][k - 1]][k - 1];
        }
        depthArr[v] = depthArr[p] + 1;
    }

    int lca_vertex(int a, int b) {
        if (depthArr[a] < depthArr[b]) swap(a, b);

        int diff = depthArr[a] - depthArr[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = upTable[a][k];
        }

        if (a == b) return a;

        for (int k = LOG - 1; k >= 0; --k) {
            if (upTable[a][k] != upTable[b][k]) {
                a = upTable[a][k];
                b = upTable[b][k];
            }
        }
        return upTable[a][0];
    }

    int dist_vertex(int a, int b) {
        int c = lca_vertex(a, b);
        return depthArr[a] + depthArr[b] - 2 * depthArr[c];
    }

    void update_diameter_with(int v) {
        int da = dist_vertex(v, diamA);
        int db = dist_vertex(v, diamB);

        if (da > diamD) {
            diamB = v;
            diamD = da;
        } else if (db > diamD) {
            diamA = diamB;
            diamB = v;
            diamD = db;
        }
    }
}

int send_message(int N, int i, int Pi) {
    if (i == 1 || encN != N || (int)upTable.size() < N) {
        init_encoder(N);
    }

    if (i == N - 1) {
        int prevA = diamA;
        int prevB = diamB;
        int prevD = diamD;

        add_vertex(i, Pi);

        int da = dist_vertex(i, prevA);
        int db = dist_vertex(i, prevB);

        if (da > prevD || db > prevD) {
            int far = (da >= db ? prevA : prevB);
            return 10001 + far;
        } else {
            return prevB + 1;
        }
    }

    add_vertex(i, Pi);
    update_diameter_with(i);

    if (i == N - 2) {
        return diamA + 1;
    }

    return 0;
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = (int)S.size();

    if (N <= 1) {
        return {0, 0};
    }

    int last = S[N - 1];

    if (last > 10000) {
        return {N - 1, last - 10001};
    }

    if (N == 2) {
        return {0, 1};
    }

    return {S[N - 2] - 1, last - 1};
}
