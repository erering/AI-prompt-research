#include <vector>
#include <utility>
#include <algorithm>

namespace {
    constexpr int OFF = 10000;
    constexpr int LOG = 15;

    int curN = 0;
    int e1 = 0, e2 = 0, diamLen = 0;
    std::vector<int> dep;
    std::vector<std::vector<int>> up;

    int lca_node(int a, int b) {
        if (dep[a] < dep[b]) std::swap(a, b);
        int diff = dep[a] - dep[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up[k][a];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up[k][a] != up[k][b]) {
                a = up[k][a];
                b = up[k][b];
            }
        }
        return up[0][a];
    }

    int dist_node(int a, int b) {
        int c = lca_node(a, b);
        return dep[a] + dep[b] - 2 * dep[c];
    }
}

int send_message(int N, int i, int Pi) {
    if (i == 1 || curN != N) {
        curN = N;
        e1 = e2 = 0;
        diamLen = 0;
        dep.assign(N, 0);
        up.assign(LOG, std::vector<int>(N, 0));
    }

    dep[i] = dep[Pi] + 1;
    up[0][i] = Pi;
    for (int k = 1; k < LOG; ++k) {
        up[k][i] = up[k - 1][up[k - 1][i]];
    }

    bool changed = false;
    int other = -1;

    int d1 = dist_node(i, e1);
    int d2 = dist_node(i, e2);

    if (d1 > diamLen) {
        int old = e1;
        e1 = i;
        e2 = old;
        diamLen = d1;
        changed = true;
        other = old;
    } else if (d2 > diamLen) {
        int old = e2;
        e1 = i;
        e2 = old;
        diamLen = d2;
        changed = true;
        other = old;
    }

    if (i == N - 2) {
        return e1 + 1;
    }

    if (i == N - 1) {
        if (changed) return OFF + other + 1;
        return e2 + 1;
    }

    return 0;
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = (int)S.size();
    if (N <= 1) return {0, 0};

    int last = S[N - 1];

    if (last > OFF) {
        return {N - 1, last - OFF - 1};
    }

    return {S[N - 2] - 1, last - 1};
}
