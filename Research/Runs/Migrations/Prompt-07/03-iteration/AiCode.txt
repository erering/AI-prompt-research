#include <vector>
#include <utility>
#include <algorithm>

namespace {
    constexpr int MAXN = 10000;
    constexpr int LOG = 15;

    int up[LOG][MAXN];
    int depth_arr[MAXN];

    int diaA = 0, diaB = 0, diaD = 0;
    int chkA = 0, chkB = 0, chkD = 0;
    bool initialized = false;

    void reset_state() {
        for (int k = 0; k < LOG; ++k) up[k][0] = 0;
        depth_arr[0] = 0;
        diaA = diaB = 0;
        diaD = 0;
        chkA = chkB = 0;
        chkD = 0;
        initialized = true;
    }

    int lca_node(int a, int b) {
        if (depth_arr[a] < depth_arr[b]) std::swap(a, b);

        int diff = depth_arr[a] - depth_arr[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up[k][a];
        }

        if (a == b) return a;

        for (int k = LOG - 1; k >= 0; --k) {
            if (up[k][a] != up[k][b]) {
                a = up[k][a];
                b = up[k][b];
            }
        }

        return up[0][a];
    }

    int dist_node(int a, int b) {
        int c = lca_node(a, b);
        return depth_arr[a] + depth_arr[b] - 2 * depth_arr[c];
    }
}

int send_message(int N, int i, int Pi) {
    if (i == 0) {
        reset_state();
        return 0;
    }

    if (i == 1 || !initialized) reset_state();

    up[0][i] = Pi;
    depth_arr[i] = depth_arr[Pi] + 1;

    for (int k = 1; k < LOG; ++k) {
        up[k][i] = up[k - 1][up[k - 1][i]];
    }

    if (i == N - 1) {
        int da = dist_node(i, chkA);
        int db = dist_node(i, chkB);

        if (da > chkD || db > chkD) {
            int other = (da >= db ? chkA : chkB);
            return 2 * other + 2;
        } else {
            return 2 * chkB + 1;
        }
    }

    int da = dist_node(i, diaA);
    int db = dist_node(i, diaB);

    if (da > diaD || db > diaD) {
        if (da >= db) {
            diaB = i;
            diaD = da;
        } else {
            diaA = i;
            diaD = db;
        }
    }

    if (i == N - 2) {
        chkA = diaA;
        chkB = diaB;
        chkD = diaD;
        return chkA + 1;
    }

    return 0;
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = (int)S.size();

    int stored = S[N - 2] - 1;
    int code = S[N - 1];

    if (code % 2 == 1) {
        int other = (code - 1) / 2;
        return {stored, other};
    } else {
        int other = (code - 2) / 2;
        return {N - 1, other};
    }
}
