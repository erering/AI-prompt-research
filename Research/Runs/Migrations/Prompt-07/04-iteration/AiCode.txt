#include <bits/stdc++.h>

static const int MAXN = 10005;
static const int LOG = 15;
static int up_tbl[LOG][MAXN];
static int dep_tbl[MAXN];
static int dia_a, dia_b, dia_d;
static int saved_a, saved_b;
static int upd_block[8];

static void reset_sender_state() {
    for (int k = 0; k < LOG; ++k) up_tbl[k][0] = 0;
    dep_tbl[0] = 0;
    dia_a = dia_b = 0;
    dia_d = 0;
    saved_a = saved_b = 0;
    for (int &x : upd_block) x = 0;
}

static int lca_sender(int u, int v) {
    if (dep_tbl[u] < dep_tbl[v]) std::swap(u, v);
    int diff = dep_tbl[u] - dep_tbl[v];
    for (int k = 0; k < LOG; ++k) {
        if (diff & (1 << k)) u = up_tbl[k][u];
    }
    if (u == v) return u;
    for (int k = LOG - 1; k >= 0; --k) {
        if (up_tbl[k][u] != up_tbl[k][v]) {
            u = up_tbl[k][u];
            v = up_tbl[k][v];
        }
    }
    return up_tbl[0][u];
}

static int dist_sender(int u, int v) {
    int w = lca_sender(u, v);
    return dep_tbl[u] + dep_tbl[v] - 2 * dep_tbl[w];
}

int send_message(int N, int i, int Pi) {
    if (i == 0) {
        reset_sender_state();
        return 0;
    }
    if (i == 1) {
        reset_sender_state();
    }

    up_tbl[0][i] = Pi;
    for (int k = 1; k < LOG; ++k) {
        up_tbl[k][i] = up_tbl[k - 1][up_tbl[k - 1][i]];
    }
    dep_tbl[i] = dep_tbl[Pi] + 1;

    int old_a = dia_a;
    int old_b = dia_b;
    int old_d = dia_d;

    int da = dist_sender(i, old_a);
    int db = dist_sender(i, old_b);

    int status = 0;
    if (da > old_d || db > old_d) {
        if (da >= db) {
            status = 1;
            dia_a = i;
            dia_b = old_a;
            dia_d = da;
        } else {
            status = 2;
            dia_a = i;
            dia_b = old_b;
            dia_d = db;
        }
    }

    const int K = 8;
    int A = N - 1 - K;

    if (i == A) {
        saved_a = dia_a;
        saved_b = dia_b;
        return saved_a + 1;
    }

    if (i >= A + 1 && i <= N - 1) {
        upd_block[i - (A + 1)] = status;
    }

    if (i == A + 1) {
        return saved_b + 1;
    }

    if (i == N - 1) {
        int code = 0;
        int pow3 = 1;
        for (int j = 0; j < K; ++j) {
            code += upd_block[j] * pow3;
            pow3 *= 3;
        }
        return code + 1;
    }

    return 0;
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = (int)S.size();
    const int K = 8;
    int A = N - 1 - K;

    int a = S[A] - 1;
    int b = S[A + 1] - 1;
    int code = S[N - 1] - 1;

    for (int j = 0; j < K; ++j) {
        int status = code % 3;
        code /= 3;

        int v = A + 1 + j;
        if (status == 1) {
            int old_a = a;
            a = v;
            b = old_a;
        } else if (status == 2) {
            int old_b = b;
            a = v;
            b = old_b;
        }
    }

    return {a, b};
}
