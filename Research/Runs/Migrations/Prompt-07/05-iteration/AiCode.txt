#include <bits/stdc++.h>
using namespace std;

namespace {
    constexpr int MAXN = 10000;
    constexpr int LOG = 15;

    int up[LOG][MAXN];
    int depth_arr[MAXN];
    int diaA = 0, diaB = 0, diaD = 0;
    int snapA = 0, snapB = 0;

    void reset_state() {
        for (int k = 0; k < LOG; ++k) up[k][0] = 0;
        depth_arr[0] = 0;
        diaA = diaB = diaD = 0;
        snapA = snapB = 0;
    }

    int lca_node(int a, int b) {
        if (depth_arr[a] < depth_arr[b]) swap(a, b);

        int diff = depth_arr[a] - depth_arr[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up[k][a];
        }

        if (a == b) return a;

        for (int k = LOG - 1; k >= 0; --k) {
            if (up[k][a] != up[k][b]) {
                a = up[k][a];
                b = up[k][b];
            }
        }
        return up[0][a];
    }

    int dist_nodes(int a, int b) {
        int c = lca_node(a, b);
        return depth_arr[a] + depth_arr[b] - 2 * depth_arr[c];
    }
}

int send_message(int N, int i, int Pi) {
    if (i == 0) {
        reset_state();
        return 0;
    }

    if (i == 1) reset_state();

    up[0][i] = Pi;
    depth_arr[i] = depth_arr[Pi] + 1;
    for (int k = 1; k < LOG; ++k) {
        up[k][i] = up[k - 1][up[k - 1][i]];
    }

    int da = dist_nodes(i, diaA);
    int db = dist_nodes(i, diaB);

    if (da > diaD) {
        diaD = da;
        diaB = diaA;
        diaA = i;
    } else if (db > diaD) {
        diaD = db;
        diaA = diaB;
        diaB = i;
    }

    if (i == N - 2) {
        snapA = diaA;
        snapB = diaB;
        return snapA + 1;
    }

    if (i == N - 1) {
        bool same = (diaA == snapA && diaB == snapB) || (diaA == snapB && diaB == snapA);

        if (same) {
            return snapB + 1;
        } else {
            int other = (diaA == i ? diaB : diaA);
            return 10001 + other;
        }
    }

    return 0;
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = (int)S.size();

    int first = S[N - 2] - 1;
    int last = S[N - 1];

    if (last > 10000) {
        return {N - 1, last - 10001};
    } else {
        return {first, last - 1};
    }
}
