#include <bits/stdc++.h>

namespace {
    constexpr int MAXN = 10005;
    constexpr int LOG = 20;

    int current_n = -1;
    int up[MAXN][LOG];
    int depth_arr[MAXN];
    int diam_a = 0;
    int diam_b = 0;
    int diam_len = 0;

    void stage1_initialize(int N) {
        current_n = N;
        diam_a = 0;
        diam_b = 0;
        diam_len = 0;
        depth_arr[0] = 0;
        for (int k = 0; k < LOG; ++k) up[0][k] = 0;
    }

    int lca_vertex(int u, int v) {
        if (depth_arr[u] < depth_arr[v]) std::swap(u, v);
        int diff = depth_arr[u] - depth_arr[v];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) u = up[u][k];
        }
        if (u == v) return u;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up[u][k] != up[v][k]) {
                u = up[u][k];
                v = up[v][k];
            }
        }
        return up[u][0];
    }

    int dist_vertices(int u, int v) {
        int w = lca_vertex(u, v);
        return depth_arr[u] + depth_arr[v] - 2 * depth_arr[w];
    }

    void stage2_reveal_edge(int i, int Pi) {
        up[i][0] = Pi;
        depth_arr[i] = depth_arr[Pi] + 1;
        for (int k = 1; k < LOG; ++k) {
            up[i][k] = up[up[i][k - 1]][k - 1];
        }
    }

    int stage3_update_diameter_and_decide(int N, int i) {
        int da = dist_vertices(i, diam_a);
        int db = dist_vertices(i, diam_b);

        if (i == N - 1) {
            if (da > diam_len) {
                int other = diam_a;
                diam_b = i;
                diam_len = da;
                return 10001 + other;
            }
            if (db > diam_len) {
                int other = diam_b;
                diam_a = i;
                diam_len = db;
                return 10001 + other;
            }
            return diam_b + 1;
        }

        if (da > diam_len) {
            diam_b = i;
            diam_len = da;
        } else if (db > diam_len) {
            diam_a = i;
            diam_len = db;
        }

        if (i == N - 2) {
            return diam_a + 1;
        }

        return 0;
    }
}

int send_message(int N, int i, int Pi) {
    if (i == 1 || current_n != N) {
        stage1_initialize(N);
    }

    stage2_reveal_edge(i, Pi);

    if (N <= 2) {
        int da = dist_vertices(i, diam_a);
        if (da > diam_len) {
            diam_b = i;
            diam_len = da;
        }
        return 0;
    }

    return stage3_update_diameter_and_decide(N, i);
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = (int)S.size();

    if (N <= 1) return {0, 0};
    if (N == 2) return {0, 1};

    int a_code = S[N - 2];
    int last_code = S[N - 1];

    if (last_code > 10000) {
        return {N - 1, last_code - 10001};
    }

    return {a_code - 1, last_code - 1};
}
