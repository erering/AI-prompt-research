#include <bits/stdc++.h>
using namespace std;

namespace Stage1 {
    static constexpr int MAXN = 10000;
    static constexpr int LOG = 15;

    int currentN = 0;
    int depth[MAXN + 5];
    int up[MAXN + 5][LOG];
    int diaA = 0, diaB = 0, diaD = 0;
    int snapA = 0, snapB = 0;

    void initialize(int N) {
        currentN = N;
        depth[0] = 0;
        for (int k = 0; k < LOG; ++k) up[0][k] = 0;
        diaA = diaB = diaD = 0;
        snapA = snapB = 0;
    }

    int lca(int a, int b) {
        if (depth[a] < depth[b]) swap(a, b);
        int diff = depth[a] - depth[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up[a][k];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up[a][k] != up[b][k]) {
                a = up[a][k];
                b = up[b][k];
            }
        }
        return up[a][0];
    }

    int dist(int a, int b) {
        int c = lca(a, b);
        return depth[a] + depth[b] - 2 * depth[c];
    }
}

namespace Stage2 {
    void reveal_edge(int i, int Pi) {
        using namespace Stage1;
        depth[i] = depth[Pi] + 1;
        up[i][0] = Pi;
        for (int k = 1; k < LOG; ++k) up[i][k] = up[up[i][k - 1]][k - 1];
    }
}

namespace Stage3 {
    int ordered_code(int a, int b) {
        int r = 1;
        for (int x = 0; x < 4; ++x) {
            for (int y = 0; y < 4; ++y) {
                if (x == y) continue;
                if (x == a && y == b) return r;
                ++r;
            }
        }
        return 1;
    }

    pair<int,int> decode_ordered_code(int code) {
        int r = 1;
        for (int x = 0; x < 4; ++x) {
            for (int y = 0; y < 4; ++y) {
                if (x == y) continue;
                if (r == code) return {x, y};
                ++r;
            }
        }
        return {0, 1};
    }

    int index_in_snapshot_set(int v, int A, int B, int x, int y) {
        if (v == A) return 0;
        if (v == B) return 1;
        if (v == x) return 2;
        return 3;
    }

    int decide(int N, int i, int oldA, int oldB, bool changed) {
        using namespace Stage1;
        int s = N - 50;

        if (i < s) return 0;

        if (i == s) {
            snapA = diaA;
            snapB = diaB;
            return snapA + 1;
        }

        if (i == s + 1) {
            return snapB + 1;
        }

        if (i == s + 2) {
            int ia = index_in_snapshot_set(diaA, snapA, snapB, s + 1, s + 2);
            int ib = index_in_snapshot_set(diaB, snapA, snapB, s + 1, s + 2);
            return ordered_code(ia, ib);
        }

        if (changed) {
            return (diaB == oldA ? 1 : 2);
        }

        return 0;
    }
}

namespace Stage4 {
    int update_and_message(int N, int i) {
        using namespace Stage1;

        int oldA = diaA, oldB = diaB;
        bool changed = false;

        int da = dist(i, diaA);
        int db = dist(i, diaB);

        if (da >= db) {
            if (da > diaD) {
                diaD = da;
                diaA = i;
                diaB = oldA;
                changed = true;
            }
        } else {
            if (db > diaD) {
                diaD = db;
                diaA = i;
                diaB = oldB;
                changed = true;
            }
        }

        return Stage3::decide(N, i, oldA, oldB, changed);
    }
}

int send_message(int N, int i, int Pi) {
    if (i == 1 || Stage1::currentN != N) Stage1::initialize(N);
    Stage2::reveal_edge(i, Pi);
    return Stage4::update_and_message(N, i);
}

namespace Stage5 {
    pair<int,int> reconstruct(vector<int> S) {
        int N = (int)S.size();
        if (N <= 1) return {0, 0};

        int s = N - 50;

        int A = S[s] - 1;
        int B = S[s + 1] - 1;

        int vertices[4] = {A, B, s + 1, s + 2};
        auto p = Stage3::decode_ordered_code(S[s + 2]);

        int curA = vertices[p.first];
        int curB = vertices[p.second];

        for (int i = s + 3; i < N; ++i) {
            if (S[i] == 1) {
                curB = curA;
                curA = i;
            } else if (S[i] == 2) {
                curA = i;
            }
        }

        return {curA, curB};
    }
}

pair<int,int> longest_path(vector<int> S) {
    return Stage5::reconstruct(std::move(S));
}
