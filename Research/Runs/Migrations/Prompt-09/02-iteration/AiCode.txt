#include <vector>
#include <utility>
#include <array>
#include <algorithm>

namespace staged_solution {
    constexpr int LOG = 15;
    constexpr int BLOCK = 9;
    constexpr int TAIL_UPDATES = 432;
    constexpr int SHORT_LIMIT = 6667;

    struct OnlineState {
        int N = -1;
        int t0 = 0;
        bool short_mode = false;
        std::vector<std::array<int, LOG>> up;
        std::vector<int> depth;
        int diaA = 0, diaB = 0, diaD = 0;
        int snapA = 0, snapB = 0;
        int pack = 0;
        int pow3[BLOCK]{};
    };

    static OnlineState st;

    static void stage1_online_initialization(int N) {
        st = OnlineState();
        st.N = N;
        st.short_mode = (N <= SHORT_LIMIT);
        st.t0 = N - 1 - TAIL_UPDATES;
        st.up.assign(std::max(N, 1), {});
        st.depth.assign(std::max(N, 1), 0);
        for (int k = 0; k < LOG; ++k) st.up[0][k] = 0;
        st.pow3[0] = 1;
        for (int k = 1; k < BLOCK; ++k) st.pow3[k] = st.pow3[k - 1] * 3;
        st.diaA = st.diaB = st.diaD = 0;
        st.snapA = st.snapB = 0;
        st.pack = 0;
    }

    static int lca(int a, int b) {
        if (st.depth[a] < st.depth[b]) std::swap(a, b);
        int diff = st.depth[a] - st.depth[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = st.up[a][k];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (st.up[a][k] != st.up[b][k]) {
                a = st.up[a][k];
                b = st.up[b][k];
            }
        }
        return st.up[a][0];
    }

    static int dist(int a, int b) {
        int c = lca(a, b);
        return st.depth[a] + st.depth[b] - 2 * st.depth[c];
    }

    static void stage2_revealed_edge(int i, int Pi) {
        st.up[i][0] = Pi;
        for (int k = 1; k < LOG; ++k) {
            st.up[i][k] = st.up[st.up[i][k - 1]][k - 1];
        }
        st.depth[i] = st.depth[Pi] + 1;
    }

    static int update_diameter_and_get_type(int i) {
        int da = dist(i, st.diaA);
        int db = dist(i, st.diaB);
        if (da > st.diaD) {
            st.diaB = i;
            st.diaD = da;
            return 2;
        }
        if (db > st.diaD) {
            st.diaA = i;
            st.diaD = db;
            return 1;
        }
        return 0;
    }

    static int stage3_decision_rule_short(int N, int i, int type) {
        if (N <= 2) return 0;
        if (i == N - 2) {
            st.snapA = st.diaA;
            st.snapB = st.diaB;
            return st.snapA + 1;
        }
        if (i == N - 1) {
            return st.snapB * 3 + type + 1;
        }
        return 0;
    }

    static int stage3_decision_rule_long(int N, int i, int type) {
        int ret = 0;
        if (i == st.t0) {
            st.snapA = st.diaA;
            st.snapB = st.diaB;
            ret = st.snapA + 1;
        } else if (i > st.t0) {
            if (i == st.t0 + 1) ret = st.snapB + 1;
            int idx = (i - (st.t0 + 1)) % BLOCK;
            st.pack += type * st.pow3[idx];
            if (idx == BLOCK - 1 || i == N - 1) {
                int v = st.pack + 1;
                st.pack = 0;
                if (ret == 0) ret = v;
            }
        }
        return ret;
    }

    static int stage4_bounded_message_output(int N, int i, int type) {
        if (st.short_mode) return stage3_decision_rule_short(N, i, type);
        return stage3_decision_rule_long(N, i, type);
    }

    static std::pair<int, int> stage5_decode_short(const std::vector<int>& S) {
        int N = (int)S.size();
        if (N <= 1) return {0, 0};
        if (N == 2) return {0, 1};
        int A = S[N - 2] - 1;
        int code = S[N - 1] - 1;
        int B = code / 3;
        int type = code % 3;
        if (type == 1) A = N - 1;
        else if (type == 2) B = N - 1;
        return {A, B};
    }

    static std::pair<int, int> stage5_decode_long(const std::vector<int>& S) {
        int N = (int)S.size();
        int t0 = N - 1 - TAIL_UPDATES;
        int A = S[t0] - 1;
        int B = S[t0 + 1] - 1;
        for (int start = t0 + 1; start <= N - 1; start += BLOCK) {
            int end = std::min(start + BLOCK - 1, N - 1);
            int code = S[end] - 1;
            for (int v = start; v <= end; ++v) {
                int type = code % 3;
                code /= 3;
                if (type == 1) A = v;
                else if (type == 2) B = v;
            }
        }
        return {A, B};
    }
}

int send_message(int N, int i, int Pi) {
    using namespace staged_solution;
    if (i <= 0) return 0;
    if (st.N != N || i == 1) stage1_online_initialization(N);
    stage2_revealed_edge(i, Pi);
    int type = update_diameter_and_get_type(i);
    return stage4_bounded_message_output(N, i, type);
}

std::pair<int, int> longest_path(std::vector<int> S) {
    using namespace staged_solution;
    int N = (int)S.size();
    if (N <= SHORT_LIMIT) return stage5_decode_short(S);
    return stage5_decode_long(S);
}
