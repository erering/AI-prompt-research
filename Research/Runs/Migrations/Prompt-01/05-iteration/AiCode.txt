#include <bits/stdc++.h>
using namespace std;

static const int MAXN = 10000;
static const int LOG = 15;

static int up[LOG][MAXN];
static int depth_arr[MAXN];
static int diaA = 0, diaB = 0, diaLen = 0;
static bool initialized = false;

static void init_state() {
    if (initialized) return;
    initialized = true;
    for (int k = 0; k < LOG; ++k) up[k][0] = 0;
    depth_arr[0] = 0;
    diaA = diaB = 0;
    diaLen = 0;
}

static int lca_node(int a, int b) {
    if (depth_arr[a] < depth_arr[b]) swap(a, b);
    int diff = depth_arr[a] - depth_arr[b];
    for (int k = 0; k < LOG; ++k) {
        if (diff & (1 << k)) a = up[k][a];
    }
    if (a == b) return a;
    for (int k = LOG - 1; k >= 0; --k) {
        if (up[k][a] != up[k][b]) {
            a = up[k][a];
            b = up[k][b];
        }
    }
    return up[0][a];
}

static int dist_nodes(int a, int b) {
    int c = lca_node(a, b);
    return depth_arr[a] + depth_arr[b] - 2 * depth_arr[c];
}

int send_message(int N, int i, int Pi) {
    init_state();

    up[0][i] = Pi;
    depth_arr[i] = depth_arr[Pi] + 1;
    for (int k = 1; k < LOG; ++k) {
        up[k][i] = up[k - 1][up[k - 1][i]];
    }

    int oldA = diaA, oldB = diaB, oldD = diaLen;
    int da = dist_nodes(i, oldA);
    int db = dist_nodes(i, oldB);

    bool changed = false;
    int other = -1;

    if (da > oldD) {
        diaA = i;
        diaB = oldA;
        diaLen = da;
        changed = true;
        other = oldA;
    } else if (db > oldD) {
        diaA = i;
        diaB = oldB;
        diaLen = db;
        changed = true;
        other = oldB;
    }

    if (i == N - 2) {
        return diaA + 1;
    }

    if (i == N - 1) {
        if (changed) return 10001 + other;
        return diaB + 1;
    }

    return 0;
}

pair<int, int> longest_path(vector<int> S) {
    int N = (int)S.size();
    int last = S[N - 1];

    if (last > 10000) {
        return {N - 1, last - 10001};
    }

    return {S[N - 2] - 1, last - 1};
}
