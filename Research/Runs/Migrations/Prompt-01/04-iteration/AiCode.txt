#include <bits/stdc++.h>
using namespace std;

namespace {
    const int MAXN = 10000;
    const int LOG = 14;

    int up[LOG][MAXN];
    int dep[MAXN];

    int diaA = 0, diaB = 0, diaD = 0;
    int snapA = 0, snapB = 0, snapD = 0;

    int lca_node(int a, int b) {
        if (dep[a] < dep[b]) swap(a, b);
        int diff = dep[a] - dep[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up[k][a];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up[k][a] != up[k][b]) {
                a = up[k][a];
                b = up[k][b];
            }
        }
        return up[0][a];
    }

    int dist_node(int a, int b) {
        int c = lca_node(a, b);
        return dep[a] + dep[b] - 2 * dep[c];
    }
}

int send_message(int N, int i, int Pi) {
    if (i == 1) {
        for (int k = 0; k < LOG; ++k) up[k][0] = 0;
        dep[0] = 0;
        diaA = diaB = 0;
        diaD = 0;
        snapA = snapB = snapD = 0;
    }

    up[0][i] = Pi;
    dep[i] = dep[Pi] + 1;
    for (int k = 1; k < LOG; ++k) {
        up[k][i] = up[k - 1][up[k - 1][i]];
    }

    int dA = dist_node(i, diaA);
    int dB = dist_node(i, diaB);

    if (dA > diaD) {
        diaB = i;
        diaD = dA;
    } else if (dB > diaD) {
        diaA = i;
        diaD = dB;
    }

    if (i == N - 2) {
        snapA = diaA;
        snapB = diaB;
        snapD = diaD;
        return snapA + 1;
    }

    if (i == N - 1) {
        if (diaD > snapD) {
            int other = (diaA == i ? diaB : diaA);
            return other + 1;
        } else {
            return snapB + 10001;
        }
    }

    return 0;
}

pair<int, int> longest_path(vector<int> S) {
    int N = (int)S.size();
    int a = S[N - 2] - 1;
    int x = S[N - 1];

    if (x <= 10000) {
        return {N - 1, x - 1};
    } else {
        return {a, x - 10001};
    }
}
