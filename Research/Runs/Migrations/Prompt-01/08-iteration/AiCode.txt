#include <bits/stdc++.h>
using namespace std;

static const int MAXN = 10000;
static const int LOG = 15;

static int up_tbl[LOG][MAXN];
static int dep_tbl[MAXN];
static int dia_a = 0, dia_b = 0, dia_len = 0;

static int lca_node(int a, int b) {
    if (dep_tbl[a] < dep_tbl[b]) swap(a, b);
    int diff = dep_tbl[a] - dep_tbl[b];
    for (int k = 0; k < LOG; ++k) {
        if (diff & (1 << k)) a = up_tbl[k][a];
    }
    if (a == b) return a;
    for (int k = LOG - 1; k >= 0; --k) {
        if (up_tbl[k][a] != up_tbl[k][b]) {
            a = up_tbl[k][a];
            b = up_tbl[k][b];
        }
    }
    return up_tbl[0][a];
}

static int dist_node(int a, int b) {
    int c = lca_node(a, b);
    return dep_tbl[a] + dep_tbl[b] - 2 * dep_tbl[c];
}

int send_message(int N, int i, int Pi) {
    if (i == 1) {
        dia_a = dia_b = 0;
        dia_len = 0;
        dep_tbl[0] = 0;
        for (int k = 0; k < LOG; ++k) up_tbl[k][0] = 0;
    }

    up_tbl[0][i] = Pi;
    dep_tbl[i] = dep_tbl[Pi] + 1;
    for (int k = 1; k < LOG; ++k) {
        up_tbl[k][i] = up_tbl[k - 1][up_tbl[k - 1][i]];
    }

    int old_a = dia_a, old_b = dia_b, old_len = dia_len;
    int da = dist_node(i, old_a);
    int db = dist_node(i, old_b);

    bool updated = false;
    int far_endpoint = old_a;

    if (da > old_len || db > old_len) {
        updated = true;
        if (da >= db) {
            far_endpoint = old_a;
            dia_a = i;
            dia_b = old_a;
            dia_len = da;
        } else {
            far_endpoint = old_b;
            dia_a = i;
            dia_b = old_b;
            dia_len = db;
        }
    }

    if (i == N - 2) {
        return dia_a + 1;
    }

    if (i == N - 1) {
        if (updated) {
            return 10000 + far_endpoint;
        } else {
            return dia_b + 1;
        }
    }

    return 0;
}

pair<int, int> longest_path(vector<int> S) {
    int N = (int)S.size();
    int a = S[N - 2] - 1;
    int x = S[N - 1];

    if (x >= 10000) {
        return {N - 1, x - 10000};
    } else {
        return {a, x - 1};
    }
}
