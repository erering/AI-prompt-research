#include <bits/stdc++.h>
using namespace std;

namespace {
    constexpr int MAXN = 10000;
    constexpr int LOG = 15;

    int up[MAXN][LOG];
    int depthArr[MAXN];

    int diaA = 0, diaB = 0, diaD = 0;

    int lca_node(int a, int b) {
        if (depthArr[a] < depthArr[b]) swap(a, b);
        int diff = depthArr[a] - depthArr[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up[a][k];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up[a][k] != up[b][k]) {
                a = up[a][k];
                b = up[b][k];
            }
        }
        return up[a][0];
    }

    int dist_node(int a, int b) {
        int c = lca_node(a, b);
        return depthArr[a] + depthArr[b] - 2 * depthArr[c];
    }
}

int send_message(int N, int i, int Pi) {
    if (i == 1) {
        diaA = diaB = diaD = 0;
        depthArr[0] = 0;
        for (int k = 0; k < LOG; ++k) up[0][k] = 0;
    }

    up[i][0] = Pi;
    for (int k = 1; k < LOG; ++k) up[i][k] = up[up[i][k - 1]][k - 1];
    depthArr[i] = depthArr[Pi] + 1;

    int prevA = diaA;
    int prevB = diaB;
    bool changed = false;
    int other = -1;

    int da = dist_node(i, prevA);
    int db = dist_node(i, prevB);

    if (da > diaD) {
        diaA = i;
        diaB = prevA;
        diaD = da;
        changed = true;
        other = prevA;
    } else if (db > diaD) {
        diaA = i;
        diaB = prevB;
        diaD = db;
        changed = true;
        other = prevB;
    }

    int start = N - 50;

    if (i == start) {
        return diaA + 1;
    }

    if (i == start + 1) {
        if (changed) return 10001 + other;
        return diaB + 1;
    }

    if (i >= start + 2) {
        if (changed) return other + 1;
        return 0;
    }

    return 0;
}

pair<int, int> longest_path(vector<int> S) {
    int N = (int)S.size();
    int start = N - 50;

    int a = S[start] - 1;
    int b = 0;

    int v = S[start + 1];
    if (v >= 10001) {
        a = start + 1;
        b = v - 10001;
    } else {
        b = v - 1;
    }

    for (int i = start + 2; i < N; ++i) {
        if (S[i] != 0) {
            a = i;
            b = S[i] - 1;
        }
    }

    return {a, b};
}
