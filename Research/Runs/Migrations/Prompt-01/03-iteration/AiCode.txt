#include <bits/stdc++.h>
using namespace std;

static const int MAXN = 10005;
static const int LOG = 16;

static int up_tbl[MAXN][LOG];
static int dep_tbl[MAXN];

static int dia_a = 0, dia_b = 0, dia_len = 0;
static int snap_a = 0, snap_b = 0;
static vector<int> trans_ops;

static int lca_node(int u, int v) {
    if (dep_tbl[u] < dep_tbl[v]) swap(u, v);
    int diff = dep_tbl[u] - dep_tbl[v];
    for (int k = 0; k < LOG; k++) {
        if (diff & (1 << k)) u = up_tbl[u][k];
    }
    if (u == v) return u;
    for (int k = LOG - 1; k >= 0; k--) {
        if (up_tbl[u][k] != up_tbl[v][k]) {
            u = up_tbl[u][k];
            v = up_tbl[v][k];
        }
    }
    return up_tbl[u][0];
}

static int dist_node(int u, int v) {
    int w = lca_node(u, v);
    return dep_tbl[u] + dep_tbl[v] - 2 * dep_tbl[w];
}

int send_message(int N, int i, int Pi) {
    int T = N - 9;

    if (i == 1) {
        memset(up_tbl, 0, sizeof(up_tbl));
        memset(dep_tbl, 0, sizeof(dep_tbl));
        dia_a = dia_b = dia_len = 0;
        snap_a = snap_b = 0;
        trans_ops.clear();
    }

    up_tbl[i][0] = Pi;
    dep_tbl[i] = dep_tbl[Pi] + 1;
    for (int k = 1; k < LOG; k++) {
        up_tbl[i][k] = up_tbl[up_tbl[i][k - 1]][k - 1];
    }

    int op = 0;
    int da = dist_node(i, dia_a);
    int db = dist_node(i, dia_b);

    if (da > dia_len) {
        int old_a = dia_a;
        dia_a = i;
        dia_b = old_a;
        dia_len = da;
        op = 1;
    } else if (db > dia_len) {
        int old_b = dia_b;
        dia_a = i;
        dia_b = old_b;
        dia_len = db;
        op = 2;
    }

    if (i == T) {
        snap_a = dia_a;
        snap_b = dia_b;
        return snap_a + 1;
    }

    if (i > T) {
        trans_ops.push_back(op);
    }

    if (i == T + 1) {
        return snap_b + 1;
    }

    if (i == N - 1) {
        int code = 0;
        int p = 1;
        for (int x : trans_ops) {
            code += x * p;
            p *= 3;
        }
        return code + 1;
    }

    return 0;
}

pair<int, int> longest_path(vector<int> S) {
    int N = (int)S.size();
    int T = N - 9;

    int a = S[T] - 1;
    int b = S[T + 1] - 1;
    int code = S[N - 1] - 1;

    for (int j = T + 1; j <= N - 1; j++) {
        int op = code % 3;
        code /= 3;

        if (op == 1) {
            int old_a = a;
            a = j;
            b = old_a;
        } else if (op == 2) {
            int old_b = b;
            a = j;
            b = old_b;
        }
    }

    return {a, b};
}
