#include <bits/stdc++.h>
using namespace std;

namespace {
    constexpr int MAXN = 10005;
    constexpr int LOG = 15;

    int par_arr[MAXN];
    int dep_arr[MAXN];
    int up_arr[LOG][MAXN];
    int upd_arr[MAXN];

    int curN = 0;
    int startPos = 0;
    int diaA = 0, diaB = 0, diaD = 0;
    int savedA = 0, savedB = 0;

    int lca_node(int a, int b) {
        if (dep_arr[a] < dep_arr[b]) swap(a, b);
        int diff = dep_arr[a] - dep_arr[b];
        for (int k = 0; k < LOG; ++k) {
            if (diff & (1 << k)) a = up_arr[k][a];
        }
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; --k) {
            if (up_arr[k][a] != up_arr[k][b]) {
                a = up_arr[k][a];
                b = up_arr[k][b];
            }
        }
        return up_arr[0][a];
    }

    int dist_node(int a, int b) {
        int c = lca_node(a, b);
        return dep_arr[a] + dep_arr[b] - 2 * dep_arr[c];
    }
}

int send_message(int N, int i, int Pi) {
    if (i == 1 || N != curN) {
        curN = N;
        startPos = N - 50;
        diaA = diaB = diaD = 0;
        savedA = savedB = 0;
        par_arr[0] = 0;
        dep_arr[0] = 0;
        for (int k = 0; k < LOG; ++k) up_arr[k][0] = 0;
    }

    par_arr[i] = Pi;
    dep_arr[i] = dep_arr[Pi] + 1;
    up_arr[0][i] = Pi;
    for (int k = 1; k < LOG; ++k) {
        up_arr[k][i] = up_arr[k - 1][up_arr[k - 1][i]];
    }

    int code = 0;
    int da = dist_node(i, diaA);
    int db = dist_node(i, diaB);

    if (da > diaD) {
        diaD = da;
        diaB = i;
        code = 1;
    } else if (db > diaD) {
        diaD = db;
        diaA = i;
        code = 2;
    }

    upd_arr[i] = code;

    if (i == startPos) {
        savedA = diaA;
        savedB = diaB;
        return savedA + 1;
    }

    if (i == startPos + 1) {
        return savedB + 1;
    }

    if (i >= startPos + 2 && i <= N - 2) {
        return upd_arr[i - 1] + 1;
    }

    if (i == N - 1) {
        return upd_arr[N - 2] * 3 + upd_arr[N - 1] + 1;
    }

    return 0;
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = (int)S.size();
    int L = N - 50;

    int A = S[L] - 1;
    int B = S[L + 1] - 1;

    auto apply_update = [&](int code, int v) {
        if (code == 1) {
            B = v;
        } else if (code == 2) {
            A = v;
        }
    };

    for (int v = L + 1; v <= N - 3; ++v) {
        int code = S[v + 1] - 1;
        apply_update(code, v);
    }

    int packed = S[N - 1] - 1;
    int code1 = packed / 3;
    int code2 = packed % 3;

    apply_update(code1, N - 2);
    apply_update(code2, N - 1);

    return {A, B};
}
