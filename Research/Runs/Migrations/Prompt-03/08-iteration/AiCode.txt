#include <bits/stdc++.h>
using namespace std;

static const int LOG = 20;

static int curN = -1;
static vector<array<int, LOG>> up_tbl;
static vector<int> dep;
static int diaA = 0, diaB = 0, diaD = 0;

static void reset_state(int N) {
    curN = N;
    up_tbl.assign(N, {});
    dep.assign(N, 0);
    for (int k = 0; k < LOG; ++k) up_tbl[0][k] = 0;
    diaA = diaB = diaD = 0;
}

static int lca_node(int a, int b) {
    if (dep[a] < dep[b]) swap(a, b);
    int diff = dep[a] - dep[b];
    for (int k = 0; k < LOG; ++k) {
        if (diff & (1 << k)) a = up_tbl[a][k];
    }
    if (a == b) return a;
    for (int k = LOG - 1; k >= 0; --k) {
        if (up_tbl[a][k] != up_tbl[b][k]) {
            a = up_tbl[a][k];
            b = up_tbl[b][k];
        }
    }
    return up_tbl[a][0];
}

static int dist_node(int a, int b) {
    int c = lca_node(a, b);
    return dep[a] + dep[b] - 2 * dep[c];
}

int send_message(int N, int i, int Pi) {
    if (curN != N || i == 1) reset_state(N);

    up_tbl[i][0] = Pi;
    dep[i] = dep[Pi] + 1;
    for (int k = 1; k < LOG; ++k) {
        up_tbl[i][k] = up_tbl[up_tbl[i][k - 1]][k - 1];
    }

    int oldA = diaA, oldB = diaB, oldD = diaD;
    int da = dist_node(i, oldA);
    int db = dist_node(i, oldB);

    bool changed = false;
    int partner = -1;

    if (da > oldD || db > oldD) {
        changed = true;
        if (da >= db) {
            diaA = i;
            diaB = oldA;
            diaD = da;
            partner = oldA;
        } else {
            diaA = i;
            diaB = oldB;
            diaD = db;
            partner = oldB;
        }
    }

    if (i == N - 2) {
        return diaA + 1;
    }

    if (i == N - 1) {
        if (changed) return partner + 10001;
        return oldB + 1;
    }

    return 0;
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = (int)S.size();
    if (N <= 1) return {0, 0};

    int last = S[N - 1];

    if (last > 10000) {
        return {N - 1, last - 10001};
    }

    return {S[N - 2] - 1, last - 1};
}
