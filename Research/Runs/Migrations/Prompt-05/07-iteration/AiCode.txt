#include <vector>
#include <utility>
#include <algorithm>

namespace {
constexpr int MAXN = 10000;
constexpr int LOG = 16;

int up[LOG][MAXN];
int depth_arr[MAXN];
int dia_a = 0, dia_b = 0, dia_d = 0;

void reset_sender() {
    dia_a = 0;
    dia_b = 0;
    dia_d = 0;
    depth_arr[0] = 0;
    for (int k = 0; k < LOG; ++k) up[k][0] = 0;
}

int lca_node(int a, int b) {
    if (depth_arr[a] < depth_arr[b]) std::swap(a, b);
    int diff = depth_arr[a] - depth_arr[b];
    for (int k = 0; k < LOG; ++k) {
        if ((diff >> k) & 1) a = up[k][a];
    }
    if (a == b) return a;
    for (int k = LOG - 1; k >= 0; --k) {
        if (up[k][a] != up[k][b]) {
            a = up[k][a];
            b = up[k][b];
        }
    }
    return up[0][a];
}

int dist_nodes(int a, int b) {
    int c = lca_node(a, b);
    return depth_arr[a] + depth_arr[b] - 2 * depth_arr[c];
}

int rank_excluding(int y, int x) {
    return y < x ? y : y - 1;
}

int unrank_excluding(int r, int x) {
    return r < x ? r : r + 1;
}
}

int send_message(int N, int i, int Pi) {
    if (i == 1) reset_sender();

    up[0][i] = Pi;
    depth_arr[i] = depth_arr[Pi] + 1;
    for (int k = 1; k < LOG; ++k) {
        up[k][i] = up[k - 1][up[k - 1][i]];
    }

    int start = N - 50;
    int old_a = dia_a;
    int old_b = dia_b;
    int old_d = dia_d;

    int change = 0;
    int da = dist_nodes(i, old_a);
    int db = dist_nodes(i, old_b);

    if (da > old_d) {
        dia_b = i;
        dia_d = da;
        change = 3;
    } else if (db > old_d) {
        dia_a = i;
        dia_d = db;
        change = 2;
    }

    if (i < start) return 0;

    if (i == start) {
        return dia_a + 1;
    }

    if (i == start + 1) {
        int x = old_a;
        int y = old_b;
        if (change == 3) {
            return 19999;
        }
        int r = rank_excluding(y, x);
        if (change == 2) {
            return 10000 + r;
        }
        return 1 + r;
    }

    if (change == 2) return 2;
    if (change == 3) return 3;
    return 1;
}

std::pair<int, int> longest_path(std::vector<int> S) {
    int N = static_cast<int>(S.size());
    int start = N - 50;

    if (N < 52 || start < 1 || S[start] == 0 || S[start + 1] == 0) {
        return {0, 0};
    }

    int a = S[start] - 1;
    int b = 0;

    int code = S[start + 1];
    if (code == 19999) {
        b = start + 1;
    } else if (code >= 1 && code <= 9999) {
        b = unrank_excluding(code - 1, a);
    } else {
        b = unrank_excluding(code - 10000, a);
        a = start + 1;
    }

    for (int i = start + 2; i < N; ++i) {
        if (S[i] == 2) {
            a = i;
        } else if (S[i] == 3) {
            b = i;
        }
    }

    return {a, b};
}
